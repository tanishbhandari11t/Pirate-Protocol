"use client";

import { useRouter } from "next/navigation";
import { useState, type FormEvent } from "react";
import { CrewBanner } from "@/components/crew/CrewBanner";
import { CrewFormShell } from "@/components/crew/CrewFormShell";
import { IdentityPreview } from "@/components/crew/IdentityPreview";
import { FlagIcon, SkullIcon } from "@/components/icons";
import { Button } from "@/components/ui/Button";
import { LoadingScreen } from "@/components/ui/Loader";
import { useNotify } from "@/components/ui/Notifications";
import { Divider } from "@/components/ui/Ornaments";
import { ParchmentCard } from "@/components/ui/ParchmentCard";
import { TextField } from "@/components/ui/TextField";
import { Tooltip } from "@/components/ui/Tooltip";
import { avatarFor } from "@/lib/avatars";
import { useCrew } from "@/lib/crew/CrewProvider";
import { randomCrewName } from "@/lib/names";
import { CAPABILITIES } from "@/lib/socket/backend";
import { CREW_NAME_MAX, USERNAME_MAX } from "@/lib/socket/contract";
import { describeError } from "@/lib/socket/errors";
import { useProfile } from "@/lib/storage";
import { normalizeName, typeUsername, validateCrewName, validateUsername } from "@/lib/validation";

export default function CreateCrewPage() {
  const router = useRouter();
  const { createCrew } = useCrew();
  const { notify } = useNotify();
  const profile = useProfile();

  const [nameDraft, setNameDraft] = useState<string | null>(null);
  const [crewName, setCrewName] = useState("");
  const [submitted, setSubmitted] = useState(false);
  const [pending, setPending] = useState(false);
  const [serverError, setServerError] = useState<string | null>(null);

  const username = nameDraft ?? profile?.username ?? "";

  const nameError = submitted ? validateUsername(username) : null;
  const crewError = submitted ? validateCrewName(crewName) : null;

  const onSubmit = async (e: FormEvent) => {
    e.preventDefault();
    setSubmitted(true);
    setServerError(null);
    if (validateUsername(username) || validateCrewName(crewName)) return;

    setPending(true);
    const res = await createCrew({ username, crewName: normalizeName(crewName) });

    if (res.ok) {
      router.push(`/lobby/${res.data.code}`);
      return;
    }
    setPending(false);
    const message = describeError(res.error);
    setServerError(message);
    notify({ tone: "danger", title: "The colours would not rise", message });
  };

  return (
    <>
      <CrewFormShell
        form={
          <ParchmentCard eyebrow="Articles of Agreement" title="Raise Your Colours">
            <form onSubmit={onSubmit} noValidate className="space-y-5 short:space-y-3">
              <TextField
                label="Your pirate name"
                placeholder="calico_jack"
                value={username}
                onChange={(e) => setNameDraft(typeUsername(e.target.value))}
                maxLength={USERNAME_MAX}
                showCount
                error={nameError}
                hint="Letters, numbers and underscores. The sea picks your likeness."
                autoComplete="nickname"
                autoCapitalize="none"
                spellCheck={false}
                autoFocus
              />
              <TextField
                label="Name of your ship"
                placeholder="The Salted Kraken"
                value={crewName}
                onChange={(e) => setCrewName(e.target.value)}
                maxLength={CREW_NAME_MAX}
                showCount
                error={crewError}
                hint="Your crew sails under this banner."
                icon={
                  <Tooltip content="Let fate name your ship">
                    <button
                      type="button"
                      onClick={() => setCrewName(randomCrewName())}
                      className="rounded p-1 transition hover:rotate-12 hover:text-blood focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ink"
                      aria-label="Random ship name"
                    >
                      <SkullIcon size={18} />
                    </button>
                  </Tooltip>
                }
              />
              <Divider className="pt-2 text-ink-soft/50!" />
              {serverError && (
                <p role="alert" className="text-center font-body italic text-blood">
                  {serverError}
                </p>
              )}
              <Button type="submit" size="lg" fullWidth loading={pending} icon={<FlagIcon size={18} />}>
                Create Crew
              </Button>
            </form>
          </ParchmentCard>
        }
        preview={
          <IdentityPreview name={username} avatarId={avatarFor(username)} role="Captain">
            <CrewBanner name={normalizeName(crewName)} />
          </IdentityPreview>
        }
      />
      <LoadingScreen show={pending} message="Raising the colours" detail="Registering your ship with the harbour master…" />
    </>
  );
}
