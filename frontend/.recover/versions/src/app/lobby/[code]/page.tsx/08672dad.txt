"use client";

import { AnimatePresence, motion } from "framer-motion";
import { useParams, useRouter } from "next/navigation";
import { useEffect, useRef, useState, useSyncExternalStore } from "react";
import { CrewChat } from "@/components/crew/CrewChat";
import { TopBar } from "@/components/layout/TopBar";
import { ArticlesPanel } from "@/components/lobby/ArticlesPanel";
import { CaptainOrders } from "@/components/lobby/CaptainOrders";
import { CrewMemberCard, EmptyBerth } from "@/components/lobby/CrewMemberCard";
import { HarbourCharter } from "@/components/lobby/HarbourCharter";
import { RoomCodePlaque } from "@/components/lobby/RoomCodePlaque";
import { ShipsLog } from "@/components/lobby/ShipsLog";
import { VoyageCountdown } from "@/components/lobby/VoyageCountdown";
import { UsersIcon } from "@/components/icons";
import { Button } from "@/components/ui/Button";
import { LoadingScreen } from "@/components/ui/Loader";
import { Modal } from "@/components/ui/Modal";
import { useNotify } from "@/components/ui/Notifications";
import { useCrew } from "@/lib/crew/CrewProvider";
import { useShanty } from "@/lib/music";
import { CAPABILITIES } from "@/lib/socket/backend";
import { describeError } from "@/lib/socket/errors";
import { seatStore } from "@/lib/storage";
import { sanitizeRoomCode } from "@/lib/validation";

const EASE = [0.22, 1, 0.36, 1] as const;

export default function LobbyPage() {
  const params = useParams<{ code: string }>();
  const code = sanitizeRoomCode(params.code ?? "");
  const router = useRouter();
  const { notify } = useNotify();
  const { state, me, isCaptain, rejoinCrew, leaveCrew, setReady, startVoyage } = useCrew();
  useShanty(state.room?.phase === "lobby" || state.room?.phase === "starting" ? "harbour" : null);
  const seat = useSyncExternalStore(seatStore.subscribe, seatStore.getSnapshot, seatStore.getServerSnapshot);

  const [readyPending, setReadyPending] = useState(false);
  const [startPending, setStartPending] = useState(false);
  const [confirmLeave, setConfirmLeave] = useState(false);
  const [leaving, setLeaving] = useState(false);
  const rejoinAttempted = useRef(false);

  const room = state.room?.code === code ? state.room : null;
  const hasSeat = seat?.roomCode === code;

  useEffect(() => {
    if (room || leaving) return;
    // Hydration renders without storage, so ask the store itself before giving up on the seat.
    if (!hasSeat && seatStore.get()?.roomCode !== code) {
      router.replace(`/crew/join?code=${code}`);
      return;
    }
    if (rejoinAttempted.current) return;
    rejoinAttempted.current = true;
    rejoinCrew(code).then((res) => {
      if (!res.ok) {
        notify({ tone: "danger", title: "Could not reclaim your berth", message: describeError(res.error.code) });
        router.replace(`/crew/join?code=${code}`);
      }
    });
  }, [room, hasSeat, code, leaving, rejoinCrew, router, notify]);

  useEffect(() => {
    if (room && (state.voyageStarted || room.phase === "in-game")) {
      router.push(`/voyage/${room.code}`);
    }
  }, [room, state.voyageStarted, router]);

  if (!room || !me) {
    return <LoadingScreen show message="Rowing back aboard" detail="Reclaiming your berth…" />;
  }

  const toggleReady = async () => {
    setReadyPending(true);
    const res = await setReady(!me.isReady);
    setReadyPending(false);
    if (!res.ok) notify({ tone: "danger", title: "Order refused", message: describeError(res.error.code) });
  };

  const start = async () => {
    setStartPending(true);
    const res = await startVoyage();
    setStartPending(false);
    if (!res.ok) notify({ tone: "danger", title: "The ship won't budge", message: describeError(res.error.code) });
  };

  const leave = async () => {
    setLeaving(true);
    await leaveCrew();
    router.push("/");
  };

  const emptyBerths = Math.max(0, room.maxPlayers - room.players.length);

  return (
    <main className="flex min-h-dvh flex-col lg:h-dvh lg:overflow-hidden">
      <TopBar />

      <div className="mx-auto grid w-full max-w-7xl flex-1 gap-6 px-4 pb-8 pt-1 sm:px-6 lg:min-h-0 lg:grid-cols-[1fr_20rem] lg:gap-8 lg:pb-6 xl:grid-cols-[1fr_22rem] xl:gap-12">
        <section className="lg:-ml-2 lg:min-h-0 lg:overflow-y-auto lg:p-2 lg:pr-3">
          <motion.header
            initial={{ opacity: 0, y: 20 }}
            animate={{ opacity: 1, y: 0 }}
            transition={{ duration: 0.9, ease: EASE }}
            className="mb-6 short:mb-4"
          >
            <p className="font-ui text-[0.6rem] uppercase tracking-[0.35em] text-brass/80 sm:text-[0.65rem] sm:tracking-[0.45em]">
              The crew manifest of
            </p>
            <h1 className="mt-2 break-words font-display text-[clamp(2.25rem,min(7vw,9vh),4.5rem)] leading-none text-gilded drop-shadow-[0_6px_16px_rgba(0,0,0,0.9)] short:mt-1">
              {room.crewName}
            </h1>
            <div className="mt-3 flex flex-wrap items-center gap-x-4 gap-y-1 font-body text-base italic text-parchment/70 sm:text-lg short:mt-2">
              <span className="inline-flex items-center gap-2">
                <UsersIcon size={18} className="text-brass" />
                {room.players.length} of {room.maxPlayers} hands aboard
              </span>
              <span className="h-1 w-1 rounded-full bg-brass/50" />
              <span>{room.phase === "lobby" ? "Moored in harbour" : "Preparing to sail"}</span>
            </div>
          </motion.header>

          <div className="grid grid-cols-2 gap-3 sm:gap-5 md:grid-cols-3 short:gap-3">
            <AnimatePresence mode="popLayout">
              {room.players.map((player) => (
                <CrewMemberCard key={player.id} player={player} isMe={player.id === me.id} />
              ))}
              {Array.from({ length: emptyBerths }, (_, i) => (
                <EmptyBerth
                  key={`berth-${room.players.length + i}`}
                  index={room.players.length + i}
                  className={i > 0 ? "max-md:hidden" : undefined}
                />
              ))}
            </AnimatePresence>
          </div>
        </section>

        <motion.aside
          initial={{ opacity: 0, x: 30 }}
          animate={{ opacity: 1, x: 0 }}
          transition={{ duration: 0.9, delay: 0.2, ease: EASE }}
          className="space-y-5 lg:min-h-0 lg:overflow-y-auto lg:pb-1 lg:pr-1 lg:pt-1 short:space-y-3"
        >
          <RoomCodePlaque code={room.code} />
          <CaptainOrders
            room={room}
            me={me}
            isCaptain={isCaptain}
            readyPending={readyPending}
            startPending={startPending}
            onToggleReady={toggleReady}
            onStart={start}
            onLeave={() => setConfirmLeave(true)}
          />
          <HarbourCharter settings={room.settings} />
          {CAPABILITIES.articles && <ArticlesPanel />}
          {CAPABILITIES.chat && <CrewChat />}
          <ShipsLog entries={state.log} defaultFilter="crew" />
        </motion.aside>
      </div>

      <Modal
        open={confirmLeave}
        onClose={() => setConfirmLeave(false)}
        eyebrow="A grave decision"
        title="Abandon Ship?"
        dismissible={!leaving}
        footer={
          <>
            <Button variant="parchment" onClick={() => setConfirmLeave(false)} disabled={leaving} data-autofocus>
              Stay aboard
            </Button>
            <Button variant="danger" onClick={leave} loading={leaving}>
              Abandon ship
            </Button>
          </>
        }
      >
        <p className="text-center">
          {isCaptain && room.players.length > 1
            ? "The helm will pass to another sailor, and your berth will be given away."
            : "Your berth will be given away. You can board again with the ship's code."}
        </p>
      </Modal>

      <AnimatePresence>
        {state.countdown && (
          <VoyageCountdown
            key="countdown"
            startsAt={state.countdown.startsAt}
            seconds={state.countdown.seconds}
            crewName={room.crewName}
          />
        )}
      </AnimatePresence>
    </main>
  );
}
