"use client";

import { AnimatePresence, motion, useReducedMotion } from "framer-motion";
import { cn } from "@/lib/cn";
import type { Mood, Reaction } from "@/lib/game/reactions";
import type { AvatarId, Outfit } from "@/lib/socket/contract";
import { AvatarFrame } from "./AvatarFrame";
import { PirateAvatar } from "./PirateAvatar";

interface LivingAvatarProps {
  avatarId: AvatarId;
  outfit?: Outfit | null;
  reaction?: Reaction | null;
  faded?: boolean;
  /** Draw the earned frame; off in tight spots like chat lines. */
  framed?: boolean;
  /** Bob gently while idle. */
  idle?: boolean;
  className?: string;
  title?: string;
}

const MOOD_BADGE: Record<Mood, { glyph: string; tone: string; label: string }> = {
  cheer: { glyph: "★", tone: "bg-brass text-ink", label: "cheering" },
  shock: { glyph: "!", tone: "bg-blood text-parchment", label: "startled" },
  grim: { glyph: "…", tone: "bg-ink text-parchment", label: "grim" },
  sly: { glyph: "♪", tone: "bg-teal text-parchment", label: "pleased with themselves" },
};

const MOOD_MOTION: Record<Mood, { y?: number[]; rotate?: number[]; scale?: number[]; x?: number[] }> = {
  cheer: { y: [0, -7, 0, -4, 0], scale: [1, 1.06, 1, 1.03, 1] },
  shock: { x: [0, -3, 3, -2, 2, 0], scale: [1, 1.08, 1] },
  grim: { y: [0, 2, 2, 0], rotate: [0, -4, -4, 0] },
  sly: { rotate: [0, 5, -3, 0] },
};

/**
 * A portrait that breathes, blinks and reacts. Reactions come from server-recorded events, so a
 * cheer means the log really shows a solve.
 */
export function LivingAvatar({
  avatarId,
  outfit,
  reaction,
  faded,
  framed = true,
  idle = true,
  className,
  title,
}: LivingAvatarProps) {
  const reduced = useReducedMotion();
  const mood = reaction?.mood ?? null;
  const badge = mood ? MOOD_BADGE[mood] : null;

  return (
    <div className={cn("relative", className)}>
      <motion.div
        key={reaction?.eventId ?? "idle"}
        className="h-full w-full"
        animate={!reduced && mood ? MOOD_MOTION[mood] : undefined}
        transition={{ duration: 0.7, ease: "easeOut" }}
      >
        <AvatarFrame frame={framed ? outfit?.frame : null} className="h-full w-full">
          <PirateAvatar
            avatarId={avatarId}
            outfit={outfit}
            expression={mood ?? "neutral"}
            animated={idle && !faded}
            faded={faded}
            title={title}
            className="h-full w-full"
          />
        </AvatarFrame>
      </motion.div>
      <AnimatePresence>
        {badge && (
          <motion.span
            key={reaction?.eventId}
            role="status"
            aria-label={`${title ?? "Sailor"} is ${badge.label}`}
            initial={{ opacity: 0, scale: 0.4, y: 6 }}
            animate={{ opacity: 1, scale: 1, y: 0 }}
            exit={{ opacity: 0, scale: 0.6, y: -6 }}
            transition={{ type: "spring", stiffness: 420, damping: 18 }}
            className={cn(
              "absolute -right-1 -top-1 grid h-5 w-5 place-items-center rounded-full border border-ink/40 text-[11px] font-bold leading-none shadow",
              badge.tone,
            )}
          >
            {badge.glyph}
          </motion.span>
        )}
      </AnimatePresence>
    </div>
  );
}
