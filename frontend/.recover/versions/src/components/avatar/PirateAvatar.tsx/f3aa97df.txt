import { useId } from "react";
import { AVATARS, type AvatarLook } from "@/lib/avatars";
import type { AvatarId } from "@/lib/socket/contract";
import { cn } from "@/lib/cn";

interface PirateAvatarProps {
  avatarId: AvatarId;
  className?: string;
  /** Desaturates the portrait, e.g. for disconnected sailors. */
  faded?: boolean;
  title?: string;
}

export function PirateAvatar({ avatarId, className, faded, title }: PirateAvatarProps) {
  const uid = useId().replace(/:/g, "");
  const look = AVATARS[avatarId]?.look ?? AVATARS.captain.look;

  return (
    <svg
      viewBox="0 0 120 120"
      role="img"
      aria-label={title ?? AVATARS[avatarId]?.title ?? "Pirate"}
      className={cn("block", faded && "grayscale opacity-60", className)}
    >
      <defs>
        <radialGradient id={`bg-${uid}`} cx="50%" cy="35%" r="75%">
          <stop offset="0%" stopColor={look.backdrop[0]} />
          <stop offset="100%" stopColor={look.backdrop[1]} />
        </radialGradient>
        <radialGradient id={`face-${uid}`} cx="45%" cy="40%" r="65%">
          <stop offset="0%" stopColor={look.skin} />
          <stop offset="100%" stopColor={look.skinShade} />
        </radialGradient>
        <clipPath id={`clip-${uid}`}>
          <circle cx="60" cy="60" r="60" />
        </clipPath>
      </defs>

      <g clipPath={`url(#clip-${uid})`}>
        <rect width="120" height="120" fill={`url(#bg-${uid})`} />
        <circle cx="60" cy="40" r="46" fill="#fff" opacity="0.04" />
        <HairBack look={look} />
        <Body look={look} />
        <Head look={look} faceFill={`url(#face-${uid})`} />
        <Hat look={look} />
      </g>
    </svg>
  );
}

function Body({ look }: { look: AvatarLook }) {
  return (
    <g>
      <rect x="51" y="72" width="18" height="16" rx="4" fill={look.skinShade} />
      <path d="M10 124 C14 96 36 86 60 86 C84 86 106 96 110 124 Z" fill={look.coat} />
      <path d="M48 87 L60 106 L72 87 Z" fill="#eadcbc" />
      <path d="M48 87 L60 106 L52 110 L40 90 Z" fill={look.coat} opacity="0.9" />
      <path d="M72 87 L60 106 L68 110 L80 90 Z" fill={look.coat} opacity="0.9" />
      <path
        d="M40 90 L52 110 M80 90 L68 110"
        stroke={look.coatTrim}
        strokeWidth="1.6"
        fill="none"
      />
      <circle cx="44" cy="104" r="1.8" fill={look.coatTrim} />
      <circle cx="76" cy="104" r="1.8" fill={look.coatTrim} />
      <circle cx="46" cy="114" r="1.8" fill={look.coatTrim} />
      <circle cx="74" cy="114" r="1.8" fill={look.coatTrim} />
    </g>
  );
}

function HairBack({ look }: { look: AvatarLook }) {
  if (look.hair === "long") {
    return (
      <path
        d="M34 48 C28 78 32 98 42 104 L78 104 C88 98 92 78 86 48 Z"
        fill={look.hairColor}
      />
    );
  }
  if (look.hair === "braids") {
    return (
      <g fill={look.hairColor}>
        <path d="M35 50 C30 70 32 86 38 96 L46 94 C42 80 42 66 44 52 Z" />
        <path d="M85 50 C90 70 88 86 82 96 L74 94 C78 80 78 66 76 52 Z" />
        <circle cx="40" cy="98" r="3.5" />
        <circle cx="80" cy="98" r="3.5" />
      </g>
    );
  }
  return null;
}

function Head({ look, faceFill }: { look: AvatarLook; faceFill: string }) {
  const eye = look.eyeColor ?? "#1a1008";
  return (
    <g>
      <ellipse cx="38.5" cy="58" rx="4" ry="6" fill={look.skinShade} />
      <ellipse cx="81.5" cy="58" rx="4" ry="6" fill={look.skinShade} />
      <ellipse cx="60" cy="56" rx="22" ry="25" fill={faceFill} />

      {look.hair === "short" && look.hat === "none" && (
        <path
          d="M38 52 C38 32 48 28 60 28 C72 28 82 32 82 52 C76 42 68 38 60 38 C52 38 44 42 38 52 Z"
          fill={look.hairColor}
        />
      )}

      {look.wrinkles && (
        <g stroke={look.skinShade} strokeWidth="1" fill="none" strokeLinecap="round">
          <path d="M50 42 Q60 39 70 42" />
          <path d="M52 46 Q60 44 68 46" />
          <path d="M43 58 L46 60 M77 58 L74 60" />
        </g>
      )}

      {/* brows */}
      <g stroke={look.beard === "none" ? "#2a1a10" : look.beardColor} strokeWidth="2.2" strokeLinecap="round">
        <path d="M45 49 Q51 46 56 49" fill="none" />
        <path d="M64 49 Q69 46 75 49" fill="none" />
      </g>

      {/* eyes */}
      {look.eyeColor ? (
        <g>
          <ellipse cx="51" cy="56" rx="3.2" ry="2.4" fill={eye} opacity="0.95" />
          <ellipse cx="69" cy="56" rx="3.2" ry="2.4" fill={eye} opacity="0.95" />
          <ellipse cx="51" cy="56" rx="6" ry="4" fill={eye} opacity="0.2" />
          <ellipse cx="69" cy="56" rx="6" ry="4" fill={eye} opacity="0.2" />
        </g>
      ) : (
        <g fill={eye}>
          <ellipse cx="51" cy="56" rx="2.4" ry="2.6" />
          <ellipse cx="69" cy="56" rx="2.4" ry="2.6" />
          <circle cx="51.8" cy="55.2" r="0.7" fill="#fff" opacity="0.7" />
          <circle cx="69.8" cy="55.2" r="0.7" fill="#fff" opacity="0.7" />
        </g>
      )}

      {look.warpaint && (
        <g stroke={look.eyeColor ?? "#7fc295"} strokeWidth="1.4" strokeLinecap="round" opacity="0.85">
          <path d="M44 62 L50 64 M44 65 L49 67" />
          <path d="M76 62 L70 64 M76 65 L71 67" />
          <path d="M60 36 L60 44" />
        </g>
      )}

      {look.freckles && (
        <g fill={look.skinShade} opacity="0.9">
          <circle cx="47" cy="63" r="0.9" />
          <circle cx="50" cy="65" r="0.9" />
          <circle cx="45" cy="66" r="0.8" />
          <circle cx="73" cy="63" r="0.9" />
          <circle cx="70" cy="65" r="0.9" />
          <circle cx="75" cy="66" r="0.8" />
        </g>
      )}

      {/* nose */}
      <path
        d="M60 57 Q64 65 58.5 67"
        stroke={look.skinShade}
        strokeWidth="1.8"
        fill="none"
        strokeLinecap="round"
      />

      {look.scar && (
        <path
          d="M70 60 L76 71 M71 64 L74 63 M73 68 L76 67"
          stroke="#9b4a40"
          strokeWidth="1.3"
          strokeLinecap="round"
        />
      )}

      <Beard look={look} />

      {look.eyepatch && (
        <g>
          <path d="M37 47 Q60 41 83 53" stroke="#0d0906" strokeWidth="2.2" fill="none" />
          <path d="M44 52 Q51 49 58 52 L57 60 Q51 64 45 60 Z" fill="#0d0906" />
        </g>
      )}

      {look.earring && (
        <circle cx="82" cy="66" r="3" fill="none" stroke="#f1d68d" strokeWidth="1.6" />
      )}
    </g>
  );
}

function Beard({ look }: { look: AvatarLook }) {
  const mouth = (
    <path
      d="M53 73 Q60 77 67 73"
      stroke="#3a1a10"
      strokeWidth="1.8"
      fill="none"
      strokeLinecap="round"
    />
  );
  const mustache = (
    <path
      d="M49 71 Q54 66 60 69 Q66 66 71 71 Q66 72.5 60 71.5 Q54 72.5 49 71 Z"
      fill={look.beardColor}
    />
  );

  switch (look.beard) {
    case "full":
      return (
        <g>
          <path
            d="M37 58 C38 86 49 98 60 98 C71 98 82 86 83 58 C80 70 74 76 60 76 C46 76 40 70 37 58 Z"
            fill={look.beardColor}
          />
          <path d="M55 75 Q60 78 65 75" stroke="#2a120a" strokeWidth="1.6" fill="none" />
          {mustache}
        </g>
      );
    case "goatee":
      return (
        <g>
          {mouth}
          <path d="M54 77 Q60 92 66 77 Q60 80 54 77 Z" fill={look.beardColor} />
          {mustache}
        </g>
      );
    case "mustache":
      return (
        <g>
          {mouth}
          {mustache}
        </g>
      );
    case "stubble":
      return (
        <g>
          <path
            d="M40 64 C42 82 50 90 60 90 C70 90 78 82 80 64 C74 74 68 78 60 78 C52 78 46 74 40 64 Z"
            fill={look.beardColor}
            opacity="0.28"
          />
          {mouth}
        </g>
      );
    default:
      return mouth;
  }
}

function Hat({ look }: { look: AvatarLook }) {
  switch (look.hat) {
    case "bicorne":
      return (
        <g>
          <path
            d="M12 42 C28 16 46 10 60 10 C74 10 92 16 108 42 C92 36 76 38 60 43 C44 38 28 36 12 42 Z"
            fill={look.hatColor}
          />
          <path
            d="M12 42 C28 36 44 38 60 43 C76 38 92 36 108 42"
            stroke={look.hatTrim}
            strokeWidth="2.2"
            fill="none"
          />
          <g transform="translate(60 27)">
            <circle r="6" fill="#efe3c5" />
            <circle cx="-2.2" cy="-0.8" r="1.4" fill={look.hatColor} />
            <circle cx="2.2" cy="-0.8" r="1.4" fill={look.hatColor} />

      return (

    case "hood":
      return (
        <g>

            d="M26 110 C20 60 30 18 60 16 C90 18 100 60 94 110 L84 110 C88 70 82 36 60 34 C38 36 32 70 36 110 Z"
            fill={look.hatColor}
          />
          <path
            d="M36 110 C32 70 38 36 60 34 C82 36 88 70 84 110"
            stroke={look.hatTrim}
            strokeWidth="1.4"
            fill="none"
            opacity="0.7"
          />
        </g>
      );
    case "knit":
      return (
        <g>
          <path d="M37 44 C37 26 48 19 60 19 C72 19 83 26 83 44 Z" fill={look.hatColor} />
          <rect x="35" y="39" width="50" height="9" rx="4" fill={look.hatTrim} />
          <g stroke={look.hatTrim} strokeWidth="1" opacity="0.6">
            <path d="M46 24 L46 39 M54 21 L54 39 M62 20 L62 39 M70 22 L70 39 M77 27 L77 39" />
          </g>
          <circle cx="60" cy="18" r="4" fill={look.hatTrim} />
        </g>
      );
    default:
      return null;
  }
}
