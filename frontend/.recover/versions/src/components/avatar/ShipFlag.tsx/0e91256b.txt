"use client";

import { motion, useReducedMotion } from "framer-motion";
import { useId } from "react";
import { cn } from "@/lib/cn";
import { FLAG_COLORS } from "@/lib/palettes";
import type { FlagEmblem, ShipFlag as Flag } from "@/lib/socket/contract";

/*
 * A ship's colours, drawn on a 60×40 cloth. `ShipFlag` is a standalone, waving <svg>;
 * `FlagCloth` is the bare <g> for embedding in other drawings, such as ships on the chart.
 */

const W = 60;
const H = 40;

/** The cloth's outline at two points in its ripple. The fly end (right) moves most. */
const CLOTH = [
  `M0 0 C15 -2 30 3 45 0 C50 -1 55 1 ${W} 0 L${W} ${H} C55 ${H + 1} 50 ${H - 1} 45 ${H} C30 ${H + 3} 15 ${H - 2} 0 ${H} Z`,
  `M0 0 C15 3 30 -2 45 2 C50 3 55 -1 ${W} 2 L${W} ${H + 2} C55 ${H - 1} 50 ${H + 3} 45 ${H + 2} C30 ${H - 2} 15 ${H + 3} 0 ${H} Z`,
];

/** Ragged fly edge for tattered flags; drawn in the field colour of the sea behind it. */
const TATTER = `M${W} 0 L${W - 4} 5 L${W} 9 L${W - 6} 15 L${W} 19 L${W - 3} 25 L${W} 29 L${W - 7} 34 L${W} ${H}`;

export function EmblemPath({ emblem, ink, field }: { emblem: FlagEmblem; ink: string; field: string }) {
  switch (emblem) {
    case "skull":
      return (
        <g>
          <path d="M-5 -3 l10 9 M5 -3 l-10 9" stroke={ink} strokeWidth="2.2" strokeLinecap="round" />
          <path d="M-6.5 -8 C-6.5 -15 6.5 -15 6.5 -8 C6.5 -4 4 -3.5 4 -1.5 L-4 -1.5 C-4 -3.5 -6.5 -4 -6.5 -8 Z" fill={ink} />
          <circle cx="-2.6" cy="-8" r="1.7" fill={field} />
          <circle cx="2.6" cy="-8" r="1.7" fill={field} />
          <path d="M-2 -1.5 v1.8 M0 -1.5 v1.8 M2 -1.5 v1.8" stroke={field} strokeWidth="0.8" />
        </g>
      );
    case "crossed-swords":
      return (
        <g stroke={ink} strokeWidth="1.8" strokeLinecap="round" fill="none">
          <path d="M-9 -9 L7 7 M9 -9 L-7 7" />
          <path d="M3 9 l6 -6 M-3 9 l-6 -6" strokeWidth="2.4" />
          <path d="M-9 -9 l-1.5 -1.5 M9 -9 l1.5 -1.5" strokeWidth="1.2" />
        </g>
      );
    case "anchor":
      return (
        <g stroke={ink} strokeWidth="1.9" fill="none" strokeLinecap="round">
          <circle cx="0" cy="-9" r="2.2" />
          <path d="M0 -6.8 V9 M-5 -3 H5" />
          <path d="M-8 3 C-7 8 -3 10 0 10 C3 10 7 8 8 3" />
          <path d="M-8 3 l-2 2.5 M8 3 l2 2.5" />
        </g>
      );
    case "compass":
      return (
        <g>
          <circle r="9" fill="none" stroke={ink} strokeWidth="1.2" />
          <path d="M0 -11 L2.2 -2.2 L11 0 L2.2 2.2 L0 11 L-2.2 2.2 L-11 0 L-2.2 -2.2 Z" fill={ink} />
          <path d="M0 -11 L2.2 -2.2 L0 0 Z M11 0 L2.2 2.2 L0 0 Z M0 11 L-2.2 2.2 L0 0 Z M-11 0 L-2.2 -2.2 L0 0 Z" fill={field} opacity="0.45" />
        </g>
      );
    case "kraken":
      return (
        <g fill={ink}>
          <path d="M-6 -4 C-6 -13 6 -13 6 -4 C6 -1 3 1 0 1 C-3 1 -6 -1 -6 -4 Z" />
          <circle cx="-2.3" cy="-5" r="1.3" fill={field} />
          <circle cx="2.3" cy="-5" r="1.3" fill={field} />
          {[-1, 1].map((s) => (
            <g key={s}>
              <path d={`M${s * 4} 0 C${s * 6} 5 ${s * 11} 4 ${s * 10} 9 C${s * 9} 6 ${s * 6} 7 ${s * 3} 1 Z`} />
              <path d={`M${s * 1.5} 1 C${s * 2} 7 ${s * 5} 10 ${s * 3} 12 C${s * 1} 9 ${s * 0} 6 0 1 Z`} />
            </g>
          ))}
        </g>
      );
    case "hourglass":
      return (
        <g stroke={ink} strokeWidth="1.6" fill="none" strokeLinejoin="round">
          <path d="M-6 -10 H6 M-6 10 H6" strokeWidth="2.2" strokeLinecap="round" />
          <path d="M-5 -10 C-5 -3 -1 -2 -1 0 C-1 2 -5 3 -5 10 M5 -10 C5 -3 1 -2 1 0 C1 2 5 3 5 10" />
          <path d="M-3 -6 H3 L0 -1 Z M-3.5 9 C-3 6 3 6 3.5 9 Z" fill={ink} stroke="none" />
        </g>
      );
    case "serpent":
      return (
        <g fill="none" stroke={ink} strokeWidth="2.4" strokeLinecap="round">
          <path d="M-10 8 C-6 2 -2 10 2 4 C6 -2 2 -8 -2 -6 C-6 -4 -2 2 3 -1 C7 -3 9 -8 8 -11" />
          <circle cx="8.4" cy="-11.5" r="1.8" fill={ink} stroke="none" />
          <path d="M10 -12 l2 -0.5 M10 -11 l2 0.8" strokeWidth="0.7" />
        </g>
      );
    case "crown":
      return (
        <g fill={ink}>
          <path d="M-10 6 L-11 -6 L-5 -1 L0 -10 L5 -1 L11 -6 L10 6 Z" />
          <rect x="-10" y="6" width="20" height="3.5" rx="1" />
          <circle cx="0" cy="2" r="1.6" fill={field} />
        </g>
      );
  }
}

interface ClothProps {
  flag: Flag;
  /** Ripple the cloth; off for print and for reduced motion. */
  waving?: boolean;
  /** Stroke for the edge, so small flags still read on parchment. */
  edge?: string;
}

/** The flag as an SVG group occupying (0,0)–(60,40). */
export function FlagCloth({ flag, waving = true, edge = "#2b1b0d" }: ClothProps) {
  const uid = useId().replace(/:/g, "");
  const reduced = useReducedMotion();
  const colors = FLAG_COLORS[flag.field];
  const animate = waving && !reduced;

  return (
    <g>
      <defs>
        <clipPath id={`${uid}-cloth`}>
          <path d={CLOTH[0]} />
        </clipPath>
        <linearGradient id={`${uid}-fold`} x1="0" x2="1">
          <stop offset="0%" stopColor="#000" stopOpacity="0" />
          <stop offset="35%" stopColor="#000" stopOpacity="0.18" />
          <stop offset="55%" stopColor="#fff" stopOpacity="0.1" />
          <stop offset="80%" stopColor="#000" stopOpacity="0.22" />
          <stop offset="100%" stopColor="#000" stopOpacity="0.05" />
        </linearGradient>
      </defs>
      <motion.path
        d={CLOTH[0]}
        fill={colors.field}
        stroke={edge}
        strokeWidth="1"
        animate={animate ? { d: CLOTH } : undefined}
        transition={animate ? { duration: 1.6, repeat: Infinity, repeatType: "mirror", ease: "easeInOut" } : undefined}
      />
      <g clipPath={`url(#${uid}-cloth)`}>
        {flag.border === "gilded" && (
          <rect x="1.5" y="1.5" width={W - 3} height={H - 3} fill="none" stroke="#e0b23c" strokeWidth="2" strokeDasharray="4 1.5" />
        )}
        <g transform={`translate(${W / 2} ${H / 2 + 1})`}>
          <EmblemPath emblem={flag.emblem} ink={colors.ink} field={colors.field} />
        </g>
        <rect width={W} height={H} fill={`url(#${uid}-fold)`} pointerEvents="none" />
      </g>
      {flag.border === "tattered" && (
        <g fill="none" stroke={edge} strokeWidth="1">
          <path d={TATTER} />
          <path d="M24 0 l2 4 M38 40 l-1.5 -4 M46 0 l1 3" />
        </g>
      )}
    </g>
  );
}

interface ShipFlagProps extends ClothProps {
  className?: string;
  /** Draw the mast the flag hangs from. */
  pole?: boolean;
  title?: string;
}

export function ShipFlag({ flag, waving, edge, className, pole = true, title }: ShipFlagProps) {
  return (
    <svg viewBox={pole ? "-8 -6 72 54" : "-2 -3 64 46"} className={cn("block overflow-visible", className)} role="img" aria-label={title ?? "Ship's flag"}>
      {pole && (
        <g>
          <rect x="-5" y="-4" width="3" height="52" rx="1.5" fill="#5a3a1c" stroke="#2b1b0d" strokeWidth="0.8" />
          <circle cx="-3.5" cy="-4.5" r="2.4" fill="#e0b23c" stroke="#7a5414" strokeWidth="0.6" />
        </g>
      )}
      <FlagCloth flag={flag} waving={waving} edge={edge} />
    </svg>
  );
}
