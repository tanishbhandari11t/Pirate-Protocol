"use client";

import { AnimatePresence, motion } from "framer-motion";
import { useMemo, useState, type ReactNode } from "react";
import { cn } from "@/lib/cn";
import {
  isUnlocked,
  pieceIn,
  SLOTS,
  UNLOCKS,
  WARDROBE_SLOTS,
  wardrobeProgress,
  wearing,
  type UnlockRule,
  type WardrobeSlot,
} from "@/lib/cosmetics";
import { careerTotals, useCareer, type CareerTotals } from "@/lib/game/career";
import type { RequestResult } from "@/lib/socket/client";
import { describeError } from "@/lib/socket/errors";
import { outfitsEqual, type AvatarId, type Outfit } from "@/lib/socket/contract";
import { CheckIcon, LockIcon } from "../icons";
import { Button } from "../ui/Button";
import { Modal } from "../ui/Modal";
import { AvatarFrame } from "./AvatarFrame";
import { PirateAvatar } from "./PirateAvatar";
import { ShipFlag } from "./ShipFlag";

interface WardrobeProps {
  open: boolean;
  onClose: () => void;
  avatarId: AvatarId;
  outfit: Outfit;
  onSave: (outfit: Outfit) => Promise<RequestResult<unknown>>;
}

const GROUPS: { label: string; slots: WardrobeSlot[] }[] = [
  { label: "Sailor", slots: ["hat", "coat", "face", "trinket", "frame"] },
  { label: "Ship's colours", slots: ["field", "emblem", "border"] },
];

const FLAG_SLOTS = new Set<WardrobeSlot>(["field", "emblem", "border"]);

/**
 * The sea chest of cosmetics. Pieces unlock from finished voyages recorded on this device; every
 * piece is cosmetic and changes nothing about how the voyage is scored.
 */
export function Wardrobe(props: WardrobeProps) {
  return (
    <Modal open={props.open} onClose={props.onClose} eyebrow="Sea chest" title="Wardrobe" size="lg">
      {props.open && <WardrobeBody {...props} />}
    </Modal>
  );
}

function WardrobeBody({ onClose, avatarId, outfit, onSave }: WardrobeProps) {
  const career = useCareer();
  const totals = useMemo(() => careerTotals(career), [career]);
  const progress = useMemo(() => wardrobeProgress(totals), [totals]);
  const [draft, setDraft] = useState<Outfit>(outfit);
  const [slot, setSlot] = useState<WardrobeSlot>("hat");
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const changed = !outfitsEqual(draft, outfit);
  const info = SLOTS[slot];

  const save = async () => {
    setSaving(true);
    setError(null);
    const result = await onSave(draft);
    setSaving(false);
    if (result.ok) onClose();
    else setError(describeError(result.error.code, result.error.message));
  };

  return (
    <div className="grid gap-6 md:grid-cols-[13rem_1fr]">
      <aside className="flex flex-col items-center gap-3">
        <AvatarFrame frame={draft.frame} className="aspect-square w-40">
          <PirateAvatar avatarId={avatarId} outfit={draft} animated className="h-full w-full bg-sea" />
        </AvatarFrame>
        <ShipFlag flag={draft.flag} className="w-28" title="Your ship's flag" />
        <p className="text-center font-ui text-[0.65rem] font-bold uppercase tracking-[0.3em] text-ink-soft/80">
          {progress.open} of {progress.total} pieces earned
        </p>
        <div className="h-1.5 w-full overflow-hidden rounded-full bg-ink/15" aria-hidden>
          <div className="h-full bg-brass" style={{ width: `${(progress.open / progress.total) * 100}%` }} />
        </div>
        <CareerLine totals={totals} />
      </aside>

      <section className="min-w-0">
        <div role="tablist" aria-label="Wardrobe slots" className="space-y-2">
          {GROUPS.map((group) => (
            <div key={group.label}>
              <p className="mb-1 font-ui text-[0.6rem] font-bold uppercase tracking-[0.3em] text-ink-soft/70">{group.label}</p>
              <div className="flex flex-wrap gap-1.5">
                {group.slots.map((s) => (
                  <button
                    key={s}
                    type="button"
                    role="tab"
                    aria-selected={slot === s}
                    onClick={() => setSlot(s)}
                    className={cn(
                      "rounded-sm border px-2.5 py-1 font-ui text-[0.68rem] font-bold uppercase tracking-[0.15em] transition",
                      slot === s
                        ? "border-ink bg-ink text-parchment-light"
                        : "border-ink/30 text-ink-soft hover:border-ink/60 hover:text-ink",
                    )}
                  >
                    {SLOTS[s].label}
                  </button>
                ))}
              </div>
            </div>
          ))}
        </div>

        <AnimatePresence mode="wait">
          <motion.ul
            key={slot}
            role="tabpanel"
            aria-label={info.label}
            initial={{ opacity: 0, y: 8 }}
            animate={{ opacity: 1, y: 0 }}
            exit={{ opacity: 0, y: -8 }}
            transition={{ duration: 0.18 }}
            className="mt-4 grid grid-cols-2 gap-2 sm:grid-cols-3"
          >
            {info.optional && (
              <PieceCard
                name="None"
                flavour="Your own likeness, unadorned."
                selected={pieceIn(draft, slot) === null}
                onPick={() => setDraft(wearing(draft, slot, null))}
              >
                <PirateAvatar avatarId={avatarId} outfit={wearing(draft, slot, null)} className="h-full w-full" />
              </PieceCard>
            )}
            {info.ids.map((id) => {
              const piece = info.catalogue[id];
              const unlocked = isUnlocked(slot, id, totals);
              const rule: UnlockRule = UNLOCKS[piece.unlock];
              const tried = wearing(draft, slot, id);
              return (
                <PieceCard
                  key={id}
                  name={piece.name}
                  flavour={piece.flavour}
                  selected={pieceIn(draft, slot) === id}
                  locked={unlocked ? null : { requirement: rule.requirement, progress: rule.progress?.(totals) ?? null }}
                  onPick={() => setDraft(tried)}
                >
                  {FLAG_SLOTS.has(slot) ? (
                    <ShipFlag flag={tried.flag} pole={false} waving={false} className="h-full w-full p-2" />
                  ) : slot === "frame" ? (
                    <AvatarFrame frame={tried.frame} className="h-full w-full">
                      <PirateAvatar avatarId={avatarId} outfit={tried} className="h-full w-full bg-sea" />
                    </AvatarFrame>
                  ) : (
                    <PirateAvatar avatarId={avatarId} outfit={tried} className="h-full w-full" />
                  )}
                </PieceCard>
              );
            })}
          </motion.ul>
        </AnimatePresence>

        {error && (
          <p role="alert" className="mt-4 text-center text-base text-blood">
            {error}
          </p>
        )}

        <div className="mt-6 flex flex-wrap justify-end gap-3">
          <Button variant="parchment" size="sm" onClick={() => setDraft(outfit)} disabled={!changed || saving}>
            Undo changes
          </Button>
          <Button size="sm" onClick={save} loading={saving} disabled={!changed}>
            Wear it
          </Button>
        </div>
      </section>
    </div>
  );
}

interface PieceCardProps {
  name: string;
  flavour: string;
  selected: boolean;
  locked?: { requirement: string; progress: [number, number] | null } | null;
  onPick: () => void;
  children: ReactNode;
}

function PieceCard({ name, flavour, selected, locked, onPick, children }: PieceCardProps) {
  return (
    <li>
      <button
        type="button"
        onClick={locked ? undefined : onPick}
        aria-disabled={locked ? true : undefined}
        aria-pressed={selected}
        title={locked ? locked.requirement : flavour}
        className={cn(
          "group relative flex w-full flex-col items-center gap-1.5 rounded-md border p-2 text-center transition",
          selected ? "border-ink bg-ink/10 shadow-[inset_0_0_0_1px_rgba(43,27,13,0.6)]" : "border-ink/20 hover:border-ink/50",
          locked && "cursor-not-allowed",
        )}
      >
        <span className={cn("relative block aspect-square w-16 overflow-hidden rounded-full bg-sea/90", locked && "opacity-45 grayscale")}>
          {children}
        </span>
        <span className="font-display text-base leading-tight text-ink">{name}</span>
        {locked ? (
          <span className="flex w-full flex-col items-center gap-1 text-xs leading-snug text-ink-soft">
            <span className="inline-flex items-center gap-1">
              <LockIcon size={12} />
              {locked.requirement}
            </span>
            {locked.progress && (
              <span className="flex w-full items-center gap-1.5" aria-label={`${locked.progress[0]} of ${locked.progress[1]}`}>
                <span className="h-1 flex-1 overflow-hidden rounded-full bg-ink/15">
                  <span
                    className="block h-full bg-brass-dark"
                    style={{ width: `${(locked.progress[0] / locked.progress[1]) * 100}%` }}
                  />
                </span>
                <span className="font-ui text-[0.6rem] tabular-nums">
                  {locked.progress[0]}/{locked.progress[1]}
                </span>
              </span>
            )}
          </span>
        ) : (
          <span className="line-clamp-2 text-xs italic leading-snug text-ink-soft/80">{flavour}</span>
        )}
        {selected && (
          <span className="absolute right-1.5 top-1.5 grid h-5 w-5 place-items-center rounded-full bg-ink text-parchment-light">
            <CheckIcon size={12} />
          </span>
        )}
      </button>
    </li>
  );
}

function CareerLine({ totals }: { totals: CareerTotals }) {
  if (totals.voyages === 0) {
    return (
      <p className="text-center text-sm italic leading-snug text-ink-soft/80">
        Finish voyages to earn new pieces. Every one is cosmetic.
      </p>
    );
  }
  return (
    <dl className="grid w-full grid-cols-3 gap-1 text-center">
      {[
        ["Voyages", totals.voyages],
        ["Vaults", totals.wins],
        ["Honours", totals.honours.size],
      ].map(([label, value]) => (
        <div key={label}>
          <dt className="font-ui text-[0.55rem] font-bold uppercase tracking-[0.2em] text-ink-soft/70">{label}</dt>
          <dd className="font-display text-xl text-ink">{value}</dd>
        </div>
      ))}
    </dl>
  );
}
