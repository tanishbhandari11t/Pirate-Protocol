"use client";

import { AnimatePresence, motion } from "framer-motion";
import { useEffect, useMemo, useRef, useState } from "react";
import { createPortal } from "react-dom";
import { CHART_H, CHART_W, chartPoint, islandPath, lanePath } from "@/lib/game/chart";
import type { Destination } from "@/lib/game/heading";
import { VAULT_KEY } from "@/lib/game/selectors";
import { seeded } from "@/lib/random";
import { playSfx } from "@/lib/sound";
import type { PublicIsland } from "@/lib/socket/contract";
import { Galleon } from "../atmosphere/Galleon";
import { CompassReveal } from "./CompassReveal";
import { Letterbox } from "./Letterbox";

/** Beat start times (ms): ocean dive, map unfolds, title and compass, hand back to the game. */
const BEATS = { dive: 0, unfold: 2_300, title: 4_700, end: 9_200 } as const;
const EASE = [0.22, 1, 0.36, 1] as const;
const PANELS = [0, 1, 2] as const;

/** Session key so the opening plays once per voyage per tab. */
export function setSailKey(roomCode: string, startedAt: number) {
  return `pp:set-sail:${roomCode}:${startedAt}`;
}

export function setSailSeen(key: string) {
  try {
    return window.sessionStorage.getItem(key) === "1";
  } catch {
    return true;
  }
}

interface SetSailCinematicProps {
  seenKey: string;
  crewName: string;
  islands: PublicIsland[];
  charted: ReadonlySet<string>;
  destination: Destination | null;
  onDone: () => void;
}

/**
 * The opening of a voyage: the camera plunges from the sky into the ocean, the chart unfolds out
 * of the deep, and a compass spins up to point the way. Plays once; any key, click or Skip ends it.
 */
export function SetSailCinematic({ seenKey, crewName, islands, charted, destination, onDone }: SetSailCinematicProps) {
  const [beat, setBeat] = useState<keyof typeof BEATS>("dive");
  const onDoneRef = useRef(onDone);

  useEffect(() => {
    onDoneRef.current = onDone;
  });

  useEffect(() => {
    try {
      window.sessionStorage.setItem(seenKey, "1");
    } catch {
      /* private mode: the opening may replay, which is harmless */
    }
    const splash = window.setTimeout(() => playSfx("splash"), 1_500);
    const sail = window.setTimeout(() => playSfx("sail"), BEATS.unfold + 300);
    const timers = (["unfold", "title"] as const).map((b) => window.setTimeout(() => setBeat(b), BEATS[b]));
    const end = window.setTimeout(() => onDoneRef.current(), BEATS.end);
    const onKey = (e: KeyboardEvent) => {
      if (e.key === "Escape" || e.key === "Enter" || e.key === " ") onDoneRef.current();
    };
    window.addEventListener("keydown", onKey);
    return () => {
      [splash, sail, end, ...timers].forEach((t) => window.clearTimeout(t));
      window.removeEventListener("keydown", onKey);
    };
  }, [seenKey]);

  return createPortal(
    <Letterbox label="The voyage begins" onSkip={onDone}>
      <AnimatePresence>
        {beat === "dive" && <OceanDive key="dive" />}
      </AnimatePresence>
      {beat !== "dive" && <ChartUnfold islands={islands} charted={charted} dimmed={beat === "title"} />}
      <AnimatePresence>
        {beat === "title" && (
          <motion.div
            key="title"
            className="absolute inset-0 z-10 flex flex-col items-center justify-center gap-6 px-4 text-center"
            initial={{ opacity: 0 }}
            animate={{ opacity: 1 }}
            exit={{ opacity: 0 }}
          >
            <motion.div initial={{ opacity: 0, y: 16, letterSpacing: "0.6em" }} animate={{ opacity: 1, y: 0, letterSpacing: "0.04em" }} transition={{ duration: 1.2, ease: EASE }}>
              <p className="font-ui text-[0.65rem] uppercase tracking-[0.45em] text-brass-light/80">The voyage of</p>
              <h2 className="mt-1 font-display text-[clamp(2.6rem,8vw,5rem)] leading-none text-gilded drop-shadow-[0_6px_18px_rgba(0,0,0,0.9)]">
                {crewName}
              </h2>
            </motion.div>
            <CompassReveal destination={destination} spins={4} />
          </motion.div>
        )}
      </AnimatePresence>
    </Letterbox>,
    document.body,
  );
}

const BUBBLES = (() => {
  const rand = seeded(8);
  return Array.from({ length: 22 }, () => ({ x: rand() * 100, size: 4 + rand() * 14, delay: 1.2 + rand() * 0.9, dur: 1 + rand() * 0.8 }));
})();

/** From the sky above the ship, down through the surface and into the blue. */
function OceanDive() {
  return (
    <motion.div className="absolute inset-0" exit={{ opacity: 0, transition: { duration: 0.6 } }} aria-hidden>
      <motion.div
        className="absolute inset-0 origin-[50%_60%]"
        initial={{ scale: 1, y: "0%" }}
        animate={{ scale: 2.6, y: "-48%" }}
        transition={{ duration: 2.1, ease: [0.6, 0, 0.9, 0.4] }}
      >
        <div className="absolute inset-0 bg-[linear-gradient(180deg,#0d1430_0%,#3a2f55_30%,#e08a5a_58%,#1f5566_58.5%,#0c2231_80%,#03070c_100%)]" />
        <div className="absolute left-1/2 top-[44%] h-24 w-24 -translate-x-1/2 rounded-full bg-[radial-gradient(circle,#fff1c1,#ffb36a_45%,transparent_70%)] opacity-80" />
        <Galleon className="absolute bottom-[40%] left-[38%] h-32 w-40" />
        <svg className="absolute inset-x-0 top-[58%] h-[42%] w-full" preserveAspectRatio="none" viewBox="0 0 100 40">
          {Array.from({ length: 9 }, (_, i) => (
            <path key={i} d={`M0 ${2 + i * 4} q12.5 -1.5 25 0 t25 0 t25 0 t25 0`} stroke="#9cc4cc" strokeOpacity={0.25 - i * 0.02} strokeWidth="0.3" fill="none" />
          ))}
        </svg>
      </motion.div>
      <motion.div
        className="absolute inset-0 bg-[linear-gradient(180deg,#0e3a4d_0%,#07131d_100%)]"
        initial={{ opacity: 0 }}
        animate={{ opacity: [0, 0, 1] }}
        transition={{ duration: 2.2, times: [0, 0.6, 1] }}
      />
      <motion.div className="absolute inset-0" initial={{ opacity: 0 }} animate={{ opacity: [0, 1, 1] }} transition={{ duration: 2.2, times: [0, 0.55, 1] }}>
        {BUBBLES.map((b, i) => (
          <motion.span
            key={i}
            className="absolute bottom-0 rounded-full border border-foam/60 bg-foam/10"
            style={{ left: `${b.x}%`, width: b.size, height: b.size }}
            initial={{ y: 0, opacity: 0 }}
            animate={{ y: "-110vh", opacity: [0, 0.9, 0] }}
            transition={{ duration: b.dur, delay: b.delay, ease: "easeOut" }}
          />
        ))}
        <div className="absolute inset-0 bg-[repeating-linear-gradient(100deg,transparent_0_40px,rgba(156,196,204,0.06)_40px_70px)]" />
      </motion.div>
    </motion.div>
  );
}

/** The chart, folded in three, opening out of the deep: side panels swing open, then it tilts flat. */
function ChartUnfold({ islands, charted, dimmed }: { islands: PublicIsland[]; charted: ReadonlySet<string>; dimmed: boolean }) {
  const ordered = useMemo(() => [...islands].filter((i) => i.key !== VAULT_KEY).sort((a, b) => a.order - b.order), [islands]);
  const art = (
    <svg viewBox={`0 0 ${CHART_W} ${CHART_H}`} className="h-full w-full" aria-hidden>
      <rect width={CHART_W} height={CHART_H} fill="#e6d2a0" />
      <rect x="14" y="14" width={CHART_W - 28} height={CHART_H - 28} fill="none" stroke="#3a2412" strokeOpacity="0.55" strokeWidth="2" />
      {ordered.slice(1).map((island, i) => (
        <path
          key={island.key}
          d={lanePath(chartPoint(ordered[i]), chartPoint(island), i % 2 ? 0.16 : -0.16)}
          stroke="#3a2412"
          strokeOpacity="0.2"
          strokeWidth="2"
          strokeDasharray="2 10"
          fill="none"
        />
      ))}
      {islands.map((island) => {
        const p = chartPoint(island);
        const known = charted.has(island.key);
        if (island.key === VAULT_KEY) return null;
        return (
          <g key={island.key} transform={`translate(${p.x} ${p.y})`}>
            <path d={islandPath(island.key, 30)} fill={known ? "#d8bd84" : "#e2cd9c"} stroke="#3a2412" strokeWidth={known ? 2 : 1.4} strokeDasharray={known ? undefined : "5 5"} opacity={known ? 1 : 0.55} />
            <text y={known ? 58 : 10} textAnchor="middle" fontSize={known ? 18 : 28} fill="#3a2412" opacity={known ? 1 : 0.5} style={{ fontFamily: "var(--font-pirata)" }}>
              {known ? island.name : "?"}
            </text>
          </g>
        );
      })}
      <text x="48" y="66" fill="#3a2412" opacity="0.85" fontSize="26" style={{ fontFamily: "var(--font-pirata)" }}>
        The Protocol Sea
      </text>
    </svg>
  );

  return (
    <motion.div
      className="absolute inset-0 flex items-center justify-center"
      style={{ perspective: 1400 }}
      animate={{ opacity: dimmed ? 0.35 : 1, filter: dimmed ? "blur(2px)" : "blur(0px)" }}
      transition={{ duration: 0.9 }}
      aria-hidden
    >
      <motion.div
        className="preserve-3d relative aspect-[5/3] w-[min(86vw,130vh)]"
        initial={{ rotateX: 62, scale: 0.55, y: "30vh" }}
        animate={{ rotateX: 0, scale: 1, y: "0vh" }}
        transition={{ duration: 1.5, ease: EASE }}
      >
        {PANELS.map((i) => (
          <motion.div
            key={i}
            className="backface-hidden absolute top-0 h-full overflow-hidden shadow-[0_20px_60px_rgba(0,0,0,0.7)]"
            style={{ left: `${(i * 100) / 3}%`, width: `${100 / 3}%`, transformOrigin: i === 0 ? "right center" : i === 2 ? "left center" : "center" }}
            initial={{ rotateY: i === 0 ? 178 : i === 2 ? -178 : 0 }}
            animate={{ rotateY: 0 }}
            transition={{ duration: 1.1, delay: 0.35 + (i === 1 ? 0 : 0.25), ease: EASE }}
          >
            <div className="absolute top-0 h-full" style={{ width: "300%", left: `${-i * 100}%` }}>
              {art}
            </div>
            {i !== 1 && (
              <motion.div
                className="absolute inset-0"
                style={{ background: `linear-gradient(${i === 0 ? 90 : 270}deg, transparent 60%, rgba(43,27,13,0.45))` }}
                initial={{ opacity: 1 }}
                animate={{ opacity: 0.2 }}
                transition={{ duration: 1.2, delay: 0.6 }}
              />
            )}
          </motion.div>
        ))}
        <motion.div
          className="pointer-events-none absolute inset-y-0 left-1/3 w-px bg-ink/40"
          initial={{ opacity: 0.8 }}
          animate={{ opacity: 0.25 }}
          transition={{ duration: 1.5, delay: 1 }}
        />
        <motion.div
          className="pointer-events-none absolute inset-y-0 left-2/3 w-px bg-ink/40"
          initial={{ opacity: 0.8 }}
          animate={{ opacity: 0.25 }}
          transition={{ duration: 1.5, delay: 1 }}
        />
      </motion.div>
    </motion.div>
  );
}
