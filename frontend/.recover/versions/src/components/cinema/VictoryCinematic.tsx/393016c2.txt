"use client";

import { AnimatePresence, motion, useReducedMotion } from "framer-motion";
import { useEffect, useRef, useState } from "react";
import { createPortal } from "react-dom";
import { cn } from "@/lib/cn";
import { findPlayer } from "@/lib/game/selectors";
import { playSfx, vibrate } from "@/lib/sound";
import type { FinishReason, PublicPlayer, RoomState } from "@/lib/socket/contract";
import { LivingAvatar } from "../avatar/LivingAvatar";
import { CoinBurst } from "../fx/CoinBurst";
import { SkullIcon } from "../icons";
import { Button } from "../ui/Button";
import { Letterbox } from "./Letterbox";

/** Beats (ms): crew name, lineup rises, winner steps forward, call to the results. */
const BEATS = [0, 1_500, 3_200, 5_400] as const;
const AUTO_ADVANCE_MS = 11_000;
const EASE = [0.22, 1, 0.36, 1] as const;

const BACKDROP: Record<FinishReason, string> = {
  treasure: "radial-gradient(ellipse at 50% 60%, rgba(255,216,115,0.35), transparent 60%), linear-gradient(180deg,#140d05,#03070c)",
  time: "radial-gradient(ellipse at 50% 70%, rgba(231,129,60,0.3), transparent 60%), linear-gradient(180deg,#0d1430,#03070c)",
  wreck: "radial-gradient(ellipse at 50% 80%, rgba(31,85,102,0.35), transparent 60%), linear-gradient(180deg,#0b1218,#020406)",
};

const EYEBROW: Record<FinishReason, string> = {
  treasure: "The Vault lies open",
  time: "The tide has turned",
  wreck: "The sea takes its due",
};

interface VictoryCinematicProps {
  room: RoomState;
  meId: string;
  onDone: () => void;
}

/**
 * The curtain call. The crew lines up in rank order and, if the server named a winner, they step
 * forward to be crowned. Nothing here decides who won; it only stages `winnerPlayerId` and `scores`.
 */
export function VictoryCinematic({ room, meId, onDone }: VictoryCinematicProps) {
  const reduced = useReducedMotion();
  const [beat, setBeat] = useState(reduced ? BEATS.length - 1 : 0);
  const onDoneRef = useRef(onDone);
  const reason: FinishReason = room.finishReason ?? "time";
  const winner = findPlayer(room, room.winnerPlayerId);
  const crowned = winner !== null;
  const lineup = rankOrder(room);

  useEffect(() => {
    onDoneRef.current = onDone;
  });

  useEffect(() => {
    const timers = reduced ? [] : BEATS.slice(1).map((at, i) => window.setTimeout(() => setBeat(i + 1), at));
    const fanfare = window.setTimeout(() => {
      if (crowned) {
        playSfx(reason === "treasure" ? "honour" : "bell");
        vibrate([40, 30, 40]);
      } else {
        playSfx("splash");
      }
    }, reduced ? 0 : BEATS[2]);
    const auto = window.setTimeout(() => onDoneRef.current(), AUTO_ADVANCE_MS);
    const onKey = (e: KeyboardEvent) => e.key === "Escape" && onDoneRef.current();
    window.addEventListener("keydown", onKey);
    return () => {
      [...timers, fanfare, auto].forEach((t) => window.clearTimeout(t));
      window.removeEventListener("keydown", onKey);
    };
  }, [reduced, reason, crowned]);

  const headline = winner
    ? reason === "treasure"
      ? `${winner.id === meId ? "You" : winner.username} claimed the hoard`
      : `${winner.id === meId ? "You" : winner.username} takes the day`
    : "No sailor came home";

  return createPortal(
    <Letterbox label="The voyage's end" onSkip={onDone} skipLabel="To the results">
      <div className="absolute inset-0" style={{ background: BACKDROP[reason] }} aria-hidden />
      {reason !== "wreck" && !reduced && <GodRays active={beat >= 2 && !!winner} />}

      <div className="relative z-10 flex h-full flex-col items-center justify-center px-4 text-center">
        <motion.div initial={{ opacity: 0, y: 20, letterSpacing: "0.5em" }} animate={{ opacity: 1, y: 0, letterSpacing: "0.05em" }} transition={{ duration: 1.2, ease: EASE }}>
          <p className="font-ui text-[0.65rem] uppercase tracking-[0.45em] text-brass-light/80">{EYEBROW[reason]}</p>
          <h2 className="mt-1 font-display text-[clamp(2.4rem,7vw,4.5rem)] leading-none text-gilded drop-shadow-[0_6px_18px_rgba(0,0,0,0.9)]">{room.name}</h2>
        </motion.div>

        <div className="relative mt-[6vh] flex items-end justify-center gap-[clamp(0.5rem,2vw,1.5rem)]">
          {lineup.map((p, i) => {
            const isWinner = p.id === winner?.id;
            const sinking = reason === "wreck";
            return (
              <motion.div
                key={p.id}
                className="relative flex flex-col items-center"
                initial={{ opacity: 0, y: 80 }}
                animate={
                  beat < 1
                    ? { opacity: 0, y: 80 }
                    : sinking && beat >= 2
                      ? { opacity: 0.35, y: 30, rotate: i % 2 ? 6 : -6 }
                      : isWinner && beat >= 2
                        ? { opacity: 1, y: -24, scale: 1.35 }
                        : { opacity: beat >= 2 && winner ? 0.6 : 1, y: 0, scale: 1 }
                }
                transition={{ duration: 0.9, delay: beat === 1 ? i * 0.12 : 0, ease: EASE }}
                style={{ zIndex: isWinner ? 2 : 1 }}
              >
                <AnimatePresence>
                  {isWinner && beat >= 2 && (
                    <motion.svg
                      key="crown"
                      viewBox="0 0 40 24"
                      className="absolute -top-7 z-10 w-12 drop-shadow-[0_4px_10px_rgba(255,216,115,0.6)]"
                      initial={{ y: -80, opacity: 0, rotate: -25 }}
                      animate={{ y: 0, opacity: 1, rotate: -6 }}
                      transition={{ type: "spring", stiffness: 200, damping: 12, delay: 0.5 }}
                      aria-hidden
                    >
                      <path d="M3 21 L1 5 L11 12 L20 1 L29 12 L39 5 L37 21 Z" fill="#ffd873" stroke="#7a5414" strokeWidth="1.4" strokeLinejoin="round" />
                      <circle cx="20" cy="14" r="2.6" fill="#c2392f" />
                      <circle cx="10" cy="16" r="1.6" fill="#1f5566" />
                      <circle cx="30" cy="16" r="1.6" fill="#1f5566" />
                    </motion.svg>
                  )}
                </AnimatePresence>
                <LivingAvatar
                  avatarId={p.avatarId}
                  outfit={p.outfit}
                  faded={p.isEliminated}
                  reaction={beat >= 2 ? { playerId: p.id, mood: isWinner ? "cheer" : sinking ? "grim" : "sly", eventId: `curtain-${beat}` } : null}
                  title={p.username}
                  className={cn("aspect-square", isWinner ? "w-[clamp(4.5rem,11vw,7rem)]" : "w-[clamp(3.5rem,8vw,5.5rem)]")}
                />
                <span className={cn("mt-2 max-w-[8rem] truncate font-display text-lg leading-none", isWinner ? "text-gold" : "text-parchment/80")}>
                  {p.id === meId ? "You" : p.username}
                </span>
                <span className="mt-0.5 flex items-center gap-1 font-ui text-[0.55rem] uppercase tracking-[0.2em] text-brass/70">
                  {p.isEliminated ? (
                    <>
                      <SkullIcon size={10} /> Lost
                    </>
                  ) : (
                    scoreOf(room, p.id)
                  )}
                </span>
                {isWinner && beat >= 2 && reason === "treasure" && <CoinBurst count={26} seed={7} spread={220} delay={0.6} />}
              </motion.div>
            );
          })}
        </div>

        <AnimatePresence>
          {beat >= 2 && (
            <motion.p
              key="headline"
              className="mt-[5vh] font-display text-[clamp(1.6rem,4.5vw,2.8rem)] leading-tight text-parchment-light drop-shadow-[0_4px_12px_rgba(0,0,0,0.9)]"
              initial={{ opacity: 0, scale: 1.2 }}
              animate={{ opacity: 1, scale: 1 }}
              transition={{ duration: 0.9, delay: 0.4, ease: EASE }}
            >
              {headline}
            </motion.p>
          )}
        </AnimatePresence>

        <AnimatePresence>
          {beat >= 3 && (
            <motion.div key="cta" initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.6 }} className="mt-6">
              <Button size="lg" onClick={onDone} data-autofocus>
                Count the plunder
              </Button>
            </motion.div>
          )}
        </AnimatePresence>
      </div>
    </Letterbox>,
    document.body,
  );
}

function rankOrder(room: RoomState): PublicPlayer[] {
  const rank = (id: string) => room.scores.find((s) => s.playerId === id)?.rank ?? Number.MAX_SAFE_INTEGER;
  const sorted = [...room.players].sort((a, b) => rank(a.id) - rank(b.id));
  const winnerAt = sorted.findIndex((p) => p.id === room.winnerPlayerId);
  if (winnerAt <= 0 || sorted.length < 3) return sorted;
  const [winner] = sorted.splice(winnerAt, 1);
  sorted.splice(Math.floor(sorted.length / 2), 0, winner);
  return sorted;
}

function scoreOf(room: RoomState, playerId: string) {
  const entry = room.scores.find((s) => s.playerId === playerId);
  return entry ? `${entry.score.toLocaleString()} pts` : "\u00a0";
}

/** Slow golden rays behind the winner; only drawn for the crowning beat. */
function GodRays({ active }: { active: boolean }) {
  return (
    <motion.div
      className="pointer-events-none absolute left-1/2 top-[58%] h-[140vmax] w-[140vmax] -translate-x-1/2 -translate-y-1/2"
      initial={{ opacity: 0 }}
      animate={{ opacity: active ? 0.55 : 0, rotate: 360 }}
      transition={{ opacity: { duration: 1.2 }, rotate: { duration: 60, repeat: Infinity, ease: "linear" } }}
      style={{ background: "repeating-conic-gradient(from 0deg, rgba(255,216,115,0.16) 0deg 6deg, transparent 6deg 18deg)", maskImage: "radial-gradient(circle, black 0%, transparent 55%)" }}
      aria-hidden
    />
  );
}
