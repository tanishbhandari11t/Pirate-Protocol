/** Artwork for a one-tap crew call; unknown calls fall back to a megaphone. */
export function QuickCallIcon({ callId, ...p }: IconProps & { callId: string }) {
  const Icon = QUICK_CALL_ICONS[callId] ?? MegaphoneIcon;
  return <Icon {...p} />;
}

type IconComponent = (p: IconProps) => JSX.Element;

const ITEM_ICONS: Record<string, IconComponent> = {
  compass: CompassIcon,
  spyglass: SpyglassIcon,
  "serpent-key": KeyIcon,
  "widow-chart": ChartIcon,
  "gold-seal": SealIcon,
  "tide-rumor": ScrollIcon,
  "cursed-coin": CoinIcon,
  "treasure-chest": ChestIcon,
};

/** Artwork for an inventory item; unknown items fall back to a scroll. */
export function ItemIcon({ itemKey, ...p }: IconProps & { itemKey: string }) {
  const Icon = ITEM_ICONS[itemKey] ?? ScrollIcon;
  return <Icon {...p} />;
}

/** Artwork for an island marker by its kind. */
export function IslandKindIcon({ kind, ...p }: IconProps & { kind: "NORMAL" | "TRAP" | "TREASURE" }) {
  if (kind === "TRAP") return <SkullIcon {...p} />;
  if (kind === "TREASURE") return <ChestIcon {...p} />;
  return <IslandIcon {...p} />;
}
