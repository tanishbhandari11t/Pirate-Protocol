"use client";

import type { ReactNode } from "react";
import { cn } from "@/lib/cn";
import { CAPABILITIES } from "@/lib/socket/backend";
import type { PlayerSnapshot, RoomSnapshot } from "@/lib/socket/contract";
import { CheckIcon, CloseIcon, DoorIcon, ShipIcon } from "../icons";
import { Button } from "../ui/Button";
import { Divider } from "../ui/Ornaments";
import { Panel } from "../ui/Panel";

interface CaptainOrdersProps {
  room: RoomSnapshot;
  me: PlayerSnapshot;
  isCaptain: boolean;
  readyPending: boolean;
  startPending: boolean;
  onToggleReady: () => void;
  onStart: () => void;
  onLeave: () => void;
}

export function CaptainOrders({
  room,
  me,
  isCaptain,
  readyPending,
  startPending,
  onToggleReady,
  onStart,
  onLeave,
}: CaptainOrdersProps) {
  const enoughHands = room.players.length >= room.minPlayers;
  const readyCount = room.players.filter((p) => p.isReady).length;
  const allReady = readyCount === room.players.length;
  const canStart = isCaptain && enoughHands && allReady && room.phase === "lobby";
  const captain = room.players.find((p) => p.id === room.captainId);

  return (
    <Panel eyebrow={isCaptain ? "You hold the helm" : `Captain ${captain?.name ?? ""}`} title="Orders">
      {CAPABILITIES.readyUp && (
        <Button
          variant={me.isReady ? "ghost" : "parchment"}
          size="lg"
          fullWidth
          loading={readyPending}
          onClick={onToggleReady}
          disabled={room.phase !== "lobby"}
          icon={me.isReady ? <CloseIcon size={16} /> : <CheckIcon size={16} />}
          aria-pressed={me.isReady}
        >
          {me.isReady ? "Stand down" : "I'm ready"}
        </Button>
      )}

      <Divider className={cn("my-5 short:my-3", !CAPABILITIES.readyUp && "mt-0")} label="To set sail" />

      <ul className="space-y-2.5">
        <Requirement met={enoughHands}>
          {`At least ${room.minPlayers} ${room.minPlayers === 1 ? "sailor" : "sailors"} aboard (${room.players.length}/${room.maxPlayers})`}
        </Requirement>
        {CAPABILITIES.readyUp && (
          <Requirement met={allReady}>
            All hands ready ({readyCount}/{room.players.length})
          </Requirement>
        )}
      </ul>

      <div className="mt-6 short:mt-4">
        {isCaptain ? (
          <Button
            size="lg"
            fullWidth
            disabled={!canStart}
            loading={startPending}
            onClick={onStart}
            icon={<ShipIcon size={18} />}
          >
            Start the Voyage
          </Button>
        ) : (
          <p className="rounded border border-brass/20 bg-abyss/40 px-4 py-3 text-center font-body italic text-parchment/60">
            Only the captain can order the ship to sail.
          </p>
        )}
      </div>

      <button
        type="button"
        onClick={onLeave}
        className="mx-auto mt-5 flex items-center gap-2 short:mt-3 font-ui text-[0.62rem] uppercase tracking-[0.25em] text-parchment/50 transition hover:text-blood-light focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-blood-light"
      >
        <DoorIcon size={14} /> Abandon ship
      </button>
    </Panel>
  );
}

function Requirement({ met, children }: { met: boolean; children: ReactNode }) {
  return (
    <li className={cn("flex items-center gap-3 font-body text-[1.02rem]", met ? "text-parchment" : "text-parchment/50")}>
      <span
        className={cn(
          "flex h-5 w-5 shrink-0 items-center justify-center rounded-full border",
          met ? "border-kelp-light bg-kelp/40 text-kelp-light" : "border-parchment/25",
        )}
      >
        {met && <CheckIcon size={12} strokeWidth={3} />}
      </span>
      {children}
    </li>
  );
}
