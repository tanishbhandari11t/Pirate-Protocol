"use client";

import type { ReactNode } from "react";
import { cn } from "@/lib/cn";
import { CAPABILITIES } from "@/lib/socket/backend";
import type { PublicPlayer, RoomState } from "@/lib/socket/contract";
import { CheckIcon, DoorIcon, ShipIcon } from "../icons";
import { Button } from "../ui/Button";
import { Divider } from "../ui/Ornaments";
import { Panel } from "../ui/Panel";

interface CaptainOrdersProps {
  room: RoomState;
  isCaptain: boolean;
  startPending: boolean;
  onStart: () => void;
  onLeave: () => void;
}

export function CaptainOrders({ room, isCaptain, startPending, onStart, onLeave }: CaptainOrdersProps) {
  const captain: PublicPlayer | undefined = room.players.find((p) => p.userId === room.hostId);
  const online = room.players.filter((p) => p.isOnline).length;
  const crewed = room.players.length >= 2;

  return (
    <Panel eyebrow={isCaptain ? "You hold the helm" : `Captain ${captain?.username ?? "ashore"}`} title="Orders">
      <ul className="space-y-2.5">
        <Requirement met={crewed}>
          {crewed ? "A crew worth the name" : "Share the code — the hunt is better with company"} ({room.players.length}/
          {room.maxPlayers})
        </Requirement>
        <Requirement met={online === room.players.length}>
          Hands on deck ({online}/{room.players.length} with lanterns lit)
        </Requirement>
      </ul>

      <Divider className="my-5 short:my-3" label="To set sail" />

      {isCaptain ? (
        <>
          <Button
            size="lg"
            fullWidth
            loading={startPending}
            disabled={room.status !== "LOBBY"}
            onClick={onStart}
            icon={<ShipIcon size={18} />}
          >
            Set Sail
          </Button>
          <p className="mt-3 text-center font-body text-sm italic text-parchment/50">
            Late sailors may still board mid-voyage with the code.
          </p>
        </>
      ) : (
        <p className="rounded border border-brass/20 bg-abyss/40 px-4 py-3 text-center font-body italic text-parchment/60">
          Stand by. When the captain sets sail, the map will awaken for the whole crew.
        </p>
      )}

      <button
        type="button"
        onClick={onLeave}
        className="mx-auto mt-5 flex items-center gap-2 font-ui text-[0.62rem] uppercase tracking-[0.25em] text-parchment/50 transition hover:text-blood-light focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-blood-light short:mt-3"
      >
        <DoorIcon size={14} /> Abandon ship
      </button>
    </Panel>
  );
}

function Requirement({ met, children }: { met: boolean; children: ReactNode }) {
  return (
    <li className={cn("flex items-center gap-3 font-body text-[1.02rem]", met ? "text-parchment" : "text-parchment/50")}>
      <span
        className={cn(
          "flex h-5 w-5 shrink-0 items-center justify-center rounded-full border",
          met ? "border-kelp-light bg-kelp/40 text-kelp-light" : "border-parchment/25",
        )}
      >
        {met && <CheckIcon size={12} strokeWidth={3} />}
      </span>
      {children}
    </li>
  );
}
