"use client";

import { AnimatePresence, motion } from "framer-motion";
import { useEffect, useRef, type ReactNode } from "react";
import { cn } from "@/lib/cn";
import type { LogEntry, LogTone } from "@/lib/crew/reducer";
import {
  AnchorIcon,
  ArrowDownIcon,
  BinocularsIcon,
  ChestIcon,
  CheckIcon,
  ClockIcon,
  CloseIcon,
  CompassIcon,
  CrownIcon,
  DoorIcon,
  FogIcon,
  InfoIcon,
  KeyIcon,
  LightbulbIcon,
  ShipIcon,
  SkullIcon,
  SlidersIcon,
  SwordsIcon,
} from "../icons";
import { Panel } from "../ui/Panel";

const TONE_ICON: Record<LogTone, { icon: ReactNode; color: string }> = {
  info: { icon: <InfoIcon size={13} />, color: "text-foam" },
  join: { icon: <AnchorIcon size={13} />, color: "text-kelp-light" },
  leave: { icon: <DoorIcon size={13} />, color: "text-ember" },
  ready: { icon: <CheckIcon size={13} />, color: "text-gold" },
  captain: { icon: <CrownIcon size={13} />, color: "text-brass-light" },
  voyage: { icon: <ShipIcon size={13} />, color: "text-blood-light" },
  move: { icon: <CompassIcon size={13} />, color: "text-foam" },
  discover: { icon: <FogIcon size={13} />, color: "text-foam" },
  solve: { icon: <CheckIcon size={13} />, color: "text-kelp-light" },
  fail: { icon: <CloseIcon size={13} />, color: "text-ember" },
  trap: { icon: <SkullIcon size={13} />, color: "text-blood-light" },
  item: { icon: <KeyIcon size={13} />, color: "text-gold" },
  trade: { icon: <SwordsIcon size={13} />, color: "text-brass-light" },
  treasure: { icon: <ChestIcon size={13} />, color: "text-gold" },
  hint: { icon: <LightbulbIcon size={13} />, color: "text-gold" },
  settings: { icon: <SlidersIcon size={13} />, color: "text-brass-light" },
  watch: { icon: <BinocularsIcon size={13} />, color: "text-foam" },
  clock: { icon: <ClockIcon size={13} />, color: "text-ember" },
};

export type LogFilter = "all" | "crew" | "voyage";

const CREW_TONES: ReadonlySet<LogTone> = new Set(["join", "leave", "ready", "captain", "settings", "info"]);

const FILTERS: { id: LogFilter; label: string }[] = [
  { id: "all", label: "All" },
  { id: "crew", label: "Crew" },
  { id: "voyage", label: "Voyage" },
];

const EMPTY_TEXT: Record<LogFilter, string> = {
  all: "The page is yet blank…",
  crew: "No word from the crew yet…",
  voyage: "The voyage has yet to begin…",
};

function matches(filter: LogFilter, tone: LogTone) {
  if (filter === "all") return true;
  return filter === "crew" ? CREW_TONES.has(tone) : !CREW_TONES.has(tone);
}

/** How close to the bottom (px) still counts as "following along". */
const STICKY_THRESHOLD = 32;

const timeFormat = new Intl.DateTimeFormat(undefined, { hour: "2-digit", minute: "2-digit" });

export function ShipsLog({
  entries,
  defaultFilter = "all",
  showFilters = true,
}: {
  entries: LogEntry[];
  defaultFilter?: LogFilter;
  showFilters?: boolean;
}) {
  const listRef = useRef<HTMLOListElement>(null);
  const [filter, setFilter] = useState<LogFilter>(defaultFilter);
  const [following, setFollowing] = useState(true);
  const [seenCount, setSeenCount] = useState(entries.length);

  const visible = useMemo(() => entries.filter((e) => matches(filter, e.tone)), [entries, filter]);
  const counts = useMemo(
    () => ({
      all: entries.length,
      crew: entries.filter((e) => CREW_TONES.has(e.tone)).length,
      voyage: entries.filter((e) => !CREW_TONES.has(e.tone)).length,
    }),
    [entries],
  );

  if (following && seenCount !== entries.length) setSeenCount(entries.length);
  const unseen = following ? 0 : Math.max(0, entries.length - seenCount);

  const scrollToEnd = useCallback((behavior: ScrollBehavior = "smooth") => {
    const list = listRef.current;
    list?.scrollTo({ top: list.scrollHeight, behavior });
  }, []);

  useEffect(() => {
    if (following) scrollToEnd();
  }, [visible.length, following, scrollToEnd]);

  const onScroll = () => {
    const list = listRef.current;
    if (!list) return;
    const atBottom = list.scrollHeight - list.scrollTop - list.clientHeight < STICKY_THRESHOLD;
    if (atBottom !== following) setFollowing(atBottom);
  };

  const jumpToLatest = () => {
    setFollowing(true);
    scrollToEnd();
  };

  return (
    <Panel eyebrow="Kept by the quartermaster" title="Ship's Log" padded={false}>
      {showFilters && (
        <div role="tablist" aria-label="Filter the log" className="flex gap-1.5 border-b border-brass/15 px-5 py-2 sm:px-6">
          {FILTERS.map((f) => (
            <button
              key={f.id}
              type="button"
              role="tab"
              aria-selected={filter === f.id}
              onClick={() => setFilter(f.id)}
              className={cn(
                "rounded-full px-2.5 py-0.5 font-ui text-[0.65rem] uppercase tracking-wider transition-colors",
                filter === f.id ? "bg-brass/25 text-parchment" : "text-brass/60 hover:text-brass-light",
              )}
            >
              {f.label}
              <span className="ml-1 tabular-nums text-brass/50">{counts[f.id]}</span>
            </button>
          ))}
        </div>
      )}

      <div className="relative">
        <ol
          ref={listRef}
          onScroll={onScroll}
          className="max-h-60 space-y-2.5 overflow-y-auto px-5 py-4 sm:px-6 short:max-h-40 short:py-3"
          aria-live="polite"
        >
          <AnimatePresence initial={false}>
            {visible.map((entry) => {
              const tone = TONE_ICON[entry.tone];
              return (
                <motion.li
                  key={entry.id}
                  initial={{ opacity: 0, x: -12 }}
                  animate={{ opacity: 1, x: 0 }}
                  className="flex items-start gap-3"
                >
                  <span className={cn("mt-1 shrink-0", tone.color)}>{tone.icon}</span>
                  <p className="flex-1 font-body leading-snug text-parchment/80">{entry.text}</p>
                  <time className="mt-0.5 shrink-0 font-ui text-[0.6rem] tabular-nums text-brass/50">
                    {timeFormat.format(entry.at)}
                  </time>
                </motion.li>
              );
            })}
          </AnimatePresence>
          {visible.length === 0 && <li className="font-body italic text-parchment/40">{EMPTY_TEXT[filter]}</li>}
        </ol>

        <AnimatePresence>
          {unseen > 0 && (
            <motion.button
              type="button"
              onClick={jumpToLatest}
              initial={{ opacity: 0, y: 8 }}
              animate={{ opacity: 1, y: 0 }}
              exit={{ opacity: 0, y: 8 }}
              className="absolute bottom-2 left-1/2 flex -translate-x-1/2 items-center gap-1.5 rounded-full bg-brass/90 px-3 py-1 font-ui text-[0.65rem] uppercase tracking-wider text-ink shadow-lg"
            >
              <ArrowDownIcon size={12} />
              {unseen} new {unseen === 1 ? "entry" : "entries"}
            </motion.button>
          )}
        </AnimatePresence>
      </div>
    </Panel>
  );
}
