"use client";

import { useState } from "react";
import { CHART_H, CHART_W, SAILOR_COLORS } from "@/lib/game/chart";
import { useMapMarks, type MarksSocket } from "@/lib/game/useMapMarks";
import { getSocket } from "@/lib/socket/client";
import type { PlayerSnapshot } from "@/lib/socket/contract";
import { MapMarksLayer, type MarkTool } from "../voyage/MapMarksLayer";
import { MarkToolbar } from "../voyage/MarkToolbar";
import { TreasureMapSketch } from "./TreasureMapSketch";

interface CrewChartProps {
  players: PlayerSnapshot[];
  meId: string | null;
}

/** The treasure map with live crew ink on top: every pin and route is relayed through the server. */
export function CrewChart({ players, meId }: CrewChartProps) {
  const [socket] = useState<MarksSocket | null>(() => (typeof window === "undefined" ? null : (getSocket() as unknown as MarksSocket)));
  const { marks, place, remove, error } = useMapMarks(socket);
  const [tool, setTool] = useState<MarkTool | null>(null);

  const colorFor = (playerId: string) => {
    const index = players.findIndex((p) => p.id === playerId);
    return SAILOR_COLORS[Math.max(0, index) % SAILOR_COLORS.length];
  };
  const nameFor = (playerId: string) =>
    playerId === meId ? "You" : (players.find((p) => p.id === playerId)?.name ?? "A sailor");

  return (
    <div>
      <div className="relative">
        <TreasureMapSketch />
        <svg
          viewBox={`0 0 ${CHART_W} ${CHART_H}`}
          preserveAspectRatio="none"
          className="absolute inset-0 h-full w-full"
          style={{ pointerEvents: tool ? "auto" : "none" }}
          role="group"
          aria-label="Crew marks on the treasure map"
        >
          {meId && (
            <MapMarksLayer
              marks={marks}
              meId={meId}
              colorFor={colorFor}
              nameFor={nameFor}
              tool={tool}
              onPlace={place}
              onRemove={remove}
              still={false}
            />
          )}
        </svg>
      </div>
      {meId && (
        <MarkToolbar tool={tool} onTool={setTool} marks={marks} meId={meId} error={error} className="mt-3" />
      )}
    </div>
  );
}
