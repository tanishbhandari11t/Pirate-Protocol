const RARITY_STYLE: Record<Rarity, string> = {
  common: "border-foam/40 text-foam",
  cursed: "border-blood-light/60 text-[#f2a097]",
  relic: "border-gold/60 text-gold",
  legendary: "border-gold text-gold shadow-[0_0_18px_rgba(255,216,115,0.4)]",
};

const RARITY_TEXT: Record<Rarity, string> = {
  common: "text-foam",
  cursed: "text-[#f2a097]",
  relic: "text-gold",
  legendary: "text-gold",
};

const ABILITY_TONE: Record<Ability["kind"], string> = {
  vault: "text-gold",
  score: "text-brass-light",
  bound: "text-parchment/50",
  hint: "text-foam",
  trade: "text-kelp-light",
  curse: "text-[#f2a097]",
  treasure: "text-gold",
  gear: "text-foam",
};

type DropTarget = `gear:${GearSlot}` | `crew:${string}` | `berth:${number}`;

/** The topmost drop zone under a viewport point. */
function dropTargetAt(x: number, y: number): DropTarget | null {
  for (const el of document.elementsFromPoint(x, y)) {
    const target = (el as HTMLElement).dataset?.drop;
    if (target) return target as DropTarget;
  }
  return null;
}

interface InventoryProps {
  room: RoomState;
  me: PublicPlayer;
  onTrade: (toPlayerId: string, itemKey: string) => Promise<boolean>;
}

/**
 * The hold as a treasure chest. Items sit in berths, tools can be fitted to the helm and lookout,
 * and tradeable goods can be dragged onto a crewmate to open the usual trade offer.
 */
export function Inventory({ room, me, onTrade }: InventoryProps) {
  const { notify } = useNotify();
  const reduced = useReducedMotion();
  const held = useMemo(() => new Set(room.you.map((i) => i.itemKey)), [room.you]);
  const { layout, gear, equip, unequip, reorder } = useHoldLayout(room.code, held);
  const items = useMemo(() => arrangeHold(room.you, layout.order), [room.you, layout.order]);
  const profiles = useMemo(() => new Map(items.map((i) => [i.itemKey, profileOf(i, room.settings)])), [items, room.settings]);

  const [selected, setSelected] = useState<string | null>(null);
  const [comparing, setComparing] = useState<string | null>(null);
  const [drag, setDrag] = useState<{ key: string; over: DropTarget | null } | null>(null);
  const [offer, setOffer] = useState<{ item: InventoryItem; to: string | null } | null>(null);

  const atSea = room.status !== "FINISHED" && !me.isEliminated;
  const crewmates = room.players.filter((p) => p.id !== me.id && !p.isEliminated);
  const focus = selected ? profiles.get(selected) ?? null : null;
  const other = comparing && comparing !== selected ? profiles.get(comparing) ?? null : null;
  const canTrade = (key: string) => atSea && itemInfo(key).tradable;

  const canDrop = (key: string, target: DropTarget) => {
    if (target.startsWith("gear:")) return gearSlotFor(key) === target.slice(5);
    if (target.startsWith("crew:")) return canTrade(key);
    return true;
  };

  const fit = (key: string) => {
    const slot = equip(key);
    if (!slot) return;
    playSfx("coin");
    notify({ tone: "info", title: `Fitted to the ${GEAR_SLOTS[slot].label}`, message: GEAR_EFFECT[key] });
  };

  const shift = (key: string, index: number) => {
    reorder(moveBerth(items.map((i) => i.itemKey), key, index));
  };

  const drop = (key: string, target: DropTarget | null) => {
    if (!target) return;
    const item = items.find((i) => i.itemKey === key);
    if (!item) return;
    if (target.startsWith("gear:")) {
      if (canDrop(key, target)) fit(key);
      else notify({ tone: "warning", title: "It won't fit", message: GEAR_SLOTS[target.slice(5) as GearSlot].empty + "." });
      return;
    }
    if (target.startsWith("crew:")) {
      if (canTrade(key)) setOffer({ item, to: target.slice(5) });
      else notify({ tone: "warning", title: "Bound to you", message: "Only rumours and cursed coin may change hands." });
      return;
    }
    shift(key, Number(target.slice(6)));
  };

  const pointOf = (event: MouseEvent | TouchEvent | PointerEvent, info: PanInfo) =>
    "clientX" in event ? { x: event.clientX, y: event.clientY } : { x: info.point.x - window.scrollX, y: info.point.y - window.scrollY };

  const pick = (key: string) => {
    if (comparing === "pending" && key !== selected) {
      setComparing(key);
      return;
    }
    setSelected((s) => (s === key ? null : key));
    setComparing(null);
  };

  const zone = (target: DropTarget) => {
    if (!drag) return "idle" as const;
    if (!canDrop(drag.key, target)) return "blocked" as const;
    return drag.over === target ? ("over" as const) : ("open" as const);
  };

  return (
    <div className="space-y-3">
      <ChestLid relics={heldRelics(room).length} used={items.length} still={!!reduced} />

      <section aria-label="Fitted gear" className="grid grid-cols-2 gap-2">
        {(Object.keys(GEAR_SLOTS) as GearSlot[]).map((slot) => {
          const key = gear[slot];
          const state = zone(`gear:${slot}`);
          return (
            <div
              key={slot}
              data-drop={`gear:${slot}`}
              className={cn(
                "relative flex items-center gap-2 rounded-md border-2 p-2 transition",
                key ? "border-brass/40 bg-brass/10" : "border-dashed border-brass/20 bg-abyss/30",
                state === "open" && "border-foam/60",
                state === "over" && "border-gold bg-gold/10 shadow-[0_0_18px_rgba(255,216,115,0.35)]",
                state === "blocked" && "opacity-40",
              )}
            >
              <span className="grid h-10 w-10 shrink-0 place-items-center rounded-full border border-brass/30 bg-abyss/60">
                {key ? <ItemArt glyph={itemInfo(key).glyph} rarity={itemInfo(key).rarity} className="h-9 w-9" halo={false} /> : <SlotEmblem slot={slot} />}
              </span>
              <span className="min-w-0 flex-1">
                <span className="block font-ui text-[0.5rem] font-bold uppercase tracking-[0.25em] text-brass/70">{GEAR_SLOTS[slot].label}</span>
                <span className="block truncate font-display text-base leading-tight text-parchment">
                  {key ? itemInfo(key).name : <span className="font-body text-xs italic text-parchment/40">{GEAR_SLOTS[slot].empty}</span>}
                </span>
              </span>
              {key && (
                <button
                  type="button"
                  onClick={() => unequip(slot)}
                  aria-label={`Unfit ${itemInfo(key).name}`}
                  className="rounded p-1 text-brass/60 transition hover:text-blood-light focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-brass"
                >
                  <CloseIcon size={14} />
                </button>
              )}
            </div>
          );
        })}
      </section>

      <section aria-label="Berths in the hold">
        <ul className="grid grid-cols-4 gap-2">
          {Array.from({ length: HOLD_BERTHS }, (_, index) => {
            const item = items[index];
            const target: DropTarget = `berth:${index}`;
            const state = zone(target);
            if (!item) {
              return (
                <li
                  key={`empty-${index}`}
                  data-drop={target}
                  className={cn(
                    "aspect-square rounded-md border-2 border-dashed border-brass/15 bg-[repeating-linear-gradient(135deg,rgba(200,155,69,0.04)_0_6px,transparent_6px_12px)] transition",
                    state === "over" && "border-gold/70 bg-gold/10",
                  )}
                  aria-label="Empty berth"
                />
              );
            }
            const profile = profiles.get(item.itemKey)!;
            return (
              <Berth
                key={item.itemKey}
                profile={profile}
                target={target}
                over={state === "over"}
                selected={selected === item.itemKey}
                comparing={comparing === item.itemKey}
                fitted={Object.values(gear).includes(item.itemKey)}
                draggable={!reduced}
                onPick={() => pick(item.itemKey)}
                onDragStart={() => setDrag({ key: item.itemKey, over: null })}
                onDrag={(event, info) => {
                  const p = pointOf(event, info);
                  const over = dropTargetAt(p.x, p.y);
                  setDrag((d) => (d && d.over !== over ? { ...d, over } : d));
                }}
                onDragEnd={(event, info) => {
                  const p = pointOf(event, info);
                  drop(item.itemKey, dropTargetAt(p.x, p.y));
                  setDrag(null);
                }}
              />
            );
          })}
        </ul>
        {items.length === 0 && (
          <p className="mt-2 text-center font-body text-sm italic text-parchment/40">
            The chest is empty. Solve an island&apos;s puzzle to claim its relic.
          </p>
        )}
      </section>

      {atSea && crewmates.length > 0 && items.some((i) => canTrade(i.itemKey)) && (
        <section aria-label="Crew dock" className="rounded-md border border-brass/15 bg-abyss/30 p-2">
          <p className="font-ui text-[0.5rem] font-bold uppercase tracking-[0.25em] text-brass/60">
            Drag a tradeable item onto a crewmate to offer it
          </p>
          <ul className="mt-2 flex flex-wrap gap-2">
            {crewmates.map((p) => {
              const state = zone(`crew:${p.id}`);
              return (
                <li
                  key={p.id}
                  data-drop={`crew:${p.id}`}
                  className={cn(
                    "flex items-center gap-1.5 rounded-full border-2 py-0.5 pl-0.5 pr-2.5 transition",
                    state === "over" ? "border-kelp-light bg-kelp/20" : state === "open" ? "border-kelp-light/40" : "border-brass/15",
                    state === "blocked" && "opacity-40",
                  )}
                >
                  <span className="h-7 w-7 overflow-hidden rounded-full border border-brass/30">
                    <PirateAvatar avatarId={p.avatarId} outfit={p.outfit} faded={!p.isOnline} className="h-full w-full" />
                  </span>
                  <span className="max-w-[6rem] truncate font-display text-sm text-parchment">{p.username}</span>
                </li>
              );
            })}
          </ul>
        </section>
      )}

      <AnimatePresence mode="wait">
        {focus && (
          <motion.section
            key={`${focus.item.itemKey}:${other?.item.itemKey ?? ""}`}
            initial={{ opacity: 0, y: 10 }}
            animate={{ opacity: 1, y: 0 }}
            exit={{ opacity: 0, y: -6 }}
            transition={{ duration: 0.2 }}
            aria-label={`${focus.item.name} details`}
          >
            {other ? (
              <ComparePanel left={focus} right={other} onClose={() => setComparing(null)} />
            ) : (
              <ItemDetails
                profile={focus}
                index={items.findIndex((i) => i.itemKey === focus.item.itemKey)}
                count={items.length}
                fittedIn={(Object.keys(gear) as GearSlot[]).find((s) => gear[s] === focus.item.itemKey) ?? null}
                canTrade={canTrade(focus.item.itemKey) && crewmates.length > 0}
                comparePending={comparing === "pending"}
                onFit={() => fit(focus.item.itemKey)}
                onUnfit={(slot) => unequip(slot)}
                onOffer={() => setOffer({ item: focus.item, to: null })}
                onCompare={() => setComparing((c) => (c === "pending" ? null : "pending"))}
                onMove={(to) => shift(focus.item.itemKey, to)}
                onClose={() => setSelected(null)}
              />
            )}
          </motion.section>
        )}
      </AnimatePresence>

      <TradeDialog
        room={room}
        me={me}
        item={offer?.item ?? null}
        initialRecipient={offer?.to ?? null}
        onClose={() => setOffer(null)}
        onTrade={onTrade}
      />
    </div>
  );
}

function ChestLid({ relics, used, still }: { relics: number; used: number; still: boolean }) {
  return (
    <div className="relative" style={{ perspective: 600 }}>
      <motion.svg
        viewBox="0 0 320 54"
        className="block w-full origin-bottom"
        initial={still ? false : { rotateX: 0 }}
        animate={{ rotateX: 38 }}
        transition={{ duration: 0.9, ease: [0.22, 1, 0.36, 1], delay: 0.1 }}
        aria-hidden
      >
        <defs>
          <linearGradient id="chest-lid-wood" x1="0" y1="0" x2="0" y2="1">
            <stop offset="0%" stopColor="#8a5a2b" />
            <stop offset="100%" stopColor="#4a2c14" />
          </linearGradient>
          <linearGradient id="chest-lid-gold" x1="0" y1="0" x2="0" y2="1">
            <stop offset="0%" stopColor="#fff1c1" />
            <stop offset="50%" stopColor="#c89b45" />
            <stop offset="100%" stopColor="#6f4f1c" />
          </linearGradient>
        </defs>
        <path d="M4 52 C4 10 316 10 316 52 Z" fill="url(#chest-lid-wood)" stroke="#2a180a" strokeWidth="2" />
        <path d="M4 52 H316" stroke="url(#chest-lid-gold)" strokeWidth="5" />
        {[40, 160, 280].map((x) => (
          <path key={x} d={`M${x} 52 C${x} 30 ${x + (x < 160 ? 8 : x > 160 ? -8 : 0)} 22 ${x + (x < 160 ? 8 : x > 160 ? -8 : 0)} 18`} stroke="url(#chest-lid-gold)" strokeWidth="6" fill="none" />
        ))}
        <path d="M30 40 C80 22 240 22 290 40" stroke="#fff" strokeOpacity="0.12" strokeWidth="3" fill="none" />
      </motion.svg>
      <div className="-mt-3 flex items-center justify-between gap-2 px-1">
        <p className="font-ui text-[0.55rem] uppercase tracking-[0.28em] text-brass/70">
          {relics}/{REQUIRED_RELICS.length} vault relics
        </p>
        <div className="flex items-center gap-1.5" aria-label={`${used} of ${HOLD_BERTHS} berths filled`}>
          <span className="font-ui text-[0.5rem] uppercase tracking-[0.25em] text-brass/60">Cargo</span>
          {Array.from({ length: HOLD_BERTHS }, (_, i) => (
            <span key={i} className={cn("h-2 w-2 rotate-45 border", i < used ? "border-gold/70 bg-gold/70" : "border-brass/30")} />
          ))}
        </div>
      </div>
    </div>
  );
}

function SlotEmblem({ slot }: { slot: GearSlot }) {
  return (
    <svg viewBox="0 0 24 24" className="h-6 w-6 text-brass/40" fill="none" stroke="currentColor" strokeWidth="1.5" aria-hidden>
      {slot === "helm" ? (
        <>
          <circle cx="12" cy="12" r="6" />
          <circle cx="12" cy="12" r="1.6" />
          {Array.from({ length: 8 }, (_, i) => (
            <path key={i} d="M12 6 V2" transform={`rotate(${i * 45} 12 12)`} strokeLinecap="round" />
          ))}
        </>
      ) : (
        <>
          <circle cx="12" cy="12" r="8" />
          <path d="M4 12 h16 M12 4 v16" strokeOpacity="0.5" />
        </>
      )}
    </svg>
  );
}

interface BerthProps {
  profile: ItemProfile;
  target: DropTarget;
  over: boolean;
  selected: boolean;
  comparing: boolean;
  fitted: boolean;
  draggable: boolean;
  onPick: () => void;
  onDragStart: () => void;
  onDrag: (event: MouseEvent | TouchEvent | PointerEvent, info: PanInfo) => void;
  onDragEnd: (event: MouseEvent | TouchEvent | PointerEvent, info: PanInfo) => void;
}

function Berth({ profile, target, over, selected, comparing, fitted, draggable, onPick, onDragStart, onDrag, onDragEnd }: BerthProps) {
  const { item, info } = profile;
  return (
    <motion.li layout data-drop={target} className={cn("relative aspect-square rounded-md", over && "ring-2 ring-gold/70")}>
      <motion.button
        type="button"
        drag={draggable}
        dragSnapToOrigin
        dragElastic={0.9}
        dragMomentum={false}
        whileDrag={{ scale: 1.12, zIndex: 50, rotate: -4, cursor: "grabbing" }}
        whileHover={{ y: -2 }}
        onDragStart={onDragStart}
        onDrag={onDrag}
        onDragEnd={onDragEnd}
        onClick={onPick}
        aria-pressed={selected}
        aria-label={`${item.name}${item.quantity > 1 ? `, ${item.quantity} held` : ""}, ${RARITY_LABEL[info.rarity]}`}
        className={cn(
          "relative flex h-full w-full touch-none flex-col items-center justify-center rounded-md border-2 bg-[radial-gradient(circle_at_50%_40%,rgba(20,54,72,0.8),rgba(3,7,12,0.9))] p-1 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-gold/80",
          RARITY_STYLE[info.rarity],
          selected && "ring-2 ring-gold",
          comparing && "ring-2 ring-foam",
          draggable && "cursor-grab",
        )}
      >
        <ItemArt glyph={info.glyph} rarity={info.rarity} className="h-[78%] w-[78%] pointer-events-none" />
        {item.quantity > 1 && (
          <span className="absolute bottom-0.5 right-1 font-ui text-[0.6rem] font-bold text-brass-light">×{item.quantity}</span>
        )}
        {fitted && (
          <span className="absolute left-0.5 top-0.5 rounded-sm bg-gold px-1 font-ui text-[0.45rem] font-bold uppercase tracking-wider text-ink">Fitted</span>
        )}
        {!info.tradable && <LockIcon size={10} className="absolute right-1 top-1 text-parchment/30" aria-hidden />}
      </motion.button>
    </motion.li>
  );
}

interface ItemDetailsProps {
  profile: ItemProfile;
  index: number;
  count: number;
  fittedIn: GearSlot | null;
  canTrade: boolean;
  comparePending: boolean;
  onFit: () => void;
  onUnfit: (slot: GearSlot) => void;
  onOffer: () => void;
  onCompare: () => void;
  onMove: (index: number) => void;
  onClose: () => void;
}

function ItemDetails({ profile, index, count, fittedIn, canTrade, comparePending, onFit, onUnfit, onOffer, onCompare, onMove, onClose }: ItemDetailsProps) {
  const { item, info, abilities, gear } = profile;
  return (
    <div className="rounded-md border border-brass/25 bg-abyss/50 p-3">
      <div className="flex gap-3">
        <span className={cn("grid h-16 w-16 shrink-0 place-items-center rounded-md border-2 bg-abyss/70", RARITY_STYLE[info.rarity])}>
          <ItemArt glyph={info.glyph} rarity={info.rarity} className="h-14 w-14" />
        </span>
        <div className="min-w-0 flex-1">
          <div className="flex items-start justify-between gap-2">
            <p className="font-display text-xl leading-tight text-parchment">{item.name}</p>
            <button type="button" onClick={onClose} aria-label="Close details" className="rounded p-0.5 text-brass/60 hover:text-parchment">
              <CloseIcon size={14} />
            </button>
          </div>
          <p className={cn("font-ui text-[0.52rem] uppercase tracking-[0.22em]", RARITY_TEXT[info.rarity])}>
            {RARITY_LABEL[info.rarity]}
            {item.quantity > 1 && <span className="text-brass-light"> · ×{item.quantity}</span>}
          </p>
          <p className="mt-1 font-body text-sm italic leading-snug text-parchment/60">{info.lore || item.description}</p>
        </div>
      </div>

      {abilities.length > 0 && (
        <ul className="mt-3 space-y-1.5">
          {abilities.map((a) => (
            <li key={a.kind} className="flex gap-2 text-sm leading-snug">
              <span className={cn("shrink-0 font-ui text-[0.55rem] font-bold uppercase tracking-[0.18em] leading-5", ABILITY_TONE[a.kind])}>{a.label}</span>
              <span className="font-body text-parchment/65">{a.detail}</span>
            </li>
          ))}
        </ul>
      )}

      <div className="mt-3 flex flex-wrap items-center gap-1.5">
        {gear &&
          (fittedIn ? (
            <Button size="sm" variant="ghost" onClick={() => onUnfit(fittedIn)}>
              Unfit
            </Button>
          ) : (
            <Button size="sm" onClick={onFit}>
              Fit to {GEAR_SLOTS[gear].label}
            </Button>
          ))}
        {canTrade && (
          <Button size="sm" variant="ghost" onClick={onOffer} icon={<SwordsIcon size={12} />}>
            Offer
          </Button>
        )}
        {count > 1 && (
          <Button size="sm" variant="ghost" onClick={onCompare} aria-pressed={comparePending}>
            {comparePending ? "Pick an item…" : "Compare"}
          </Button>
        )}
        <span className="ml-auto flex gap-1">
          <button
            type="button"
            onClick={() => onMove(index - 1)}
            disabled={index <= 0}
            aria-label="Move to an earlier berth"
            className="rounded border border-brass/25 p-1 text-brass/70 transition hover:text-gold disabled:opacity-30"
          >
            <ArrowLeftIcon size={14} />
          </button>
          <button
            type="button"
            onClick={() => onMove(index + 1)}
            disabled={index >= count - 1}
            aria-label="Move to a later berth"
            className="rounded border border-brass/25 p-1 text-brass/70 transition hover:text-gold disabled:opacity-30"
          >
            <ArrowRightIcon size={14} />
          </button>
        </span>
      </div>
    </div>
  );
}

function ComparePanel({ left, right, onClose }: { left: ItemProfile; right: ItemProfile; onClose: () => void }) {
  const rows = compareItems(left, right);
  return (
    <div className="rounded-md border border-foam/30 bg-abyss/50 p-3">
      <div className="flex items-center justify-between">
        <p className="font-ui text-[0.55rem] font-bold uppercase tracking-[0.3em] text-foam">Side by side</p>
        <button type="button" onClick={onClose} aria-label="Stop comparing" className="rounded p-0.5 text-brass/60 hover:text-parchment">
          <CloseIcon size={14} />
        </button>
      </div>
      <div className="mt-2 grid grid-cols-[1fr_auto_1fr] items-center gap-x-2 gap-y-1.5 text-center">
        {[left, right].map((p, i) => (
          <div key={p.item.itemKey} className={cn("flex flex-col items-center", i === 1 && "col-start-3")}>
            <ItemArt glyph={p.info.glyph} rarity={p.info.rarity} className="h-12 w-12" />
            <span className="font-display text-base leading-tight text-parchment">{p.item.name}</span>
          </div>
        ))}
        {rows.map((row) => (
          <CompareLine key={row.label} {...row} />
        ))}
      </div>
    </div>
  );
}

function CompareLine({ label, left, right, better }: ReturnType<typeof compareItems>[number]) {
  const cell = (side: "left" | "right", text: string) => (
    <span
      className={cn(
        "rounded px-1 py-0.5 font-ui text-[0.65rem] capitalize",
        better === side ? "bg-gold/15 font-bold text-gold" : "text-parchment/70",
      )}
    >
      {text}
    </span>
  );
  return (
    <>
      {cell("left", left)}
      <span className="font-ui text-[0.48rem] uppercase tracking-[0.2em] text-brass/60">{label}</span>
      {cell("right", right)}
    </>
  );
}

interface TradeDialogProps {
  room: RoomState;
  me: PublicPlayer;
  item: InventoryItem | null;
  /** Preselected when an item was dropped onto a crewmate. */
  initialRecipient: string | null;
  onClose: () => void;
  onTrade: (toPlayerId: string, itemKey: string) => Promise<boolean>;
}

function TradeDialog({ room, me, item, initialRecipient, onClose, onTrade }: TradeDialogProps) {
  const [picked, setPicked] = useState<{ item: string | null; to: string } | null>(null);
  const [pending, setPending] = useState(false);
  const crewmates = room.players.filter((p) => p.id !== me.id && !p.isEliminated);
  const recipient = picked && picked.item === (item?.itemKey ?? null) ? picked.to : initialRecipient;
  const chosen = crewmates.some((p) => p.id === recipient) ? recipient : null;
  const info = item ? itemInfo(item.itemKey) : null;

  const close = () => {
    setPicked(null);
    onClose();
  };

  const offer = async () => {
    if (!item || !chosen) return;
    setPending(true);
    const ok = await onTrade(chosen, item.itemKey);
    setPending(false);
    if (ok) close();
  };

  return (
    <Modal
      open={!!item}
      onClose={close}
      eyebrow="Barter on the open sea"
      title="Make an Offer"
      dismissible={!pending}
      footer={
        <>
          <Button variant="parchment" onClick={close} disabled={pending}>
            Keep it
          </Button>
          <Button onClick={offer} loading={pending} disabled={!chosen} icon={<SwordsIcon size={16} />}>
            Offer
          </Button>
        </>
      }
    >
      {item && info && (
        <div>
          <div className="flex items-center justify-center gap-3">
            <span className="flex h-14 w-14 items-center justify-center rounded-full border-2 border-ink/40 bg-sea/80">
              <ItemArt glyph={info.glyph} rarity={info.rarity} className="h-12 w-12" />
            </span>
            <div className="text-left">
              <p className="font-ui text-[0.6rem] font-bold uppercase tracking-[0.3em] text-ink-soft/70">You offer</p>
              <p className="font-display text-2xl leading-none text-ink">{item.name}</p>
            </div>
          </div>
          <p className="mt-4 text-center text-base">
            {item.quantity > 1 ? `All ${item.quantity} pass` : "It passes"} straight into their hold. The harbour master allows no
            take-backs.
          </p>
          <p className="mt-5 font-ui text-[0.6rem] font-bold uppercase tracking-[0.3em] text-ink-soft/80">Choose a crewmate</p>
          <div role="radiogroup" aria-label="Choose a crewmate" className="mt-2 grid gap-2 sm:grid-cols-2">
            {crewmates.map((p) => (
              <button
                key={p.id}
                type="button"
                role="radio"
                aria-checked={chosen === p.id}
                onClick={() => setPicked({ item: item.itemKey, to: p.id })}
                className={cn(
                  "flex items-center gap-3 rounded border-2 px-3 py-2 text-left transition focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ink/40",
                  chosen === p.id ? "border-blood bg-blood/10" : "border-ink/20 hover:border-ink/40",
                )}
              >
                <span className="h-9 w-9 shrink-0 overflow-hidden rounded-full border border-ink/40">
                  <PirateAvatar avatarId={p.avatarId} outfit={p.outfit} faded={!p.isOnline} className="h-full w-full" />
                </span>
                <span className="min-w-0">
                  <span className="block truncate font-display text-xl leading-none text-ink">{p.username}</span>
                  <span className="font-body text-xs italic text-ink-soft">{p.isOnline ? "Aboard" : "In the fog"}</span>
                </span>
              </button>
            ))}
            {crewmates.length === 0 && (
              <p className="col-span-full text-center italic">No crewmate can receive it right now.</p>
            )}
          </div>
        </div>
      )}
    </Modal>
  );
}
