"use client";

import { AnimatePresence, motion } from "framer-motion";
import { useState } from "react";
import { cn } from "@/lib/cn";
import { heldRelics } from "@/lib/game/selectors";
import { RARITY_LABEL, itemInfo, type Rarity } from "@/lib/game/world";
import { REQUIRED_RELICS, type InventoryItem, type PublicPlayer, type RoomState } from "@/lib/socket/contract";
import { PirateAvatar } from "../avatar/PirateAvatar";
import { SwordsIcon } from "../icons";
import { Button } from "../ui/Button";
import { Modal } from "../ui/Modal";
import { ItemGlyph } from "./ItemGlyph";

const RARITY_STYLE: Record<Rarity, string> = {
  common: "border-foam/40 text-foam",
  cursed: "border-blood-light/60 text-[#f2a097]",
  relic: "border-gold/60 text-gold",
  legendary: "border-gold text-gold shadow-[0_0_18px_rgba(255,216,115,0.4)]",
};

interface InventoryProps {
  room: RoomState;
  me: PublicPlayer;
  onTrade: (toPlayerId: string, itemKey: string) => Promise<boolean>;
}

export function Inventory({ room, me, onTrade }: InventoryProps) {
  const [trading, setTrading] = useState<InventoryItem | null>(null);
  const items = [...room.you].sort((a, b) => rank(a.itemKey) - rank(b.itemKey));

  return (
    <div>
      <p className="font-ui text-[0.58rem] uppercase tracking-[0.3em] text-brass/70">
        In the hold · {heldRelics(room).length}/{REQUIRED_RELICS.length} vault relics
      </p>
      <ul className="mt-2 space-y-2">
        <AnimatePresence initial={false}>
          {items.map((item) => {
            const info = itemInfo(item.itemKey);
            return (
              <motion.li
                key={item.itemKey}
                layout
                initial={{ opacity: 0, x: 20 }}
                animate={{ opacity: 1, x: 0 }}
                exit={{ opacity: 0, x: -20 }}
                className="flex gap-3 rounded-md border border-brass/15 bg-abyss/40 p-3"
              >
                <span className={cn("flex h-11 w-11 shrink-0 items-center justify-center rounded-full border-2 bg-abyss/70", RARITY_STYLE[info.rarity])}>
                  <ItemGlyph glyph={info.glyph} size={22} />
                </span>
                <div className="min-w-0 flex-1">
                  <div className="flex items-baseline justify-between gap-2">
                    <p className="truncate font-display text-lg leading-tight text-parchment">{item.name}</p>
                    {item.quantity > 1 && <span className="font-ui text-xs text-brass-light">×{item.quantity}</span>}
                  </div>
                  <p className="font-body text-sm leading-snug text-parchment/60">{item.description}</p>
                  <div className="mt-1.5 flex flex-wrap items-center gap-2">
                    <span className={cn("font-ui text-[0.52rem] uppercase tracking-[0.22em]", RARITY_STYLE[info.rarity].split(" ").pop())}>
                      {RARITY_LABEL[info.rarity]}
                    </span>
                    <span className="font-ui text-[0.52rem] uppercase tracking-[0.22em] text-parchment/40">
                      {info.tradable ? "· Tradeable" : "· Bound"}
                    </span>
                    {info.tradable && !me.isEliminated && room.status !== "FINISHED" && (
                      <button
                        type="button"
                        onClick={() => setTrading(item)}
                        className="ml-auto inline-flex items-center gap-1 rounded border border-kelp-light/40 px-2 py-0.5 font-ui text-[0.55rem] uppercase tracking-[0.2em] text-kelp-light transition hover:bg-kelp/20 focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-kelp-light"
                      >
                        <SwordsIcon size={11} /> Offer
                      </button>
                    )}
                  </div>
                </div>
              </motion.li>
            );
          })}
        </AnimatePresence>
        {items.length === 0 && (
          <li className="rounded-md border border-dashed border-brass/15 px-4 py-6 text-center font-body italic text-parchment/40">
            The hold is empty. Solve an island&apos;s puzzle to claim its relic.
          </li>
        )}
      </ul>

      <TradeDialog room={room} me={me} item={trading} onClose={() => setTrading(null)} onTrade={onTrade} />
    </div>
  );
}

function rank(itemKey: string) {
  const order: Rarity[] = ["legendary", "relic", "common", "cursed"];
  return order.indexOf(itemInfo(itemKey).rarity);
}

interface TradeDialogProps {
  room: RoomState;
  me: PublicPlayer;
  item: InventoryItem | null;
  onClose: () => void;
  onTrade: (toPlayerId: string, itemKey: string) => Promise<boolean>;
}

function TradeDialog({ room, me, item, onClose, onTrade }: TradeDialogProps) {
  const [recipient, setRecipient] = useState<string | null>(null);
  const [pending, setPending] = useState(false);
  const crewmates = room.players.filter((p) => p.id !== me.id && !p.isEliminated);
  const chosen = crewmates.some((p) => p.id === recipient) ? recipient : null;
  const info = item ? itemInfo(item.itemKey) : null;

  const close = () => {
    setRecipient(null);
    onClose();
  };

  const offer = async () => {
    if (!item || !chosen) return;
    setPending(true);
    const ok = await onTrade(chosen, item.itemKey);
    setPending(false);
    if (ok) close();
  };

  return (
    <Modal
      open={!!item}
      onClose={close}
      eyebrow="Barter on the open sea"
      title="Make an Offer"
      dismissible={!pending}
      footer={
        <>
          <Button variant="parchment" onClick={close} disabled={pending}>
            Keep it
          </Button>
          <Button onClick={offer} loading={pending} disabled={!chosen} icon={<SwordsIcon size={16} />}>
            Offer
          </Button>
        </>
      }
    >
      {item && info && (
        <div>
          <div className="flex items-center justify-center gap-3">
            <span className="flex h-12 w-12 items-center justify-center rounded-full border-2 border-ink/40 text-ink">
              <ItemGlyph glyph={info.glyph} size={24} />
            </span>
            <div className="text-left">
              <p className="font-ui text-[0.6rem] font-bold uppercase tracking-[0.3em] text-ink-soft/70">You offer</p>
              <p className="font-display text-2xl leading-none text-ink">{item.name}</p>
            </div>
          </div>
          <p className="mt-4 text-center text-base">
            {item.quantity > 1 ? `All ${item.quantity} pass` : "It passes"} straight into their hold. The harbour master allows no
            take-backs.
          </p>
          <p className="mt-5 font-ui text-[0.6rem] font-bold uppercase tracking-[0.3em] text-ink-soft/80">Choose a crewmate</p>
          <div role="radiogroup" aria-label="Choose a crewmate" className="mt-2 grid gap-2 sm:grid-cols-2">
            {crewmates.map((p) => (
              <button
                key={p.id}
                type="button"
                role="radio"
                aria-checked={chosen === p.id}
                onClick={() => setRecipient(p.id)}
                className={cn(
                  "flex items-center gap-3 rounded border-2 px-3 py-2 text-left transition focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ink/40",
                  chosen === p.id ? "border-blood bg-blood/10" : "border-ink/20 hover:border-ink/40",
                )}
              >
                <span className="h-9 w-9 shrink-0 overflow-hidden rounded-full border border-ink/40">
                  <PirateAvatar avatarId={p.avatarId} faded={!p.isOnline} className="h-full w-full" />
                </span>
                <span className="min-w-0">
                  <span className="block truncate font-display text-xl leading-none text-ink">{p.username}</span>
                  <span className="font-body text-xs italic text-ink-soft">{p.isOnline ? "Aboard" : "In the fog"}</span>
                </span>
              </button>
            ))}
            {crewmates.length === 0 && (
              <p className="col-span-full text-center italic">No crewmate can receive it right now.</p>
            )}
          </div>
        </div>
      )}
    </Modal>
  );
}
