"use client";

import { cn } from "@/lib/cn";
import { useCrew } from "@/lib/crew/CrewProvider";
import { watchersOf } from "@/lib/crew/reducer";
import { islandByKey } from "@/lib/game/selectors";
import { CAPABILITIES } from "@/lib/socket/backend";
import type { RoomState } from "@/lib/socket/contract";
import { describeError } from "@/lib/socket/errors";
import { PirateAvatar } from "../avatar/PirateAvatar";
import { BinocularsIcon, CloseIcon, EyeIcon } from "../icons";
import { useNotify } from "../ui/Notifications";

/**
 * For sailors the sea has claimed: pick a crewmate to follow from the rigging. For everyone still
 * sailing: a quiet note of who is watching them.
 */
export function SpectatorBar({ room }: { room: RoomState }) {
  const { state, sailor, follow, pending } = useCrew();
  const { notify } = useNotify();
  if (!sailor) return null;

  const watchers = watchersOf(state, sailor.id);
  const canSpectate = sailor.isEliminated || room.status === "FINISHED";

  if (!CAPABILITIES.spectate) {
    if (!sailor.isEliminated) return null;
    return (
      <p className="flex items-center gap-2 rounded border border-foam/25 bg-abyss/50 px-3 py-2 font-body text-sm italic text-parchment/65">
        <BinocularsIcon size={14} className="shrink-0 text-foam/80" />
        The sea has claimed you. Follow your crew&apos;s progress on the chart.
      </p>
    );
  }

  if (!canSpectate) {
    if (watchers.length === 0) return null;
    return (
      <p className="flex items-center gap-2 rounded border border-brass/15 bg-abyss/40 px-3 py-2 font-body text-sm italic text-parchment/60">
        <EyeIcon size={14} className="shrink-0 text-brass" />
        {watchers.map((w) => w.name).join(", ")} {watchers.length === 1 ? "is" : "are"} watching you from the rigging.
      </p>
    );
  }

  const following = room.players.find((p) => p.id === sailor.followingId) ?? null;
  const candidates = room.players.filter((p) => p.id !== sailor.id && !p.isEliminated);
  const busy = pending.has("spectate:follow");

  const choose = async (playerId: string | null) => {
    const res = await follow(playerId);
    if (!res.ok) notify({ tone: "warning", title: "The spyglass slips", message: describeError(res.error.code, res.error.message) });
  };

  return (
    <section className="rounded-md border border-foam/25 bg-abyss/50 p-3" aria-label="Watch from the rigging">
      <p className="flex items-center gap-2 font-ui text-[0.58rem] uppercase tracking-[0.3em] text-foam/80">
        <BinocularsIcon size={13} /> From the rigging
      </p>
      {following ? (
        <div className="mt-2 flex items-center gap-3">
          <span className="h-9 w-9 shrink-0 overflow-hidden rounded-full border border-foam/50">
            <PirateAvatar avatarId={following.avatarId} className="h-full w-full" />
          </span>
          <p className="min-w-0 flex-1 font-body text-parchment/80">
            Watching <span className="font-display text-lg text-parchment">{following.username}</span>
            <span className="block truncate text-sm italic text-parchment/50">
              {islandByKey(room, following.currentIslandKey)?.name ?? "Adrift"}
            </span>
          </p>
          <button
            type="button"
            onClick={() => choose(null)}
            disabled={busy}
            aria-label="Stop watching"
            className="rounded p-1.5 text-parchment/50 transition hover:text-parchment disabled:opacity-40"
          >
            <CloseIcon size={15} />
          </button>
        </div>
      ) : (
        <p className="mt-1 font-body text-sm italic text-parchment/55">
          {candidates.length > 0 ? "Choose a crewmate to follow across the map." : "No one is left sailing."}
        </p>
      )}

      {candidates.length > 0 && (
        <div className="mt-2.5 flex flex-wrap gap-1.5">
          {candidates.map((p) => (
            <button
              key={p.id}
              type="button"
              onClick={() => choose(p.id)}
              disabled={busy || p.id === sailor.followingId}
              aria-pressed={p.id === sailor.followingId}
              className={cn(
                "flex items-center gap-1.5 rounded-full border py-0.5 pl-0.5 pr-2.5 font-ui text-[0.6rem] uppercase tracking-wider transition",
                p.id === sailor.followingId
                  ? "border-foam/70 bg-foam/15 text-foam"
                  : "border-brass/25 text-brass-light enabled:hover:border-foam/60",
              )}
            >
              <span className="h-5 w-5 overflow-hidden rounded-full">
                <PirateAvatar avatarId={p.avatarId} className="h-full w-full" />
              </span>
              {p.username}
            </button>
          ))}
        </div>
      )}
    </section>
  );
}
