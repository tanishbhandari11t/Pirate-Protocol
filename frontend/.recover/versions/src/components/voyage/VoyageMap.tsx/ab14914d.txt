import { CHART_H, CHART_W, SAILOR_COLORS, chartPoint, islandPath, lanePath } from "@/lib/game/chart";

export interface MapCallout {
  id: string;
  text: string;
  /** Quick calls get a louder scroll than free text. */
  call: boolean;
}

export interface KrakenSighting {
  id: string;
  islandKey: string;
}

const INK = "#3a2412";
const LAND = "#d8bd84";
const EASE = [0.45, 0, 0.25, 1] as const;
const MAX_DARK = 0.5;

const seedOf = (text: string) => [...text].reduce((h, c) => (Math.imul(h, 31) + c.charCodeAt(0)) | 0, 7) >>> 0;

const sailorColor = (index: number, isMe: boolean) => (isMe ? "#ffd873" : SAILOR_COLORS[(index + 1) % SAILOR_COLORS.length]);

/** How dark the chart is (0 at noon, MAX_DARK at midnight), from the voyage clock, refreshed every few seconds. */
function useDusk(room: RoomState, enabled: boolean, sky: Sky | undefined) {
  const [now, setNow] = useState(() => Date.now());
  useEffect(() => {
    if (!enabled || sky) return;
    const timer = window.setInterval(() => setNow(Date.now()), 15_000);
    return () => window.clearInterval(timer);
  }, [enabled, sky]);
  if (!enabled) return 0;
  const darkness = sky?.darkness ?? skyFor({ startedAt: voyageStartedAt(room), endsAt: room.endsAt, now }).darkness;
  return darkness * MAX_DARK;
}

interface VoyageMapProps {
  room: RoomState;
  meId: string;
  charted: ReadonlySet<string>;
  vaultOpen: boolean;
  selectedKey: string | null;
  sailingTo: string | null;
  fogSurge: boolean;
  onSelect: (islandKey: string) => void;
  /** Rain and lightning sweep the chart while true. */
  storm?: boolean;
  /** Live sea state: drives the wind rose, gusts and storms. */
  weather?: Weather;
  /** The latest words from each sailor, keyed by player id. */
  callouts?: Record<string, MapCallout>;
  kraken?: KrakenSighting | null;
  dayCycle?: boolean;
  /** The voyage's sky; when given, the chart darkens with it instead of keeping its own clock. */
  sky?: Sky;
  /** Charted trap islands whose waters run dark. */
  dangerZones?: TrapZone[];
  /** A fitted chart's course from this sailor to their next destination. */
  course?: { from: string; to: string } | null;
  /** A fitted spyglass's soundings: distances from this island to every charted one. */
  soundingsFrom?: string | null;
  interactive?: boolean;
  className?: string;
}

export function VoyageMap({
  room,
  meId,
  charted,
  vaultOpen,
  selectedKey,
  sailingTo,
  fogSurge,
  onSelect,
  storm = false,
  weather,
  callouts,
  kraken,
  dayCycle = true,
  sky,
  dangerZones,
  course,
  soundingsFrom,
  interactive = true,
  className,
}: VoyageMapProps) {
  const stormy = storm || !!weather?.storm;
  const uid = useId().replace(/:/g, "");
  const reduced = useReducedMotion();
  const islands = useMemo(() => [...room.islands].sort((a, b) => a.order - b.order), [room.islands]);
  const solved = useMemo(() => solvedKeys(room, meId), [room, meId]);
  const trails = useMemo(() => trailsFrom(room), [room]);
  const dusk = useDusk(room, dayCycle, sky);
  const target = sailingTo ? islands.find((i) => i.key === sailingTo) : null;
  const focus = target ? chartPoint(target) : null;
  const camera = focus
    ? { scale: 1.14, x: (CHART_W / 2 - focus.x) * 0.22, y: (CHART_H / 2 - focus.y) * 0.22 }
    : { scale: 1, x: 0, y: 0 };
  const cameraStyle = { transformBox: "view-box" as const, transformOrigin: `${CHART_W / 2}px ${CHART_H / 2}px` };
  const point = (key: string) => {
    const island = islands.find((i) => i.key === key);
    return island ? chartPoint(island) : null;
  };

  const lanes = islands.filter((i) => i.key !== VAULT_KEY);
  const vault = islands.find((i) => i.key === VAULT_KEY);
  const vaultFrom = lanes[lanes.length - 1];
  const krakenIsland = kraken ? islands.find((i) => i.key === kraken.islandKey) : null;
  const krakenAt = krakenIsland ? chartPoint(krakenIsland) : null;

  return (
    <div className={cn("relative", className)}>
      <div aria-hidden className="parchment absolute inset-0 rounded-[3px] shadow-[0_24px_60px_rgba(0,0,0,0.6)]" />
      <svg
        viewBox={`0 0 ${CHART_W} ${CHART_H}`}
        className="relative block h-full w-full"
        role="group"
        aria-label="The living treasure map"
      >
        <defs>
          <radialGradient id={`${uid}-fog`}>
            <stop offset="0%" stopColor="#f3e6c4" stopOpacity="0.95" />
            <stop offset="60%" stopColor="#e9d6a6" stopOpacity="0.7" />
            <stop offset="100%" stopColor="#e9d6a6" stopOpacity="0" />
          </radialGradient>
          <radialGradient id={`${uid}-glow`}>
            <stop offset="0%" stopColor="#ffd873" stopOpacity="0.85" />
            <stop offset="100%" stopColor="#ffd873" stopOpacity="0" />
          </radialGradient>
          <radialGradient id={`${uid}-danger`}>
            <stop offset="0%" stopColor="#7a1414" stopOpacity="0.38" />
            <stop offset="55%" stopColor="#5a1018" stopOpacity="0.18" />
            <stop offset="100%" stopColor="#5a1018" stopOpacity="0" />
          </radialGradient>
          <radialGradient id={`${uid}-vignette`} cx="50%" cy="50%" r="70%">
            <stop offset="60%" stopColor="#2b1b0d" stopOpacity="0" />
            <stop offset="100%" stopColor="#2b1b0d" stopOpacity="0.45" />
          </radialGradient>
          <mask id={`${uid}-vault-route`} maskUnits="userSpaceOnUse" x="0" y="0" width={CHART_W} height={CHART_H}>
            {vault && vaultFrom && vaultOpen && (
              <motion.path
                d={lanePath(chartPoint(vaultFrom), chartPoint(vault), -0.25)}
                stroke="#fff"
                strokeWidth="14"
                fill="none"
                initial={{ pathLength: 0 }}
                animate={{ pathLength: 1 }}
                transition={{ duration: 2.4, ease: "easeInOut", delay: 0.4 }}
              />
            )}
          </mask>
        </defs>

        <SeaDecor />
        <SeaLife />

        <motion.g animate={camera} transition={{ duration: 1.4, ease: EASE }} style={cameraStyle}>
          {dangerZones?.map((zone) => (
            <g key={`danger-${zone.key}`} pointerEvents="none">
              <motion.ellipse
                cx={zone.x}
                cy={zone.y}
                rx="120"
                ry="86"
                fill={`url(#${uid}-danger)`}
                initial={{ opacity: 0 }}
                animate={reduced ? { opacity: zone.spent ? 0.35 : 1 } : { opacity: zone.spent ? 0.35 : [0.7, 1, 0.7] }}
                transition={reduced || zone.spent ? { duration: 1.5 } : { duration: 5, repeat: Infinity, ease: "easeInOut" }}
              />
              {!zone.spent &&
                [0, 1, 2].map((ring) => (
                  <ellipse
                    key={ring}
                    cx={zone.x}
                    cy={zone.y}
                    rx={58 + ring * 18}
                    ry={40 + ring * 13}
                    fill="none"
                    stroke="#7a1414"
                    strokeOpacity={0.22 - ring * 0.06}
                    strokeWidth="1.4"
                    strokeDasharray="3 9"
                  />
                ))}
            </g>
          ))}

          {lanes.slice(1).map((island, i) => {
            const from = lanes[i];
            const known = charted.has(from.key) && charted.has(island.key);
            return (
              <path
                key={`${from.key}-${island.key}`}
                d={lanePath(chartPoint(from), chartPoint(island), i % 2 ? 0.16 : -0.16)}
                stroke={known ? "#8e1f1a" : INK}
                strokeOpacity={known ? 0.75 : 0.18}
                strokeWidth={known ? 3 : 2}
                strokeDasharray={known ? "1 9" : "2 10"}
                strokeLinecap="round"
                fill="none"
              />
            );
          })}

          {vault && vaultFrom && vaultOpen && (
            <g mask={`url(#${uid}-vault-route)`}>
              <path
                d={lanePath(chartPoint(vaultFrom), chartPoint(vault), -0.25)}
                stroke="#ffd873"
                strokeOpacity="0.45"
                strokeWidth="12"
                fill="none"
                strokeLinecap="round"
              />
              <path
                d={lanePath(chartPoint(vaultFrom), chartPoint(vault), -0.25)}
                stroke="#b8860b"
                strokeWidth="3.5"
                strokeDasharray="2 8"
                strokeLinecap="round"
                fill="none"
              />
            </g>
          )}

          {room.players.map((player, index) => {
            const stops = (trails.get(player.id) ?? []).map(point).filter((p) => p !== null);
            if (stops.length < 2) return null;
            const shift = (index - 1) * 4;
            const d = stops.map((p, i) => `${i ? "L" : "M"}${p.x + shift} ${p.y + shift}`).join(" ");
            return (
              <g key={`wake-${player.id}`} pointerEvents="none" opacity={player.isOnline ? 0.9 : 0.45}>
                <path d={d} fill="none" stroke={INK} strokeOpacity="0.3" strokeWidth="5" strokeLinecap="round" strokeLinejoin="round" />
                <motion.path
                  d={d}
                  fill="none"
                  stroke={sailorColor(index, player.id === meId)}
                  strokeWidth="2.5"
                  strokeDasharray="3 7"
                  strokeLinecap="round"
                  strokeLinejoin="round"
                  initial={{ pathLength: 0 }}
                  animate={{ pathLength: 1 }}
                  transition={{ duration: reduced ? 0 : 1.2 }}
                />
              </g>
            );
          })}

          {islands.map((island) => (
            <IslandMark
              key={island.key}
              interactive={interactive}
              uid={uid}
              island={island}
              charted={island.key === VAULT_KEY ? vaultOpen : charted.has(island.key)}
              hidden={island.key === VAULT_KEY && !vaultOpen}
              selected={selectedKey === island.key}
              solved={island.puzzles.some((p) => solved.has(p.key))}
              onSelect={onSelect}
            />
          ))}

          {vault && vaultOpen && <GoldDust {...chartPoint(vault)} />}

          {course && <CourseLine from={point(course.from)} to={point(course.to)} still={!!reduced} />}

          {soundingsFrom &&
            (() => {
              const origin = point(soundingsFrom);
              if (!origin) return null;
              return islands
                .filter((i) => i.key !== soundingsFrom && (i.key === VAULT_KEY ? vaultOpen : charted.has(i.key)))
                .map((island) => {
                  const p = chartPoint(island);
                  return (
                    <g key={`sounding-${island.key}`} transform={`translate(${p.x - 46} ${p.y - 42})`} pointerEvents="none">
                      <motion.g initial={{ opacity: 0, scale: 0.6 }} animate={{ opacity: 1, scale: 1 }} transition={{ duration: 0.5 }}>
                        <circle r="13" fill="#f5e8c6" stroke={INK} strokeWidth="1.2" />
                        <text y="4.5" textAnchor="middle" fontSize="12" fontWeight="700" fill={INK} style={{ fontFamily: "var(--font-cinzel)" }}>
                          {leagues(Math.hypot(p.x - origin.x, p.y - origin.y))}
                        </text>
                      </motion.g>
                    </g>
                  );
                });
            })()}

          <AnimatePresence>
            {kraken && krakenAt && <Kraken key={kraken.id} x={krakenAt.x} y={krakenAt.y} seed={seedOf(kraken.id)} />}
          </AnimatePresence>

          {room.players.map((player, index) => (
            <SailorShip
              key={player.id}
              player={player}
              index={index}
              isMe={player.id === meId}
              islandKey={player.id === meId && sailingTo ? sailingTo : player.currentIslandKey}
              islands={islands}
              sailing={player.id === meId && !!sailingTo}
              callout={callouts?.[player.id] ?? null}
              still={!!reduced}
            />
          ))}
        </motion.g>

        <DriftingFog uid={uid} />
        {weather && <Gusts weather={weather} />}

        <motion.rect
          width={CHART_W}
          height={CHART_H}
          fill="#0b1a2e"
          pointerEvents="none"
          style={{ mixBlendMode: "multiply" }}
          initial={false}
          animate={{ opacity: dusk }}
          transition={{ duration: 8 }}
        />
        <motion.g animate={camera} transition={{ duration: 1.4, ease: EASE }} style={cameraStyle} pointerEvents="none">
          {islands
            .filter((i) => (i.key === VAULT_KEY ? vaultOpen : charted.has(i.key)))
            .map((island) => {
              const p = chartPoint(island);
              return (
                <motion.circle
                  key={`lantern-${island.key}`}
                  cx={p.x + 22}
                  cy={p.y - 20}
                  r="16"
                  fill={`url(#${uid}-glow)`}
                  initial={false}
                  animate={{ opacity: Math.min(1, (dusk / MAX_DARK) * 1.3) }}
                  transition={{ duration: 8 }}
                />
              );
            })}
        </motion.g>

        <AnimatePresence>
          {stormy && <Storm key="storm" still={!!reduced} fierce={weather?.state === "tempest"} />}
        </AnimatePresence>

        {weather && <WindRose weather={weather} />}

        <AnimatePresence>
          {fogSurge && (
            <motion.rect
              key="surge"
              x="0"
              y="0"
              width={CHART_W}
              height={CHART_H}
              fill="#efe2bf"
              initial={{ opacity: 0 }}
              animate={{ opacity: [0, 0.85, 0.7] }}
              exit={{ opacity: 0, transition: { duration: 1.6 } }}
              transition={{ duration: 0.9 }}
            />
          )}
        </AnimatePresence>

        <rect x="0" y="0" width={CHART_W} height={CHART_H} fill={`url(#${uid}-vignette)`} pointerEvents="none" />
        <rect x="14" y="14" width={CHART_W - 28} height={CHART_H - 28} fill="none" stroke={INK} strokeOpacity="0.55" strokeWidth="2" />
      </svg>
    </div>
  );
}

const SeaDecor = memo(function SeaDecor() {
  const waves = useMemo(() => {
    const rand = seeded(77);
    return Array.from({ length: 34 }, () => ({ x: 50 + rand() * 900, y: 50 + rand() * 500 }));
  }, []);
  return (
    <g pointerEvents="none">
      {waves.map((w, i) => (
        <path key={i} d={`M${w.x.toFixed(0)} ${w.y.toFixed(0)} q7 -7 14 0 t14 0`} stroke={INK} strokeOpacity="0.28" strokeWidth="1.4" fill="none" />
      ))}
      <text x="48" y="66" fill={INK} opacity="0.85" fontSize="26" style={{ fontFamily: "var(--font-pirata)" }}>
        The Protocol Sea
      </text>
    </g>
  );
});

function DriftingFog({ uid }: { uid: string }) {
  const banks = [
    { cx: 260, cy: 140, rx: 220, ry: 60, dur: 38 },
    { cx: 760, cy: 470, rx: 260, ry: 70, dur: 46 },
    { cx: 520, cy: 300, rx: 200, ry: 50, dur: 54 },
  ];
  return (
    <g pointerEvents="none" opacity="0.35">
      {banks.map((b, i) => (
        <motion.ellipse
          key={i}
          cx={b.cx}
          cy={b.cy}
          rx={b.rx}
          ry={b.ry}
          fill={`url(#${uid}-fog)`}
          animate={{ x: [-60, 60, -60] }}
          transition={{ duration: b.dur, repeat: Infinity, ease: "easeInOut" }}
        />
      ))}
    </g>
  );
}

const RAIN = (() => {
  const rand = seeded(5);
  return Array.from({ length: 90 }, () => ({ x: rand() * 1100 - 50, y: rand() * 660 - 60, len: 14 + rand() * 16 }));
})();

function Storm({ still, fierce }: { still: boolean; fierce: boolean }) {
  return (
    <motion.g
      pointerEvents="none"
      initial={{ opacity: 0 }}
      animate={{ opacity: 1 }}
      exit={{ opacity: 0, transition: { duration: 1.2 } }}
    >
      <rect width={CHART_W} height={CHART_H} fill="#1b2a35" opacity={fierce ? 0.5 : 0.35} />
      {!still && (
        <motion.g
          stroke="#dfe9ee"
          strokeOpacity="0.55"
          strokeWidth={fierce ? 1.8 : 1.4}
          animate={{ y: [0, 60] }}
          transition={{ duration: fierce ? 0.2 : 0.3, repeat: Infinity, ease: "linear" }}
        >
          {RAIN.map((drop, i) => (
            <line key={i} x1={drop.x} y1={drop.y} x2={drop.x - drop.len * 0.35} y2={drop.y + drop.len} />
          ))}
        </motion.g>
      )}
      {fierce && !still && <LightningBolt />}
      <motion.rect
        width={CHART_W}
        height={CHART_H}
        fill="#fffbe8"
        initial={{ opacity: 0 }}
        animate={{ opacity: [0, 0.75, 0, 0.45, 0] }}
        transition={{
          duration: 0.9,
          times: [0, 0.1, 0.3, 0.4, 1],
          ...(fierce && !still ? { repeat: Infinity, repeatDelay: 5.5 } : {}),
        }}
      />
    </motion.g>
  );
}

const BOLT = "M612 0 L586 120 L622 128 L570 270 L604 278 L548 420";

function LightningBolt() {
  return (
    <motion.g
      initial={{ opacity: 0 }}
      animate={{ opacity: [0, 1, 0, 0.8, 0] }}
      transition={{ duration: 0.9, times: [0, 0.08, 0.25, 0.35, 1], repeat: Infinity, repeatDelay: 5.5 }}
    >
      <path d={BOLT} fill="none" stroke="#fffbe8" strokeWidth="9" strokeOpacity="0.35" strokeLinejoin="bevel" />
      <path d={BOLT} fill="none" stroke="#fffbe8" strokeWidth="3" strokeLinejoin="bevel" />
      <path d="M604 278 L640 330 L628 370" fill="none" stroke="#fffbe8" strokeWidth="2" />
    </motion.g>
  );
}

interface IslandMarkProps {
  interactive: boolean;
  uid: string;
  island: PublicIsland;
  charted: boolean;
  hidden: boolean;
  selected: boolean;
  solved: boolean;
  onSelect: (key: string) => void;
}

const IslandMark = memo(function IslandMark({ interactive, uid, island, charted, hidden, selected, solved, onSelect }: IslandMarkProps) {
  const { x, y } = chartPoint(island);
  const isVault = island.key === VAULT_KEY;
  const radius = isVault ? 40 : 34;
  const d = useMemo(() => islandPath(island.key, radius), [island.key, radius]);
  const label = charted ? island.name : "Uncharted";
  const labelW = label.length * 10 + 26;

  return (
    <g
      transform={`translate(${x} ${y})`}
      {...(interactive
        ? {
            role: "button",
            tabIndex: 0,
            "aria-label": charted ? `${island.name}${solved ? ", solved" : ""}` : "Uncharted island shrouded in fog",
            "aria-pressed": selected,
            onClick: () => onSelect(island.key),
            onKeyDown: (e: React.KeyboardEvent) => {
              if (e.key === "Enter" || e.key === " ") {
                e.preventDefault();
                onSelect(island.key);
              }
            },
          }
        : { "aria-hidden": true })}
      className={cn("outline-none [&:focus-visible_.focus-ring]:opacity-100", interactive && "cursor-pointer")}
    >
      <circle r={radius + 26} fill="transparent" />
      {isVault && (
        <motion.circle
          r={radius + 30}
          fill={`url(#${uid}-glow)`}
          animate={{ opacity: hidden ? [0.15, 0.3, 0.15] : [0.5, 0.9, 0.5] }}
          transition={{ duration: 3.5, repeat: Infinity }}
        />
      )}

      {!hidden && (
        <motion.g
          initial={false}
          animate={{ opacity: charted ? 1 : 0.45 }}
          whileHover={{ scale: 1.06 }}
          transition={{ duration: 0.6 }}
        >
          <path
            d={d}
            fill={charted ? (isVault ? "#c9a25c" : LAND) : "#e2cd9c"}
            stroke={INK}
            strokeWidth={charted ? 2.2 : 1.5}
            strokeDasharray={charted ? undefined : "5 5"}
          />
          {charted && <IslandDetail kind={island.kind} isVault={isVault} />}
        </motion.g>
      )}

      <AnimatePresence>
        {!charted && (
          <motion.g key="fog" initial={{ opacity: 1 }} exit={{ opacity: 0, scale: 1.6, transition: { duration: 1.4 } }}>
            <ellipse rx={radius * 1.9} ry={radius * 1.2} fill={`url(#${uid}-fog)`} />
            <text
              y="10"
              textAnchor="middle"
              fontSize="30"
              fill={INK}
              opacity="0.55"
              style={{ fontFamily: "var(--font-pirata)" }}
            >
              ?
            </text>
          </motion.g>
        )}
      </AnimatePresence>

      {solved && (
        <motion.path
          d="M-11 -11 L11 11 M11 -11 L-11 11"
          stroke="#8e1f1a"
          strokeWidth="5"
          strokeLinecap="round"
          initial={{ scale: 0, opacity: 0 }}
          animate={{ scale: 1, opacity: 1 }}
          transition={{ type: "spring", stiffness: 320, damping: 14 }}
        />
      )}

      {selected && (
        <motion.circle
          r={radius + 18}
          fill="none"
          stroke="#8e1f1a"
          strokeWidth="2.2"
          strokeDasharray="6 7"
          animate={{ rotate: 360 }}
          transition={{ duration: 18, repeat: Infinity, ease: "linear" }}
        />
      )}
      <circle className="focus-ring" r={radius + 22} fill="none" stroke="#8e1f1a" strokeWidth="2" opacity="0" />

      {!hidden && (
        <g transform={`translate(0 ${radius + 24})`} pointerEvents="none">
          <rect x={-labelW / 2} y="-13" width={labelW} height="24" rx="3" fill="#f1e2bb" stroke={INK} strokeOpacity="0.55" />
          <text
            y="5"
            textAnchor="middle"
            fontSize="17"
            fill={INK}
            opacity={charted ? 1 : 0.6}
            style={{ fontFamily: "var(--font-pirata)" }}
          >
            {label}
          </text>
        </g>
      )}
    </g>
  );
});

function IslandDetail({ kind, isVault }: { kind: PublicIsland["kind"]; isVault: boolean }) {
  if (isVault) {
    return (
      <g stroke={INK} strokeWidth="1.8" fill="none">
        <path d="M-12 8 V-6 a12 12 0 0 1 24 0 V8 Z" fill="#2b1b0d" fillOpacity="0.85" />
        <circle cx="0" cy="0" r="2.4" fill="#ffd873" stroke="none" />
      </g>
    );
  }
  return (
    <g stroke={INK} strokeWidth="1.5" fill="none" opacity="0.8">
      {kind === "TRAP" ? (
        <path d="M-8 4 l8 -14 l8 14 M-2 -2 h4" />
      ) : (
        <path d="M2 6 v-14 M2 -8 q-8 -2 -11 5 M2 -8 q8 -2 11 5 M2 -8 q-3 -8 -10 -8 M2 -8 q3 -8 10 -8" />
      )}
    </g>
  );
}

/** The widow's chart: a salt-drawn course from the sailor to where they should go next. */
function CourseLine({ from, to, still }: { from: { x: number; y: number } | null; to: { x: number; y: number } | null; still: boolean }) {
  if (!from || !to || (from.x === to.x && from.y === to.y)) return null;
  const d = lanePath(from, to, 0.22);
  return (
    <g pointerEvents="none">
      <path d={d} stroke="#f5e8c6" strokeOpacity="0.7" strokeWidth="7" fill="none" strokeLinecap="round" />
      <motion.path
        d={d}
        stroke="#1f5566"
        strokeWidth="2.6"
        strokeDasharray="10 8"
        fill="none"
        strokeLinecap="round"
        initial={{ pathLength: 0, strokeDashoffset: 0 }}
        animate={still ? { pathLength: 1 } : { pathLength: 1, strokeDashoffset: [0, -36] }}
        transition={still ? { duration: 0 } : { pathLength: { duration: 1.6, ease: "easeInOut" }, strokeDashoffset: { duration: 2, repeat: Infinity, ease: "linear" } }}
      />
      <circle cx={to.x} cy={to.y} r="46" fill="none" stroke="#1f5566" strokeOpacity="0.55" strokeWidth="1.6" strokeDasharray="4 6" />
    </g>
  );
}

interface SailorShipProps {
  player: PublicPlayer;
  index: number;
  isMe: boolean;
  islandKey: string | null;
  islands: PublicIsland[];
  sailing: boolean;
  callout: MapCallout | null;
  still: boolean;
}

function SailorShip({ player, index, isMe, islandKey, islands, sailing, callout, still }: SailorShipProps) {
  const island = islands.find((i) => i.key === islandKey);
  if (!island) return null;
  const base = chartPoint(island);
  const angle = (index / 6) * Math.PI * 2 - Math.PI / 2;
  const offset = isMe ? { x: -36, y: -34 } : { x: Math.cos(angle) * 52, y: Math.sin(angle) * 40 - 8 };
  const color = sailorColor(index, isMe);
  const faded = !player.isOnline || player.isEliminated;

  return (
    <motion.g
      initial={false}
      animate={{ x: base.x + offset.x, y: base.y + offset.y, opacity: faded ? 0.45 : 1 }}
      transition={{ duration: sailing ? 1.5 : 0.9, ease: EASE }}
      pointerEvents="none"
      aria-hidden
    >
      {sailing && <ShipWake still={still} />}
      <motion.g animate={{ rotate: [-4, 4, -4], y: [0, -2, 0] }} transition={{ duration: 3 + index * 0.3, repeat: Infinity }}>
        {player.isEliminated ? (
          <g stroke={INK} strokeWidth="1.6" fill="#6b5a44">
            <path d="M-14 4 q14 8 28 0 l-4 6 h-20 Z" />
            <path d="M-2 4 l6 -14" fill="none" />
          </g>
        ) : (
          <g>
            <path d="M-15 3 q15 9 30 0 l-4 7 h-22 Z" fill="#5a3a1c" stroke={INK} strokeWidth="1.4" />
            <path d="M0 3 v-24" stroke={INK} strokeWidth="1.6" />
            <path d="M1 -20 q11 7 0 17 Z" fill="#f4e7c7" stroke={INK} strokeWidth="1.2" />
            <path d="M-1 -18 q-9 6 0 14 Z" fill="#efe0bb" stroke={INK} strokeWidth="1.1" />
            {player.outfit ? (
              <>
                <path d="M-12 6 q12 6.5 24 0" stroke={color} strokeWidth="2.2" fill="none" />
                <g transform="translate(0.6 -29) scale(0.24)">
                  <FlagCloth flag={player.outfit.flag} waving={!still} edge={INK} />
                </g>
              </>
            ) : (
              <path d="M0 -24 l9 3 l-9 3 Z" fill={color} stroke={INK} strokeWidth="0.8" />
            )}
          </g>
        )}
      </motion.g>
      <text
        y="24"
        textAnchor="middle"
        fontSize={isMe ? 13 : 11}
        fontWeight="700"
        fill={isMe ? "#8e1f1a" : INK}
        style={{ fontFamily: "var(--font-cinzel)", letterSpacing: "0.08em" }}
      >
        {isMe ? "YOU" : player.username.slice(0, 10).toUpperCase()}
      </text>
      <AnimatePresence>{callout && <Callout key={callout.id} text={callout.text} call={callout.call} />}</AnimatePresence>
    </motion.g>
  );
}
