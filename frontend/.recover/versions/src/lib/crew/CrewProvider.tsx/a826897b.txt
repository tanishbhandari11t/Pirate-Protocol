  useCallback,

  useMemo,

import { connectSocket, getSocket, onServerEvent, request, type RequestResult } from "../socket/client";
import type {

  JoinCrewPayload,

import { CAPABILITIES } from "../socket/backend";
import { outfitStore, profileStore, savedOutfit, seatStore } from "../storage";
import {
  crewReducer,
  describeSettingsChange,
  formatClock,
  initialCrewState,
  selectMe,

/* ------------------------------------------------------------------ */
/* Public API                                                          */
/* ------------------------------------------------------------------ */

interface CrewApi {
  state: CrewState;
  me: PlayerSnapshot | null;
  isCaptain: boolean;
  /** The voyage as the game screen reads it; `null` until the crew sets sail. */
  voyage: RoomState | null;
  /** This sailor's place in the voyage. */
  sailor: PublicPlayer | null;
  /** Requests currently awaiting the server, so buttons can show a spinner. */
  pending: ReadonlySet<ClientRequestName>;

  createCrew: (payload: CreateCrewPayload) => Promise<RequestResult<SeatGrant>>;
  joinCrew: (payload: JoinCrewPayload) => Promise<RequestResult<SeatGrant>>;
  rejoinCrew: (roomCode: RoomCode) => Promise<RequestResult<SeatGrant>>;
  leaveCrew: () => Promise<void>;
  setReady: (ready: boolean) => Promise<RequestResult<PlayerSnapshot>>;
  /** Saves the outfit on this device and, while in harbour, shows it to the crew. */
  dressUp: (outfit: Outfit) => Promise<RequestResult<PlayerSnapshot | null>>;
  startVoyage: () => Promise<RequestResult<null>>;
  kickSailor: (playerId: PlayerId) => Promise<RequestResult<null>>;
  configureVoyage: (patch: Partial<VoyageSettings>) => Promise<RequestResult<VoyageSettings>>;

  move: (islandKey: string) => Promise<RequestResult<MoveResult>>;
  submitAnswer: (puzzleKey: string, answer: string) => Promise<RequestResult<AnswerVerdict>>;
  trade: (toPlayerId: PlayerId, itemKey: string) => Promise<RequestResult<TradeReceipt>>;
  buyHint: (puzzleKey: string, payWith: HintPayment) => Promise<RequestResult<HintGrant>>;
  follow: (playerId: PlayerId | null) => Promise<RequestResult<FollowResult>>;

  sendChat: (text: string) => Promise<RequestResult<ChatMessage>>;
  sendCall: (callId: QuickCallId) => Promise<RequestResult<ChatMessage>>;
  /** Tell the provider whether the chat panel is on screen, so unread counts stay honest. */
  setChatOpen: (open: boolean) => void;
}

const CrewContext = createContext<CrewApi | null>(null);

const SYNC_ATTEMPTS = 3;
const SYNC_BACKOFF_MS = 600;

function localError<T>(code: ClientErrorCode, message?: string): RequestResult<T> {
  return { ok: false, error: { code, message: message ?? describeError(code) } };
}

const wait = (ms: number) => new Promise<void>((resolve) => setTimeout(resolve, ms));

/** Dresses a crew request in the saved outfit, unless the server has no use for one. */
function withOutfit<T extends CreateCrewPayload | JoinCrewPayload>(payload: T): T {
  if (!CAPABILITIES.outfits) return payload;
  return { ...payload, outfit: payload.outfit ?? savedOutfit() };
}

/* ------------------------------------------------------------------ */
/* Provider                                                            */
/* ------------------------------------------------------------------ */

export function CrewProvider({ children }: { children: ReactNode }) {
  const [state, dispatch] = useReducer(crewReducer, initialCrewState);
  const [pending, setPending] = useState<ReadonlySet<ClientRequestName>>(() => new Set());
  const { notify } = useNotify();
  const stateRef = useRef(state);
  const inFlight = useRef(new Set<ClientRequestName>());

  useEffect(() => {
    stateRef.current = state;
  });

  /** Runs one request at a time per event name, so a double-click never sends two orders. */
  const exclusive = useCallback(
    async <T,>(name: ClientRequestName, run: () => Promise<RequestResult<T>>): Promise<RequestResult<T>> => {
      if (inFlight.current.has(name)) {
        return localError<T>("RATE_LIMITED", "Hold fast — that order is already underway.");
      }
      inFlight.current.add(name);
      setPending(new Set(inFlight.current));
      try {
        const res = await run();
        if (!res.ok && isSeatLost(res.error.code) && stateRef.current.room) {
          seatStore.clear();
          dispatch({ type: "reset" });
        }
        return res;
      } finally {
        inFlight.current.delete(name);
        setPending(new Set(inFlight.current));
      }
    },
    [],
  );

  const seat = useCallback((grant: SeatGrant) => {
    const me = grant.room.players.find((p) => p.id === grant.playerId);
    seatStore.set({ roomCode: grant.room.code, playerId: grant.playerId, sessionToken: grant.sessionToken });
    if (me) profileStore.set({ playerName: me.name, avatarId: me.avatarId });
    dispatch({ type: "seated", grant, at: Date.now() });
  }, []);

  /* ---------------- Phase 1: crews & lobby ---------------- */

  const createCrew = useCallback(
    (payload: CreateCrewPayload) =>
      exclusive("crew:create", async () => {
        const res = await request("crew:create", withOutfit(payload));
        if (res.ok) seat(res.data);
        return res;
      }),
    [exclusive, seat],
  );

  const joinCrew = useCallback(
    (payload: JoinCrewPayload) =>
      exclusive("crew:join", async () => {
        const res = await request("crew:join", withOutfit({ ...payload, roomCode: payload.roomCode.toUpperCase() }));
        if (res.ok) seat(res.data);
        return res;
      }),
    [exclusive, seat],
  );

  const rejoinCrew = useCallback(
    async (roomCode: RoomCode): Promise<RequestResult<SeatGrant>> => {
      const stored = seatStore.get();
      if (!stored || stored.roomCode !== roomCode) return localError("SESSION_EXPIRED");
      const res = await request("crew:rejoin", { roomCode, sessionToken: stored.sessionToken });
      if (res.ok) seat(res.data);
      else if (!isRetryable(res.error.code)) seatStore.clear();
      return res;
    },
    [seat],
  );

  const leaveCrew = useCallback(async () => {
    if (stateRef.current.room) await request("crew:leave", {});
    seatStore.clear();
    dispatch({ type: "reset" });
  }, []);

  const setReady = useCallback(
    (ready: boolean) => exclusive("player:ready", () => request("player:ready", { ready })),
    [exclusive],
  );

  const dressUp = useCallback(
    async (outfit: Outfit): Promise<RequestResult<PlayerSnapshot | null>> => {
      const clean = sanitizeOutfit(outfit);
      if (!clean) return localError("INVALID_PAYLOAD", "That outfit has a piece the tailor does not recognise.");
      outfitStore.set(clean);
      const room = stateRef.current.room;
      if (!CAPABILITIES.outfits || !room || room.phase !== "lobby") return { ok: true, data: null };
      return exclusive("player:outfit", () => request("player:outfit", { outfit: clean }));
    },
    [exclusive],
  );

  const startVoyage = useCallback(() => exclusive("game:start", () => request("game:start", {})), [exclusive]);

  const kickSailor = useCallback(
    (playerId: PlayerId) => {
      if (playerId === stateRef.current.playerId) return Promise.resolve(localError<null>("INVALID_PAYLOAD"));
      return exclusive("crew:kick", () => request("crew:kick", { playerId }));
    },
    [exclusive],
  );

  const configureVoyage = useCallback(
    (patch: Partial<VoyageSettings>) => {
      const room = stateRef.current.room;
      if (!room) return Promise.resolve(localError<VoyageSettings>("NOT_IN_ROOM"));
      if (room.captainId !== stateRef.current.playerId) return Promise.resolve(localError<VoyageSettings>("NOT_CAPTAIN"));
      if (room.phase !== "lobby") return Promise.resolve(localError<VoyageSettings>("GAME_IN_PROGRESS"));
      if (!mergeSettings(room.settings, patch)) return Promise.resolve(localError<VoyageSettings>("INVALID_PAYLOAD"));
      return exclusive("crew:configure", () => request("crew:configure", { settings: patch }));
    },
    [exclusive],
  );

  /* ---------------- Phase 2: the voyage ---------------- */

  const move = useCallback(
    (islandKey: string) => {
      const voyage = stateRef.current.room?.voyage;
      if (!voyage) return Promise.resolve(localError<MoveResult>("NOT_SAILING"));
      if (!voyage.islands.some((i) => i.key === islandKey)) {
        return Promise.resolve(localError<MoveResult>("ISLAND_NOT_FOUND"));
      }
      return exclusive("map:move", () => request("map:move", { islandKey }));
    },
    [exclusive],
  );

  const submitAnswer = useCallback(
    (puzzleKey: string, answer: string) => {
      const cleaned = answer.replace(/\s+/g, " ").trim().slice(0, ANSWER_MAX);
      if (!cleaned) return Promise.resolve(localError<AnswerVerdict>("INVALID_PAYLOAD", "Write an answer first."));
      if (!stateRef.current.room?.voyage) return Promise.resolve(localError<AnswerVerdict>("NOT_SAILING"));
      return exclusive("puzzle:submit", () => request("puzzle:submit", { puzzleKey, answer: cleaned }));
    },
    [exclusive],
  );

  const trade = useCallback(
    (toPlayerId: PlayerId, itemKey: string) => {
      if (!isTradableKey(itemKey)) return Promise.resolve(localError<TradeReceipt>("NOT_TRADABLE"));
      if (toPlayerId === stateRef.current.playerId) return Promise.resolve(localError<TradeReceipt>("BAD_RECIPIENT"));
      return exclusive("trade:offer", () => request("trade:offer", { toPlayerId, itemKey }));
    },
    [exclusive],
  );

  /* ---------------- Phase 3: crew life ---------------- */

  const buyHint = useCallback(
    (puzzleKey: string, payWith: HintPayment) => {
      const room = stateRef.current.room;
      if (!room?.voyage) return Promise.resolve(localError<HintGrant>("NOT_SAILING"));
      if (!room.settings.hintsEnabled) return Promise.resolve(localError<HintGrant>("HINTS_DISABLED"));
      if (payWith === "tide-rumor" && !stateRef.current.you.some((i) => i.itemKey === "tide-rumor")) {
        return Promise.resolve(localError<HintGrant>("CANNOT_AFFORD", "You hold no Tide Rumor to pay with."));
      }
      return exclusive("puzzle:hint", () => request("puzzle:hint", { puzzleKey, payWith }));
    },
    [exclusive],
  );

  const follow = useCallback(
    (playerId: PlayerId | null) => {
      if (playerId === stateRef.current.playerId) return Promise.resolve(localError<FollowResult>("BAD_RECIPIENT"));
      return exclusive("spectate:follow", () => request("spectate:follow", { playerId }));
    },
    [exclusive],
  );

  const sendChat = useCallback(
    (text: string) => {
      const cleaned = text.replace(/\s+/g, " ").trim().slice(0, CHAT_MAX);
      if (!cleaned) return Promise.resolve(localError<ChatMessage>("INVALID_PAYLOAD", "Say something first."));
      return exclusive("chat:send", async () => {
        const res = await request("chat:send", { text: cleaned });
        if (res.ok) dispatch({ type: "chat-message", message: res.data });
        return res;
      });
    },
    [exclusive],
  );

  const sendCall = useCallback(
    (callId: QuickCallId) =>
      exclusive("chat:send", async () => {
        const res = await request("chat:send", { callId });
        if (res.ok) dispatch({ type: "chat-message", message: res.data });
        return res;
      }),
    [exclusive],
  );

  const setChatOpen = useCallback((open: boolean) => dispatch({ type: "chat-visibility", open }), []);

  /** Asks the server for missed events, retrying transport hiccups with a short backoff. */
  const syncVoyage = useCallback(async () => {
    for (let attempt = 1; attempt <= SYNC_ATTEMPTS; attempt += 1) {
      const res: RequestResult<SyncResult> = await request("voyage:sync", { sinceSeq: stateRef.current.lastSeq });
      if (res.ok) {
        dispatch({ type: "events-replayed", events: res.data.events, lastSeq: res.data.lastSeq });
        return;
      }
      if (!isRetryable(res.error.code)) return;
      await wait(SYNC_BACKOFF_MS * attempt);
    }
  }, []);

  useEffect(() => {
    if (!state.needsSync) return;
    dispatch({ type: "sync-started" });
    void syncVoyage();
  }, [state.needsSync, syncVoyage]);

  /* ---------------- Server pushes & connection ---------------- */

  useEffect(() => {
    connectSocket();
    const at = () => Date.now();

    const unsubscribers = [
      onServerEvent("room:state", (room) => dispatch({ type: "room-state", room })),
      onServerEvent("room:player-joined", ({ player }) => dispatch({ type: "player-joined", player, at: at() })),
      onServerEvent("room:player-left", ({ playerId, name, reason }) => {
        if (playerId === stateRef.current.playerId) {
          seatStore.clear();
          dispatch({ type: "reset" });
          notify({
            tone: "danger",
            title: reason === "kicked" ? "Cast overboard" : "Left the ship",
            message: reason === "kicked" ? "The captain removed you from the crew." : undefined,
          });
          return;
        }
        dispatch({ type: "player-left", playerId, name, reason, at: at() });
      }),
      onServerEvent("room:player-updated", ({ player }) => dispatch({ type: "player-updated", player, at: at() })),
      onServerEvent("room:captain-changed", ({ captainId }) => {
        dispatch({ type: "captain-changed", captainId, at: at() });
        if (captainId === stateRef.current.playerId) {
          notify({ tone: "warning", title: "You hold the helm", message: "The crew now answers to you, Captain." });
        }
      }),
      onServerEvent("room:settings-changed", ({ settings, changedBy }) => {
        const previous = stateRef.current.room?.settings;
        dispatch({ type: "settings-changed", settings, changedBy, at: at() });
        if (previous && changedBy !== stateRef.current.playerId) {
          const changes = describeSettingsChange(previous, settings);
          if (changes.length > 0) {
            notify({
              tone: "info",
              title: "The articles were rewritten",
              message: `${changes.join(", ")}. Ready up again when you agree.`,
            });
          }
        }
      }),
      onServerEvent("game:starting", ({ startsAt, seconds }) =>
        dispatch({ type: "voyage-starting", startsAt, seconds, at: at() }),
      ),
      onServerEvent("game:started", () => dispatch({ type: "voyage-started", at: at() })),
      onServerEvent("game:event", (event) => {
        if (event.roomCode !== stateRef.current.room?.code) return;
        dispatch({ type: "game-event", event });
        if (isGameEvent(event, "TIME_WARNING")) {
          const m = event.payload.minutesLeft;
          notify({
            tone: m <= 1 ? "danger" : "warning",
            title: "The tide is turning",
            message: `${m} ${m === 1 ? "minute remains" : "minutes remain"}.`,
          });
        }
      }),
      onServerEvent("game:finished", ({ roomCode, winnerPlayerId, finishedAt, reason, scores }) => {
        if (roomCode !== stateRef.current.room?.code) return;
        dispatch({ type: "voyage-finished", winnerPlayerId, reason, scores, at: finishedAt });
      }),
      onServerEvent("chat:message", (message) => {
        const current = stateRef.current;
        if (message.roomCode !== current.room?.code) return;
        dispatch({ type: "chat-message", message });
        if (message.kind === "call" && message.playerId !== current.playerId && !current.chatOpen) {
          const who = current.room.players.find((p) => p.id === message.playerId)?.name ?? "A sailor";
          notify({ tone: message.callId === "trap-ahead" ? "warning" : "info", title: who, message: message.text });
        }
      }),
      onServerEvent("server:notice", ({ level, message }) =>
        notify({ tone: level, title: level === "info" ? "Word from the harbour" : "Harbour warning", message }),
      ),
    ];

    const socket = getSocket();
    let hasConnectedBefore = socket.connected;

    const onConnect = async () => {
      const current = stateRef.current.room;
      if (hasConnectedBefore && current) {
        const res = await rejoinCrew(current.code);
        if (!res.ok) {
          dispatch({ type: "reset" });
          notify({
            tone: errorTone(res.error.code),
            title: errorTitle(res.error.code),
            message: describeError(res.error.code, res.error.message),
          });
        } else {
          notify({ tone: "success", title: "Back aboard", message: "The connection to your crew ship was restored." });
        }
      }
      hasConnectedBefore = true;
    };

    const onDisconnect = (reason: string) => {
      if (reason === "io client disconnect" || !stateRef.current.room) return;
      notify({
        tone: "warning",
        title: "Connection to the crew ship lost…",
        message: "Hold fast. Your lantern is signalling for a reconnect.",
      });
    };

    /** Phones freeze background tabs; nudge the socket awake when the player returns. */
    const onVisible = () => {
      if (document.visibilityState === "visible" && !socket.connected) connectSocket();
    };

    socket.on("connect", onConnect);
    socket.on("disconnect", onDisconnect);
    document.addEventListener("visibilitychange", onVisible);

    return () => {
      unsubscribers.forEach((off) => off());
      socket.off("connect", onConnect);
      socket.off("disconnect", onDisconnect);
      document.removeEventListener("visibilitychange", onVisible);
    };
  }, [notify, rejoinCrew]);

  /* ---------------- Derived values ---------------- */

  const me = useMemo(() => selectMe(state), [state]);
  const voyage = useMemo(
    () => toVoyageRoom(state.room, state.you, state.hints),
    [state.room, state.you, state.hints],
  );
  const sailor = useMemo(
    () => voyage?.players.find((p) => p.id === state.playerId) ?? null,
    [voyage, state.playerId],
  );

  const api = useMemo<CrewApi>(
    () => ({
      state,
      me,
      isCaptain: !!me && state.room?.captainId === me.id,
      voyage,
      sailor,
      pending,
      createCrew,
      joinCrew,
      rejoinCrew,
      leaveCrew,
      setReady,
      dressUp,
      startVoyage,
      kickSailor,
      configureVoyage,
      move,
      submitAnswer,
      trade,
      buyHint,
      follow,
      sendChat,
      sendCall,
      setChatOpen,
    }),
    [
      state,
      me,
      voyage,
      sailor,
      pending,
      createCrew,
      joinCrew,
      rejoinCrew,
      leaveCrew,
      setReady,
      dressUp,
      startVoyage,
      kickSailor,
      configureVoyage,
      move,
      submitAnswer,
      trade,
      buyHint,
      follow,
      sendChat,
      sendCall,
      setChatOpen,
    ],
  );

  return <CrewContext.Provider value={api}>{children}</CrewContext.Provider>;
}

export function useCrew() {
  const ctx = useContext(CrewContext);
  if (!ctx) throw new Error("useCrew must be used inside <CrewProvider>");
  return ctx;
}

/* ------------------------------------------------------------------ */
/* Voyage clock                                                        */
/* ------------------------------------------------------------------ */

/** One shared one-second ticker, running only while something on screen reads the clock. */
const clockListeners = new Set<() => void>();
let clockTimer: ReturnType<typeof setInterval> | null = null;
let clockNow = 0;

const clockStore = {
  subscribe(listener: () => void) {
    clockListeners.add(listener);
    if (!clockTimer) {
      clockNow = Date.now();
      clockTimer = setInterval(() => {
        clockNow = Date.now();
        clockListeners.forEach((l) => l());
      }, 1000);
    }
    return () => {
      clockListeners.delete(listener);
      if (clockListeners.size === 0 && clockTimer) {
        clearInterval(clockTimer);
        clockTimer = null;
      }
    };
  },
  getSnapshot: () => clockNow,
  getServerSnapshot: () => 0,
};

export interface VoyageClock {
  /** Milliseconds left on a timed voyage; `null` when untimed or not sailing. */
  remainingMs: number | null;
  elapsedMs: number | null;
  /** Under a minute left. */
  urgent: boolean;
  /** Ready-to-render text: the countdown on timed voyages, time at sea otherwise. */
  label: string | null;
}

export function useVoyageClock(): VoyageClock {
  const { state } = useCrew();
  const now = useSyncExternalStore(clockStore.subscribe, clockStore.getSnapshot, clockStore.getServerSnapshot);
  const voyage = state.room?.voyage;
  if (!voyage || !now) return { remainingMs: null, elapsedMs: null, urgent: false, label: null };

  const reference = voyage.finishedAt ?? now;
  const elapsedMs = Math.max(0, reference - voyage.startedAt);
  const remainingMs = voyage.endsAt ? Math.max(0, voyage.endsAt - reference) : null;
  return {
    remainingMs,
    elapsedMs,
    urgent: remainingMs !== null && remainingMs < 60_000,
    label: remainingMs !== null ? formatClock(remainingMs) : formatClock(elapsedMs),
  };
}

/** The wording of a quick call, for buttons that preview what the crew will see. */
export function quickCallText(callId: QuickCallId) {
  return QUICK_CALLS[callId];
}
