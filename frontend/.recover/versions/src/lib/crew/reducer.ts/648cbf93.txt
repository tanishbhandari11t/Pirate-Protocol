export type LogTone =
  | "info"
  | "join"
  | "leave"
  | "ready"
  | "captain"
  | "voyage"
  | "move"
  | "discover"
  | "solve"
  | "fail"
  | "trap"
  | "item"
  | "trade"
  | "treasure"
  | "hint"
  | "settings"
  | "watch"
  | "clock";

export interface LogEntry {
  id: number;
  at: number;
  tone: LogTone;
  text: string;
  /** Set when the entry came from a server `game:event`, so replays never double-log it. */
  eventId?: string;
}

export interface VoyageOutcome {
  winnerPlayerId: PlayerId | null;
  reason: FinishReason;
  scores: ScoreEntry[];
  at: number;
}

export interface CrewState {
  room: RoomSnapshot | null;
  playerId: PlayerId | null;
  /** This sailor's private hold, delivered with every `room:state`. */
  you: InventoryItem[];
  /** Hints this sailor has bought, delivered with every `room:state`. */
  hints: HintEntry[];
  countdown: { startsAt: number; seconds: number } | null;
  voyageStarted: boolean;
  voyageFinished: boolean;
  outcome: VoyageOutcome | null;
  log: LogEntry[];
  logSeq: number;
  /** Ids of voyage events already written to the log. */
  seenEvents: string[];
  /** Highest voyage event `seq` applied; a jump past `lastSeq + 1` means something was missed. */
  lastSeq: number;
  needsSync: boolean;
  chat: ChatMessage[];
  chatOpen: boolean;
  unreadChat: number;
}

export const initialCrewState: CrewState = {
  room: null,
  playerId: null,
  you: [],
  hints: [],
  countdown: null,
  voyageStarted: false,
  voyageFinished: false,
  outcome: null,
  log: [],
  logSeq: 0,
  seenEvents: [],
  lastSeq: 0,
  needsSync: false,
  chat: [],
  chatOpen: false,
  unreadChat: 0,
};

export type CrewAction =
  | { type: "seated"; grant: SeatGrant; at: number }
  | { type: "room-state"; room: RoomStatePush }
  | { type: "player-joined"; player: PlayerSnapshot; at: number }
  | { type: "player-left"; playerId: PlayerId; name: string; reason: LeaveReason; at: number }
  | { type: "player-updated"; player: PlayerSnapshot; at: number }
  | { type: "captain-changed"; captainId: PlayerId; at: number }
  | { type: "settings-changed"; settings: VoyageSettings; changedBy: PlayerId; at: number }
  | { type: "voyage-starting"; startsAt: number; seconds: number; at: number }
  | { type: "voyage-started"; at: number }
  | { type: "game-event"; event: GameEvent }
  | { type: "events-replayed"; events: GameEvent[]; lastSeq: number }
  | { type: "sync-started" }
  | {
      type: "voyage-finished";
      winnerPlayerId: PlayerId | null;
      reason: FinishReason;
      scores: ScoreEntry[];
      at: number;
    }
  | { type: "chat-message"; message: ChatMessage }
  | { type: "chat-visibility"; open: boolean }
  | { type: "reset" };

const LOG_LIMIT = 60;
const SEEN_LIMIT = 200;

/* ------------------------------------------------------------------ */
/* Log helpers                                                         */
/* ------------------------------------------------------------------ */

function appendLog(state: CrewState, at: number, tone: LogTone, text: string, eventId?: string): CrewState {
  const logSeq = state.logSeq + 1;
  return { ...state, logSeq, log: [...state.log, { id: logSeq, at, tone, text, eventId }].slice(-LOG_LIMIT) };
}

function nameOf(room: RoomSnapshot | null, playerId: unknown, meId: PlayerId | null) {
  if (typeof playerId !== "string") return "A sailor";
  if (playerId === meId) return "You";
  return room?.players.find((p) => p.id === playerId)?.name ?? "A departed sailor";
}

function possessive(name: string) {
  return name === "You" ? "your" : `${name}'s`;
}

function islandName(room: RoomSnapshot | null, key: unknown) {
  if (typeof key !== "string") return "an unknown shore";
  return room?.voyage?.islands.find((i) => i.key === key)?.name ?? "an unknown shore";
}

function itemName(key: unknown, you: InventoryItem[]) {
  if (typeof key !== "string") return "something";
  return you.find((i) => i.itemKey === key)?.name ?? key.replace(/-/g, " ").replace(/\b\w/g, (c) => c.toUpperCase());
}

function settingsOf(room: RoomSnapshot | null): VoyageSettings {
  return room?.settings ?? DEFAULT_SETTINGS;
}

/** Turns one server event into a line for the ship's log. Returns `null` for events the log skips. */
export function describeVoyageEvent(
  room: RoomSnapshot | null,
  event: GameEvent,
  meId: PlayerId | null,
  you: InventoryItem[] = [],
): { tone: LogTone; text: string } | null {
  const who = nameOf(room, event.playerId, meId);
  const verb = (mine: string, theirs: string) => (who === "You" ? mine : theirs);

  if (isGameEvent(event, "MOVED")) {
    return { tone: "move", text: `${who} dropped anchor at ${islandName(room, event.payload.islandKey)}.` };
  }
  if (isGameEvent(event, "ISLAND_DISCOVERED")) {
    return { tone: "discover", text: `${islandName(room, event.payload.islandKey)} emerged from the fog.` };
  }
  if (isGameEvent(event, "PUZZLE_SOLVED")) {
    return { tone: "solve", text: `${who} cracked the riddle of ${islandName(room, event.payload.islandKey)}.` };
  }
  if (isGameEvent(event, "PUZZLE_FAILED")) {
    return { tone: "fail", text: `${who} answered wrongly at ${islandName(room, event.payload.islandKey)}.` };
  }
  if (isGameEvent(event, "TRAP_TRIGGERED")) {
    const { strikes, eliminated } = event.payload;
    const limit = settingsOf(room).maxStrikes;
    const tail = eliminated ? ` ${verb("The sea has claimed you.", "The sea has claimed them.")}` : "";
    return {
      tone: "trap",
      text: `${who} sprang a trap at ${islandName(room, event.payload.islandKey)} — strike ${strikes}/${limit}.${tail}`,
    };
  }
  if (isGameEvent(event, "ITEM_GRANTED")) {
    const item = itemName(event.payload.itemKey, you);
    if (event.payload.itemKey === "cursed-coin") {
      return { tone: "trap", text: `${who} ${verb("were", "was")} handed a ${item}.` };
    }
    return { tone: "item", text: `${who} recovered the ${item}.` };
  }
  if (isGameEvent(event, "TRADED")) {
    const { fromPlayerId, toPlayerId, itemKey, quantity } = event.payload;
    const count = quantity > 1 ? `${quantity} × ` : "a ";
    return {
      tone: "trade",
      text: `${nameOf(room, fromPlayerId, meId)} passed ${count}${itemName(itemKey, you)} to ${nameOf(room, toPlayerId, meId)}.`,
    };
  }
  if (isGameEvent(event, "VAULT_AWAKENED")) {
    return { tone: "treasure", text: `Five relics hum in ${possessive(who)} pack. The Vault has awakened.` };
  }
  if (isGameEvent(event, "TREASURE_FOUND")) {
    return { tone: "treasure", text: `${who} opened the Vault. The hoard is claimed!` };
  }
  if (isGameEvent(event, "HINT_BOUGHT")) {
    const price = event.payload.paidWith === "strike" ? "a strike" : "a Tide Rumor";
    return {
      tone: "hint",
      text: `${who} paid ${price} for a whisper about ${islandName(room, event.payload.islandKey)}.`,
    };
  }
  if (isGameEvent(event, "SPECTATING")) {
    const target = event.payload.targetId;
    if (!target) return { tone: "watch", text: `${who} stopped watching from the rigging.` };
    return { tone: "watch", text: `${who} now ${verb("watch", "watches")} ${nameOf(room, target, meId)} from the rigging.` };
  }
  if (isGameEvent(event, "TIME_WARNING")) {
    const m = event.payload.minutesLeft;
    return { tone: "clock", text: `${m} ${m === 1 ? "minute remains" : "minutes remain"} before the tide turns.` };
  }
  if (isGameEvent(event, "TIME_UP")) {
    return { tone: "clock", text: "The tide has turned. The voyage is over." };
  }
  if (isGameEvent(event, "CREW_WRECKED")) {
    return { tone: "trap", text: "The sea has claimed every sailor." };
  }
  // Joins and departures are already logged from the lobby pushes.
  if (event.type === "PLAYER_JOINED" || event.type === "PLAYER_LEFT") return null;
  return { tone: "info", text: `${who}: ${event.type.toLowerCase().replace(/_/g, " ")}.` };
}

function logEvent(state: CrewState, event: GameEvent): CrewState {
  const lastSeq = Math.max(state.lastSeq, event.seq);
  if (state.seenEvents.includes(event.id)) return lastSeq === state.lastSeq ? state : { ...state, lastSeq };
  const seenEvents = [...state.seenEvents, event.id].slice(-SEEN_LIMIT);
  const line = describeVoyageEvent(state.room, event, state.playerId, state.you);
  const next = { ...state, seenEvents, lastSeq };
  if (!line) return next;
  return appendLog(next, Date.parse(event.createdAt) || Date.now(), line.tone, line.text, event.id);
}

/** After a refresh mid-voyage, rebuild the log from the events the server still remembers. */
function backfillVoyageLog(state: CrewState): CrewState {
  const events = [...(state.room?.voyage?.log ?? [])].sort((a, b) => a.seq - b.seq);
  return events.reduce(logEvent, state);
}

function minutesLabel(minutes: number) {
  return minutes ? `${minutes} minutes` : "no limit";
}

/** One readable line per article the captain changed. */
export function describeSettingsChange(previous: VoyageSettings, next: VoyageSettings): string[] {
  const changes: string[] = [];
  if (previous.maxStrikes !== next.maxStrikes) {
    changes.push(`strikes allowed ${previous.maxStrikes} → ${next.maxStrikes}`);
  }
  if (previous.trapsEnabled !== next.trapsEnabled) {
    changes.push(next.trapsEnabled ? "traps armed" : "traps disarmed");
  }
  if (previous.hintsEnabled !== next.hintsEnabled) {
    changes.push(next.hintsEnabled ? "hints for sale" : "hints forbidden");
  }
  if (previous.timeLimitMinutes !== next.timeLimitMinutes) {
    changes.push(`clock ${minutesLabel(previous.timeLimitMinutes)} → ${minutesLabel(next.timeLimitMinutes)}`);
  }
  return changes;
}

/** A one-line summary of the articles, for the lobby and the results screen. */
export function describeSettings(settings: VoyageSettings): string {
  return [
    `${settings.maxStrikes} ${settings.maxStrikes === 1 ? "strike" : "strikes"}`,
    settings.trapsEnabled ? "traps armed" : "traps disarmed",
    settings.hintsEnabled ? "hints for sale" : "no hints",
    settings.timeLimitMinutes ? `${settings.timeLimitMinutes}-minute clock` : "no clock",
  ].join(" · ");
}

/* Snapshot helpers                                                    */
/* ------------------------------------------------------------------ */

function upsertPlayer(room: RoomSnapshot, player: PlayerSnapshot): RoomSnapshot {
  const exists = room.players.some((p) => p.id === player.id);
  return {
    ...room,
    players: exists ? room.players.map((p) => (p.id === player.id ? player : p)) : [...room.players, player],
  };
}

function splitPush(push: RoomStatePush): { room: RoomSnapshot; you: InventoryItem[]; hints: HintEntry[] } {
  const { you, yourHints, ...room } = push;
  return { room, you, hints: yourHints };
}

/** Inserts a message in `seq` order, ignoring duplicates, keeping the newest `CHAT_HISTORY_LIMIT`. */
function mergeChat(chat: ChatMessage[], incoming: ChatMessage[]): ChatMessage[] {
  const byId = new Map(chat.map((m) => [m.id, m]));
  for (const m of incoming) byId.set(m.id, m);
  return [...byId.values()].sort((a, b) => a.seq - b.seq).slice(-CHAT_HISTORY_LIMIT);
}

function outcomeFrom(room: RoomSnapshot, at: number): VoyageOutcome | null {
  const v = room.voyage;
  if (!v || v.status !== "finished" || !v.finishReason) return null;
  return { winnerPlayerId: v.winnerPlayerId, reason: v.finishReason, scores: v.scores, at: v.finishedAt ?? at };
}

/* Reducer                                                             */
/* ------------------------------------------------------------------ */

export function crewReducer(state: CrewState, action: CrewAction): CrewState {
  switch (action.type) {
    case "seated": {
      const { room, playerId, chat } = action.grant;
      const me = room.players.find((p) => p.id === playerId);
      const sameShip = state.room?.code === room.code;
      const base: CrewState = {
        ...initialCrewState,
        room,
        playerId,
        you: sameShip ? state.you : [],
        hints: sameShip ? state.hints : [],
        voyageStarted: room.phase === "in-game" || room.phase === "finished",
        voyageFinished: room.phase === "finished",
        outcome: outcomeFrom(room, action.at),
        log: sameShip ? state.log : [],
        logSeq: state.logSeq,
        seenEvents: sameShip ? state.seenEvents : [],
        lastSeq: sameShip ? state.lastSeq : 0,
        chat: mergeChat(sameShip ? state.chat : [], chat),
        chatOpen: state.chatOpen,
        unreadChat: sameShip ? state.unreadChat : 0,
      };
      const seated = sameShip
        ? base
        : appendLog(base, action.at, "join", `${me?.name ?? "You"} came aboard the ${bareShipName(room.crewName)}.`);
      return backfillVoyageLog(seated);
    }

    case "room-state": {
      if (!state.room || action.room.code !== state.room.code) return state;
      const { room, you, hints } = splitPush(action.room);
      return {
        ...state,
        room,
        you,
        hints,
        countdown: room.phase === "lobby" ? null : state.countdown,
        voyageStarted: state.voyageStarted || room.phase === "in-game" || room.phase === "finished",
        voyageFinished: room.phase === "finished",
        outcome: state.outcome ?? outcomeFrom(room, Date.now()),
      };
    }

    case "player-joined": {
      if (!state.room) return state;
      const next = { ...state, room: upsertPlayer(state.room, action.player) };
      return appendLog(next, action.at, "join", `${action.player.name} climbed aboard.`);
    }

    case "player-left": {
      if (!state.room) return state;
      const room = { ...state.room, players: state.room.players.filter((p) => p.id !== action.playerId) };
      const verb =
        action.reason === "kicked"
          ? "was cast overboard"
          : action.reason === "disconnected"
            ? "vanished into the fog"
            : "went ashore";
      return appendLog({ ...state, room }, action.at, "leave", `${action.name} ${verb}.`);
    }

    case "player-updated": {
      if (!state.room) return state;
      const previous = state.room.players.find((p) => p.id === action.player.id);
      const next = { ...state, room: upsertPlayer(state.room, action.player) };
      if (previous && previous.isReady !== action.player.isReady) {
        return appendLog(
          next,
          action.at,
          "ready",
          action.player.isReady
            ? `${action.player.name} is ready to weigh anchor.`
            : `${action.player.name} needs a moment more.`,
        );
      }
      if (previous && previous.isConnected !== action.player.isConnected) {
        return appendLog(
          next,
          action.at,
          action.player.isConnected ? "join" : "leave",
          action.player.isConnected
            ? `${action.player.name} emerged from the fog.`
            : `${action.player.name} lost sight of the ship.`,
        );
      }
      return next;
    }

    case "captain-changed": {
      if (!state.room) return state;
      const room: RoomSnapshot = {
        ...state.room,
        captainId: action.captainId,
        players: state.room.players.map((p) => ({ ...p, isCaptain: p.id === action.captainId })),
      };
      const captain = room.players.find((p) => p.id === action.captainId);
      return appendLog({ ...state, room }, action.at, "captain", `${captain?.name ?? "A new sailor"} now holds the helm.`);
    }

    case "settings-changed": {
      if (!state.room) return state;
      const changes = describeSettingsChange(state.room.settings, action.settings);
      const room: RoomSnapshot = {
        ...state.room,
        settings: action.settings,
        players: state.room.players.map((p) => (p.id === action.changedBy ? p : { ...p, isReady: false })),
      };
      if (changes.length === 0) return { ...state, room };
      const who = nameOf(state.room, action.changedBy, state.playerId);
      return appendLog(
        { ...state, room },
        action.at,
        "settings",
        `${who} rewrote the articles: ${changes.join(", ")}. Every sailor must ready up again.`,
      );
    }

    case "voyage-starting": {
      const next: CrewState = {
        ...state,
        countdown: { startsAt: action.startsAt, seconds: action.seconds },
        room: state.room ? { ...state.room, phase: "starting" } : state.room,
      };
      return appendLog(next, action.at, "voyage", "The captain orders: weigh anchor!");
    }

    case "voyage-started": {
      const next: CrewState = {
        ...state,
        voyageStarted: true,
        countdown: null,
        lastSeq: 0,
        needsSync: false,
        room: state.room ? { ...state.room, phase: "in-game" } : state.room,
      };
      return appendLog(next, action.at, "voyage", "The ships leave harbour. The map awakens.");
    }

    case "game-event": {
      const gap = state.lastSeq > 0 && action.event.seq > state.lastSeq + 1;
      const next = logEvent(state, action.event);
      return gap ? { ...next, needsSync: true } : next;
    }

    case "sync-started":
      return state.needsSync ? { ...state, needsSync: false } : state;

    case "events-replayed": {
      const ordered = [...action.events].sort((a, b) => a.seq - b.seq);
      const next = ordered.reduce(logEvent, state);
      return { ...next, lastSeq: Math.max(next.lastSeq, action.lastSeq), needsSync: false };
    }

    case "voyage-finished": {
      if (state.outcome && state.voyageFinished) return state;
      const outcome: VoyageOutcome = {
        winnerPlayerId: action.winnerPlayerId,
        reason: action.reason,
        scores: action.scores,
        at: action.at,
      };
      const next: CrewState = {
        ...state,
        voyageFinished: true,
        outcome,
        room: state.room ? { ...state.room, phase: "finished" } : state.room,
      };
      if (action.reason === "wreck" || !action.winnerPlayerId) {
        const text =
          action.reason === "wreck"
            ? "The voyage ends in wreckage. No one claims the hoard."
            : "The tide turned with no clear victor.";
        return appendLog(next, action.at, action.reason === "wreck" ? "trap" : "clock", text);
      }
      const winner = nameOf(state.room, action.winnerPlayerId, state.playerId);
      const how = action.reason === "time" ? "takes the day on points" : "claimed the hoard";
      return appendLog(next, action.at, "treasure", `${winner === "You" ? "You" : winner} ${how}.`);
    }

    case "chat-message": {
      if (state.chat.some((m) => m.id === action.message.id)) return state;
      const fromOther = action.message.playerId !== state.playerId;
      return {
        ...state,
        chat: mergeChat(state.chat, [action.message]),
        unreadChat: !state.chatOpen && fromOther ? state.unreadChat + 1 : state.unreadChat,
      };
    }

    case "chat-visibility":
      return { ...state, chatOpen: action.open, unreadChat: action.open ? 0 : state.unreadChat };

    case "reset":
      return { ...initialCrewState, logSeq: state.logSeq };

    default:
      return state;
  }
}

export function selectMe(state: CrewState): PlayerSnapshot | null {
  return state.room?.players.find((p) => p.id === state.playerId) ?? null;
}

export function selectCaptain(state: CrewState): PlayerSnapshot | null {
  return state.room?.players.find((p) => p.id === state.room?.captainId) ?? null;
}

export function selectSettings(state: CrewState): VoyageSettings {
  return settingsOf(state.room);
}

export function readyCount(room: RoomSnapshot) {
  return room.players.filter((p) => p.isReady).length;
}

/**
 * Why the captain cannot give the order to sail yet, in the crew's own words. This only shapes
 * the button; the server re-checks every rule when `game:start` arrives.
 */
export function startBlockers(room: RoomSnapshot): string[] {
  const blockers: string[] = [];
  if (room.phase !== "lobby") blockers.push("The ship has already weighed anchor.");
  if (room.players.length < room.minPlayers) {
    const missing = room.minPlayers - room.players.length;
    blockers.push(`${missing} more ${missing === 1 ? "hand is" : "hands are"} needed on deck.`);
  }
  const waiting = room.players.filter((p) => !p.isReady);
  if (waiting.length > 0) {
    blockers.push(`Waiting on ${waiting.map((p) => p.name).join(", ")}.`);
  }
  if (room.players.some((p) => !p.isConnected)) blockers.push("A sailor is lost in the fog.");
  return blockers;
}

export function heldRelicCount(state: CrewState) {
  const held = new Set(state.you.map((i) => i.itemKey));
  return REQUIRED_RELICS.filter((r) => held.has(r)).length;
}

/** Hints this sailor has bought for one puzzle, gentlest first. */
export function hintsFor(state: CrewState, puzzleKey: string): HintEntry[] {
  return state.hints.filter((h) => h.puzzleKey === puzzleKey).sort((a, b) => a.level - b.level);
}

/** Sailors in the rigging currently watching `playerId`. */
export function watchersOf(state: CrewState, playerId: PlayerId): PlayerSnapshot[] {
  const sailors = state.room?.voyage?.sailors ?? [];
  const ids = new Set(sailors.filter((s) => s.followingId === playerId).map((s) => s.playerId));
  return state.room?.players.filter((p) => ids.has(p.id)) ?? [];
}

export function selectScore(state: CrewState, playerId: PlayerId | null): ScoreEntry | null {
  if (!playerId) return null;
  const scores = state.outcome?.scores ?? state.room?.voyage?.scores ?? [];
  return scores.find((s) => s.playerId === playerId) ?? null;
}

/** `m:ss` for under an hour, `h:mm:ss` beyond; never negative. */
export function formatClock(ms: number): string {
  const total = Math.max(0, Math.ceil(ms / 1000));
  const h = Math.floor(total / 3600);
  const m = Math.floor((total % 3600) / 60);
  const s = total % 60;
  const ss = String(s).padStart(2, "0");
  return h > 0 ? `${h}:${String(m).padStart(2, "0")}:${ss}` : `${m}:${ss}`;
}

function toPublicPlayer(room: RoomSnapshot, player: PlayerSnapshot): PublicPlayer {
  const sailor = room.voyage?.sailors.find((s) => s.playerId === player.id);
  return {
    id: player.id,
    userId: player.id,
    username: player.name,
    avatarId: player.avatarId,
    isHost: player.isCaptain,
    isOnline: player.isConnected,
    isEliminated: sailor?.isEliminated ?? false,
    strikes: sailor?.strikes ?? 0,
    hintsUsed: sailor?.hintsUsed ?? 0,
    followingId: sailor?.followingId ?? null,
    lastSeenAt: new Date(player.joinedAt).toISOString(),
    currentIslandKey: sailor?.currentIslandKey ?? null,
    ...(player.outfit ? { outfit: player.outfit } : {}),
  };
}

/** Flattens the latest `room:state` into the single view the voyage screen renders. */
export function toVoyageRoom(room: RoomSnapshot | null, you: InventoryItem[], hints: HintEntry[] = []): RoomState | null {
  const voyage = room?.voyage;
  if (!room || !voyage) return null;
  return {
    code: room.code,
    name: room.crewName,
    status: voyage.status === "finished" ? "FINISHED" : "ACTIVE",
    maxPlayers: room.maxPlayers,
    hostId: room.captainId,
    createdAt: new Date(room.createdAt).toISOString(),
    settings: room.settings,
    players: room.players.map((p) => toPublicPlayer(room, p)),
    islands: voyage.islands,
