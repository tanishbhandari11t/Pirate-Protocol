import type { PlayerId, PublicIsland, RoomState } from "../socket/contract";
import { chartPoint } from "./chart";
import { findPlayer, isVaultReady, islandByKey, solvedKeys, VAULT_KEY } from "./selectors";
import { compassPoint } from "./weather";
import { ISLAND_RELIC } from "./world";

/**
 * Where this sailor should sail next, worked out from public information only: island order and
 * kind, the sailor's own solves, and whether their hold carries every relic. It never knows an answer.
 */

export interface Destination {
  island: PublicIsland;
  /** Whether the crew has charted it; uncharted islands are pointed at but not named. */
  charted: boolean;
  /** Degrees clockwise from north, from the sailor's position (or the chart centre when adrift). */
  bearing: number;
  point: string;
  /** Chart units to sail. */
  distance: number;
  reason: "vault" | "relic" | "unsolved";
}

/** One league on the chart, for the spyglass's distance marks. */
export const CHART_LEAGUE = 40;

export function bearingBetween(from: { x: number; y: number }, to: { x: number; y: number }) {
  return ((Math.atan2(to.x - from.x, from.y - to.y) * 180) / Math.PI + 360) % 360;
}

export function nextDestination(room: RoomState, meId: PlayerId, charted: ReadonlySet<string>): Destination | null {
  const me = findPlayer(room, meId);
  if (!me || me.isEliminated || room.status === "FINISHED") return null;

  const solved = solvedKeys(room, meId);
  const unsolved = (i: PublicIsland) => i.puzzles.some((p) => !solved.has(p.key));
  const ordered = [...room.islands].filter((i) => i.key !== VAULT_KEY).sort((a, b) => a.order - b.order);

  let target: PublicIsland | null = null;
  let reason: Destination["reason"] = "unsolved";
  const vault = islandByKey(room, VAULT_KEY);
  if (vault && isVaultReady(room)) {
    target = vault;
    reason = "vault";
  } else {
    target = ordered.find((i) => ISLAND_RELIC[i.key] && unsolved(i)) ?? null;
    if (target) reason = "relic";
    else target = ordered.find((i) => i.kind !== "TRAP" && unsolved(i)) ?? null;
  }
  if (!target) return null;

  const here = islandByKey(room, me.currentIslandKey);
  const from = here ? chartPoint(here) : { x: 500, y: 300 };
  const to = chartPoint(target);
  const bearing = bearingBetween(from, to);
  return {
    island: target,
    charted: target.key === VAULT_KEY ? reason === "vault" : charted.has(target.key),
    bearing,
    point: compassPoint(bearing),
    distance: Math.hypot(to.x - from.x, to.y - from.y),
    reason,
  };
}

export function leagues(distance: number) {
  return Math.max(0, Math.round(distance / CHART_LEAGUE));
}

/** How the compass names a destination: the island's name if charted, otherwise only a direction. */
export function destinationLabel(dest: Destination) {
  if (dest.distance < 1) return `Here, at ${dest.charted ? dest.island.name : "this shore"}`;
  return dest.charted ? `${dest.island.name}, ${dest.point}` : `Uncharted waters to the ${dest.point}`;
}
