/**
 * Pirate Protocol — Socket.IO wire contract (Phase 1: crews & lobby).
 *
 * Every client → server event is a request that MUST be acknowledged with an `Ack<T>`.
 * Every server → client event is a push with a single payload object.
 * The server is authoritative: the client never decides whether an answer is right, whether
 * a trap fired, what a hint says, what anyone scored, or who won. It renders what it is sent.
 */

export interface ShipFlag {
  field: FlagField;
  emblem: FlagEmblem;
  border: FlagBorder;
}

/** `null` in a slot keeps whatever the base likeness wears there. */
export interface Outfit {
  hat: OutfitHat | null;
  coat: OutfitCoat | null;
  face: OutfitFace | null;
  trinket: OutfitTrinket | null;
  frame: AvatarFrame | null;
  flag: ShipFlag;
}

export const DEFAULT_FLAG: ShipFlag = { field: "sable", emblem: "skull", border: "plain" };

export const DEFAULT_OUTFIT: Outfit = {
  hat: null,
  coat: null,
  face: null,
  trinket: null,
  frame: null,
  flag: DEFAULT_FLAG,
};

function oneOf<T extends string>(list: readonly T[], value: unknown): value is T {
  return typeof value === "string" && (list as readonly string[]).includes(value);
}

function slot<T extends string>(list: readonly T[], value: unknown): T | null | undefined {
  if (value === null || value === undefined) return null;
  return oneOf(list, value) ? value : undefined;
}

/**
 * Returns a clean copy of an outfit, or `null` if any piece is unknown. Extra keys are dropped.
 * The mock server runs the same rules, so a valid outfit here is accepted there.
 */
export function sanitizeOutfit(value: unknown): Outfit | null {
  if (!value || typeof value !== "object") return null;
  const v = value as Record<string, unknown>;
  const flag = v.flag as Record<string, unknown> | undefined;
  if (!flag || typeof flag !== "object") return null;
  if (!oneOf(FLAG_FIELDS, flag.field) || !oneOf(FLAG_EMBLEMS, flag.emblem) || !oneOf(FLAG_BORDERS, flag.border)) {
    return null;
  }
  const hat = slot(OUTFIT_HATS, v.hat);
  const coat = slot(OUTFIT_COATS, v.coat);
  const face = slot(OUTFIT_FACES, v.face);
  const trinket = slot(OUTFIT_TRINKETS, v.trinket);
  const frame = slot(AVATAR_FRAMES, v.frame);
  if (hat === undefined || coat === undefined || face === undefined || trinket === undefined || frame === undefined) {
    return null;
  }
  return { hat, coat, face, trinket, frame, flag: { field: flag.field, emblem: flag.emblem, border: flag.border } };
}

export function outfitsEqual(a: Outfit, b: Outfit) {
  return (
    a.hat === b.hat &&
    a.coat === b.coat &&
    a.face === b.face &&
    a.trinket === b.trinket &&
    a.frame === b.frame &&
    a.flag.field === b.flag.field &&
    a.flag.emblem === b.flag.emblem &&
    a.flag.border === b.flag.border
  );
}

/* Voyage settings — the ship's articles the captain signs             */
/* ------------------------------------------------------------------ */

export const STRIKE_LIMIT_MIN = 1;
export const STRIKE_LIMIT_MAX = 5;
/** Minutes; `0` means the voyage runs until the Vault opens or the crew is wrecked. */
export const TIME_LIMIT_OPTIONS = [0, 10, 15, 20, 30] as const;
export type TimeLimitMinutes = (typeof TIME_LIMIT_OPTIONS)[number];

export interface VoyageSettings {
  /** Strikes a sailor may take before elimination. */
  maxStrikes: number;
  /** When off, a wrong answer on a trap island counts as an ordinary miss. */
  trapsEnabled: boolean;
  /** When off, `puzzle:hint` is refused for the whole voyage. */
  hintsEnabled: boolean;
  timeLimitMinutes: TimeLimitMinutes;
}

export const DEFAULT_SETTINGS: VoyageSettings = {
  maxStrikes: MAX_STRIKES,
  trapsEnabled: true,
  hintsEnabled: true,
  timeLimitMinutes: 0,
};

export function isTimeLimit(value: unknown): value is TimeLimitMinutes {
  return typeof value === "number" && (TIME_LIMIT_OPTIONS as readonly number[]).includes(value);
}

/**
 * Merges a partial update onto the current articles, or returns `null` if any field is invalid.
 * The client uses it to grey out bad choices; the server runs the same rules again.
 */
export function mergeSettings(current: VoyageSettings, patch: Partial<VoyageSettings>): VoyageSettings | null {
  const next = { ...current, ...patch };
  if (!Number.isInteger(next.maxStrikes)) return null;
  if (next.maxStrikes < STRIKE_LIMIT_MIN || next.maxStrikes > STRIKE_LIMIT_MAX) return null;
  if (typeof next.trapsEnabled !== "boolean" || typeof next.hintsEnabled !== "boolean") return null;
  if (!isTimeLimit(next.timeLimitMinutes)) return null;
  return next;
}

export function settingsEqual(a: VoyageSettings, b: VoyageSettings) {
  return (
    a.maxStrikes === b.maxStrikes &&
    a.trapsEnabled === b.trapsEnabled &&
    a.hintsEnabled === b.hintsEnabled &&
    a.timeLimitMinutes === b.timeLimitMinutes
  );
}

/**
 * One entry in the voyage log. `payload` is loosely typed so older clients survive new event
 * types; use `isGameEvent` to narrow it to a known shape.
 */
export interface GameEvent {
  id: string;
  /** Increases by one per event in a voyage; a gap means the client missed something. */
  seq: number;
  type: GameEventType | (string & {});
  payload: Record<string, unknown>;
  playerId: PlayerId | null;
  /** ISO timestamp. */
  createdAt: string;
}

export type TypedGameEvent<T extends GameEventType> = Omit<GameEvent, "type" | "payload"> & {
  type: T;
  payload: GameEventPayloads[T];
};

export function isGameEvent<T extends GameEventType>(event: GameEvent, type: T): event is TypedGameEvent<T> {
  return event.type === type;
}

  "room:player-left": { playerId: PlayerId; name: string; reason: LeaveReason };
  "room:player-updated": { player: PlayerSnapshot };
  "room:captain-changed": { captainId: PlayerId };
  /** The captain rewrote the articles; every sailor's ready flag was cleared. */
  "room:settings-changed": { settings: VoyageSettings; changedBy: PlayerId };
  /** Captain launched the voyage; clients show a countdown until `startsAt` (epoch ms). */
  "game:starting": { startsAt: number; seconds: number };
  "game:started": { roomCode: RoomCode; startedAt: number; endsAt: number | null };
  /** One thing that just happened on the voyage. Always sent after the `room:state` it caused. */
  "game:event": GameEvent & { roomCode: RoomCode };
  "game:finished": {
    roomCode: RoomCode;
    winnerPlayerId: PlayerId | null;
    finishedAt: number;
    reason: FinishReason;
    scores: ScoreEntry[];
  };
  "chat:message": ChatMessage & { roomCode: RoomCode };
  "server:notice": { level: NoticeLevel; message: string };
}

export type ServerEventName = keyof ServerEvents;
