export function describeError(code: ClientErrorCode, fallback?: string): string {
  return PIRATE_MESSAGES[code] ?? fallback ?? PIRATE_MESSAGES.INTERNAL;
}

/** Transport hiccups worth retrying automatically for requests that are safe to repeat. */
const RETRYABLE: ReadonlySet<ClientErrorCode> = new Set(["OFFLINE", "TIMEOUT", "INTERNAL"]);

export function isRetryable(code: ClientErrorCode) {
  return RETRYABLE.has(code);
}

/** Errors meaning this tab no longer holds a seat, so local state must be dropped. */
const SEAT_LOST: ReadonlySet<ClientErrorCode> = new Set(["SESSION_EXPIRED", "NOT_IN_ROOM", "ROOM_NOT_FOUND"]);

export function isSeatLost(code: ClientErrorCode) {
  return SEAT_LOST.has(code);
}

/** Mistakes the player can fix by acting differently, as opposed to things going wrong. */
const PLAYER_MISTAKES: ReadonlySet<ClientErrorCode> = new Set([
  "INVALID_PAYLOAD",
  "NAME_TAKEN",
  "NOT_ON_ISLAND",
  "ALREADY_SOLVED",
  "MISSING_RELICS",
  "NOT_TRADABLE",
  "ITEM_NOT_HELD",
  "BAD_RECIPIENT",
  "ISLAND_HIDDEN",
  "CANNOT_AFFORD",
  "NO_HINTS_LEFT",
  "RATE_LIMITED",
]);

export type ErrorTone = "warning" | "danger";

/** Gentle amber for player mistakes, red for everything that actually went wrong. */
export function errorTone(code: ClientErrorCode): ErrorTone {
  return PLAYER_MISTAKES.has(code) ? "warning" : "danger";
}

/** A short title for a toast about this error. */
export function errorTitle(code: ClientErrorCode): string {
  switch (code) {
    case "OFFLINE":
    case "TIMEOUT":
      return "Signal lost";
    case "SESSION_EXPIRED":
    case "NOT_IN_ROOM":
    case "ROOM_NOT_FOUND":
      return "Lost at sea";
    case "NOT_CAPTAIN":
      return "Not your order to give";
    case "ELIMINATED":
      return "Claimed by the sea";
    case "VOYAGE_OVER":
      return "The voyage is over";
    case "RATE_LIMITED":
      return "Hold fast";
    default:
      return PLAYER_MISTAKES.has(code) ? "Not quite, sailor" : "Rough waters";
  }
}
