"use client";

import { useSyncExternalStore } from "react";
import type { AvatarId, RoomCode } from "./socket/contract";

type StorageKind = "local" | "session";

function createStorageStore<T>(kind: StorageKind, key: string) {
  const listeners = new Set<() => void>();
  let cachedRaw: string | null | undefined;
  let cachedValue: T | null = null;

  const storage = () => {
    if (typeof window === "undefined") return null;
    try {
      return kind === "local" ? window.localStorage : window.sessionStorage;
    } catch {
      return null;
    }
  };

  const read = (): T | null => {
    const raw = storage()?.getItem(key) ?? null;
    if (raw === cachedRaw) return cachedValue;
    cachedRaw = raw;
    try {
      cachedValue = raw ? (JSON.parse(raw) as T) : null;
    } catch {
      cachedValue = null;
    }
    return cachedValue;
  };

  const emit = () => listeners.forEach((l) => l());

  return {
    get: read,
    set(value: T) {
      storage()?.setItem(key, JSON.stringify(value));
      emit();
    },
    clear() {
      storage()?.removeItem(key);
      emit();
    },
    subscribe(listener: () => void) {
      listeners.add(listener);
      const onStorage = (e: StorageEvent) => e.key === key && listener();
      window.addEventListener("storage", onStorage);
      return () => {
        listeners.delete(listener);
        window.removeEventListener("storage", onStorage);
      };
    },
    getSnapshot: read,
    getServerSnapshot: (): T | null => null,
  };
}

export interface SeatRecord {
  roomCode: RoomCode;
  playerId: string;
  sessionToken: string;
}

export interface PirateProfile {
  playerName: string;
  avatarId: AvatarId;
}

/** Per-tab seat so a refresh can reclaim the same berth. */
export const seatStore = createStorageStore<SeatRecord>("session", "pp:seat");

/** Remembered identity used to prefill the crew forms. */
export const profileStore = createStorageStore<PirateProfile>("local", "pp:profile");

export function useProfile() {
  return useSyncExternalStore(
    profileStore.subscribe,
    profileStore.getSnapshot,
    profileStore.getServerSnapshot,
  );
}

/** The outfit this device last dressed in, carried into every crew it creates or joins. */
export const outfitStore = createStorageStore<Outfit>("local", "pp:outfit");

/** The saved outfit, falling back to plain dress if nothing (or something stale) is stored. */
export function savedOutfit(): Outfit {
  return sanitizeOutfit(outfitStore.get()) ?? DEFAULT_OUTFIT;
}

export function useSavedOutfit(): Outfit {
  const stored = useSyncExternalStore(outfitStore.subscribe, outfitStore.getSnapshot, outfitStore.getServerSnapshot);
  return useMemo(() => (stored && sanitizeOutfit(stored)) ?? DEFAULT_OUTFIT, [stored]);
}
