import {
  CREW_NAME_MAX,
  ROOM_CODE_ALPHABET,
  ROOM_CODE_LENGTH,
  USERNAME_MAX,
  USERNAME_MIN,
  USERNAME_PATTERN,
} from "./socket/contract";
import { BACKEND } from "./socket/backend";
import { NEST_USERNAME_MIN, toNestUsername } from "./socket/nest/translate";

export function normalizeName(value: string) {
  return value.replace(/\s+/g, " ").trim();
}

/** The server stores usernames trimmed and lowercased. */
export function normalizeUsername(value: string) {
  return value.trim().toLowerCase();
}

/** Shapes keystrokes into the server's username alphabet as the sailor types. */
export function typeUsername(value: string) {
  return value
    .toLowerCase()
    .replace(/\s+/g, "_")
    .replace(/[^a-z0-9_]/g, "")
    .slice(0, USERNAME_MAX);
}

export function validateUsername(value: string): string | null {
  const name = normalizeUsername(value);
  if (name.length < USERNAME_MIN) return `Every pirate needs a name of at least ${USERNAME_MIN} letters.`;
  if (name.length > USERNAME_MAX) return `Keep it under ${USERNAME_MAX} letters, sailor.`;
  if (!USERNAME_PATTERN.test(name)) return "Only letters, numbers and underscores — no spaces.";
  return null;
}

export function validateCrewName(value: string): string | null {
  const name = normalizeName(value);
  if (name.length < 3) return "A crew name needs at least 3 letters.";
  if (name.length > CREW_NAME_MAX) return `Keep it under ${CREW_NAME_MAX} letters.`;
  return null;
}

export function sanitizeRoomCode(value: string) {
  const allowed = new Set(ROOM_CODE_ALPHABET);
  return value
    .toUpperCase()
    .split("")
    .filter((c) => allowed.has(c))
    .join("")
    .slice(0, ROOM_CODE_LENGTH);
}

export function validateRoomCode(value: string): string | null {
  if (sanitizeRoomCode(value).length !== ROOM_CODE_LENGTH) {
    return `The code has ${ROOM_CODE_LENGTH} runes.`;
  }
  return null;
}
